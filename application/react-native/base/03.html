3.2.5 TextInput组件

  1. 代码

    import { StyleSheet, View,TextInput,Dimensions,Button, Alert } from 'react-native'
import React, { useState } from 'react'

function doLogin(text:{username:string,password:string}) {
    Alert.alert('账户信息',`用户名：${text.username}\n密码：${text.password}`)
}

export default function Index() {

    const [text,setText] = useState({username:'',password:''})
    return (
        <View style={[styles.container]}>
            <TextInput
                style={[styles.input]}
                placeholder='请输入用户名'
                value={text.username}
                onChangeText={val=>setText(prev=>({...prev,username:val}))}
            />
            <TextInput
                style={[styles.input]}
                placeholder='请输入密码'
                secureTextEntry={true}  // 密码输入框
                value={text.password}
                onChangeText={val=>setText(prev=>({...prev,password:val}))}
            />
            <TextInput
                style={[styles.input]}
                placeholder='手机号'
                keyboardType='numeric'  // 数字键盘
                maxLength={13}          // 最大长度
            />
            <TextInput
                style={[styles.input,{height:100}]} // 必须设置高度，否则无法显示多行
                placeholder='请输入自我介绍'
                editable
                multiline               // 多行输入
                numberOfLines={5}      // 显示行数
                textAlignVertical='top' // 多行输入时，文字在顶部显示，ios默认顶部，安卓默认第二行
            />
            <View style={[styles.btn]}>
                <Button  
                    title='登录' 
                    onPress={()=>{doLogin(text)}}
                />
            </View>
        </View>
    )
}

3.2.6 Touchable 组件

  1. 介绍

    • TouchableHighlight 触碰后，高亮显示
    • TouchableOpactity 触碰后，透明度降低
    • TouchableWithoutFeedback 触碰后，无任何响应

  2. 代码示例

    import {
    StyleSheet,
    Text,
    View,
    TouchableHighlight,
    TouchableOpacity,
    TouchableWithoutFeedback,
} from 'react-native';
import React from 'react';

export default function index() {
    return (
        <View style={[styles.container]}>
            <TouchableHighlight onPress={() => console.log('触碰高亮显示')}>
                <View style={[styles.item]}>
                    <Text>触碰高亮</Text>
                </View>
            </TouchableHighlight>
            <TouchableOpacity onPress={() => console.log('触碰透明度变化')}>
                <View style={[styles.item]}>

3.2.8 SelectList组件 (重点)

  1. 代码示例

    import { useState } from 'react';
import {
    StyleSheet,
    Text,
    View,
    SectionList,
    StatusBar,
    Alert,
} from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';

const DATA = [
    {
        title: '魏国',
        data: ['曹操', '司马懿', '张辽'],
    },
    {
        title: '蜀国',
        data: ['刘备', '关羽', '张飞'],
    },
    {
        title: '吴国',
        data: ['孙权', '周瑜', '吕蒙'],
    },
    {
        title: '群雄',
        data: ['张角', '袁绍', '董卓'],
    },
];


export default function Index() {
    // 开启加载动画
    const [isFresh, setIsFresh] = useState(false);
    return (
        <SafeAreaView style={styles.container}>
        <SectionList
            sections={DATA} // 数据源
            keyExtractor={(item, index) => item + index} // key
            // 每一行的渲染
            renderItem={({ item }) => (
                <View style={styles.item}>
                    <Text style={styles.title}>{item}</Text>
                </View>
            )}
            // 每个分组头的渲染
            renderSectionHeader={({ section: { title } }) => (
                <Text style={styles.header}>{title}</Text>
            )}
            // 每一行的分割线
            ItemSeparatorComponent={() => (
                <View style={{ borderBottomWidth: 1, backgroundColor: 'red' }} />
            )}
            // 列表数据为空时展示的组件
            ListEmptyComponent={() => {
                return <Text style={{ fontSize: 30 }}>空空如也！</Text>;
            }}
            // 下拉刷新
            refreshing={isFresh} // 是否正在刷新
            onRefresh={()=>{
                setIsFresh(true); // 开启加载动画

                setTimeout(() => {
                    setIsFresh(false); // 关闭加载动画
                    Alert.alert('刷新完成！');
                }, 2000);
            }}  // 下拉刷新触发的事件

            // 上拉刷新
            onEndReachedThreshold={0.1} // 距离底部多少触发上拉加载 0.1 代表10%
            onEndReached={()=>{
                Alert.alert('加载更多！');
            }}

            // 头部组件
            ListHeaderComponent={()=>{
                return <Text style={{fontSize:40}}>三国人物</Text>
            }}
            // 尾部组件
            ListFooterComponent={()=>{
                return <Text style={{fontSize:30, textAlign:'center'}}>我是有底线的！</Text>
            }}
        />
        </SafeAreaView>
    );
}

const styles = StyleSheet.create({
    container: {

        marginVertical: 8,

        <View style={styles.item}>
        <Text style={styles.title}>{item.title}</Text>

3.2.10 Animated

  1. RN中可以直接使用的动画组件

    • Animated.View
    • Animated.Text
    • Animated.ScrollView
    • Animated.Image

  2. 如何创建动画（步骤）

    • 创建初始值

      • Animated.Value() 单个值
      • Animated.ValueXY() 向量值

    • 将初始值绑定的动画组件上

      一般将其绑定到某个样式属性下，例如：opacity,translate

    • 通过动画类型API,一帧一帧地更改初始值

      • Animated.decay() 加速效果
      • Animated.spring() 弹跳效果
      • Animated.timing() 时间渐变效果

  3. 示例1：淡变展示

    • 代码

      import React, { useRef } from 'react';
import { SafeAreaView } from 'react-native-safe-area-context';
import { Animated, Text, View, StyleSheet, Button } from 'react-native';

export default function Index() {
    // fadeAnim将用作不透明度的值。初始值：0
    const fadeAnim = useRef(new Animated.Value(0)).current;

    const fadeIn = () => {
        console.log('fadeIn - start');

        // 将在5秒内将fadeAnim值更改为1
        Animated.timing(fadeAnim, {
            toValue: 1, // 目标值
            duration: 5000, // 动画时间
            useNativeDriver: true, // 使用原生方式渲染动画
        }).start(() => {
            console.log('fadeIn - end');
        }); // 开始执行动画
    };

    const fadeOut = () => {
        // 将在3秒内将fadeAnim值更改为0
        Animated.timing(fadeAnim, {
            toValue: 0,
            duration: 3000,
            useNativeDriver: true,
        }).start();
    };

    return (
        <SafeAreaView style={styles.container}>
            <Animated.View
                style={[
                styles.fadingContainer,
                {
                    // 将不透明度绑定到动画值
                    opacity: fadeAnim,
                },
                ]}
            >
                <Text style={styles.fadingText}>正在淡变地视图!</Text>
            </Animated.View>
            <View style={styles.buttonRow}>
                <Button title="淡变显示" onPress={fadeIn} />
                <Button title="淡变隐藏" onPress={fadeOut} />
            </View>
        </SafeAreaView>
    );
}

const styles = StyleSheet.create({
    container: {
        height: '100%',
        flex: 1,
        alignItems: 'center',

封装 AsyncStorage

  1. 组件使用

    • 增 set(key,value)
    • 删 delete(key) | clear()
    • 改 update(key,value) 主要考虑value 是对象
    • 查 get(key)

  2. 代码示例

    • storage.ts

      import AsyncStorage from '@react-native-async-storage/async-storage';

class Storage{
    // 添加数据
    static set(key:string,value: number | string | object ){
        return AsyncStorage.setItem(key,JSON.stringify(value))
    }

    // 获取数据
    static get(key:string){
        return AsyncStorage.getItem(key).then(value=>{
            if (value && value !== '') {
                const jsonValue = JSON.parse(value)
                return jsonValue;
            }
        }).catch(()=>null)
    }

    // 更新数据
    static update(key:string,newValue: number | string | object){
        return AsyncStorage.getItem(key).then(oldValue =>{
            newValue = typeof newValue === 'string' ? newValue : Object.assign({},oldValue,newValue);
            return AsyncStorage.setItem(key,JSON.stringify(newValue))
        })
    }

    // 删除指定的key
    static delete(key:string){
        return AsyncStorage.removeItem(key);
    }

    // 清空所有数据 
    static clear(){
        return AsyncStorage.clear()
    }
}

export default Storage

  1. 组件使用

  2. 代码示例

    import React, { useState } from 'react';
import {
    View,
    Text,
    TouchableOpacity,
    StyleSheet,
    Image,
    Alert,
    ActivityIndicator,
} from 'react-native';
import {
    launchCamera,
    launchImageLibrary,
    CameraOptions,
    ImageLibraryOptions,
    ImagePickerResponse,
} from 'react-native-image-picker';

interface AvatarPickerProps {
    onAvatarSelected?: (avatarUri: string) => void;
    initialAvatar?: string;
    size?: number;
}

const AvatarPicker: React.FC<AvatarPickerProps> = ({
    onAvatarSelected,
    initialAvatar,
    size = 120,
}) => {
    const [avatarUri, setAvatarUri] = useState<string | null>(initialAvatar || null);
    const [loading, setLoading] = useState<boolean>(false);

    // 头像配置 - 专注于头像使用的优化配置
    const cameraOptions: CameraOptions = {
        mediaType: 'photo',
        includeBase64: false,
        maxHeight: 800,  // 适合头像的尺寸
        maxWidth: 800,
        quality: 0.7,    // 适中的质量
        saveToPhotos: false, // 头像不需要保存到相册
        cameraType: 'front', // 头像通常用前置摄像头
    };

    const libraryOptions: ImageLibraryOptions = {
        mediaType: 'photo',
        includeBase64: false,
        maxHeight: 800,
        maxWidth: 800,
        quality: 0.7,
        selectionLimit: 1, // 头像只能选一张
    };

    // 处理选择结果
    const handleResponse = (response: ImagePickerResponse) => {
        setLoading(false);

        if (response.didCancel) {
            return;
        }

        if (response.errorCode) {
            let errorMessage = '选择失败';
            
            switch (response.errorCode) {
                case 'camera_unavailable':
                errorMessage = '相机不可用';
                break;
                case 'permission':
                errorMessage = '没有相机或相册权限';
                break;
            }

            Alert.alert('错误', errorMessage);
            return;
        }

        if (response.assets && response.assets.length > 0) {
            const asset = response.assets[0];
            if (asset.uri) {
                setAvatarUri(asset.uri);
                onAvatarSelected?.(asset.uri);
                Alert.alert('成功', '头像已更新');
            }
        }
    };

    // 打开相机拍照
    const takePhoto = () => {
        setLoading(true);
        launchCamera(cameraOptions, handleResponse);
    };

    // 打开相册选择
    const chooseFromGallery = () => {
        setLoading(true);
        launchImageLibrary(libraryOptions, handleResponse);
    };

    // 删除头像
    const removeAvatar = () => {
        Alert.alert(
            '删除头像',
            '确定要删除当前头像吗？',
            [
                { text: '取消', style: 'cancel' },
                { 
                    text: '删除', 
                    style: 'destructive',
                    onPress: () => {
                        setAvatarUri(null);
                        onAvatarSelected?.('');
                    }
                },
            ]
        );
    };

    // 显示选择选项
    const showAvatarOptions = () => {
        Alert.alert(
            '选择头像',
            '请选择获取头像的方式',
            [
                { text: '取消', style: 'cancel' },
                { text: '拍照', onPress: takePhoto },
                { text: '从相册选择', onPress: chooseFromGallery },
                avatarUri ? { text: '删除头像', style: 'destructive', onPress: removeAvatar } : null,
            ].filter(Boolean) as any
        );
    };

    const avatarSize = {
        width: size,
        height: size,
        borderRadius: size / 2,
    };

    return (
        <View style={styles.container}>
            <TouchableOpacity 
                style={[styles.avatarContainer, avatarSize]}
                onPress={showAvatarOptions}
                disabled={loading}
            >
                {loading ? (
                    <View style={[styles.loadingContainer, avatarSize]}>
                        <ActivityIndicator size="small" color="#007AFF" />
                    </View>
                    ) : avatarUri ? (
                    <Image 
                        source={{ uri: avatarUri }} 
                        style={[styles.avatarImage, avatarSize]}
                        resizeMode="cover"
                    />
                    ) : (
                    <View style={[styles.placeholder, avatarSize]}>
                        <Text style={styles.placeholderText}>点击选择头像</Text>
                    </View>
                )}
                
                {/* 编辑图标 */}
                <View style={styles.editIcon}>
                    <Text style={styles.editIconText}>✎</Text>
                </View>
            </TouchableOpacity>

            <Text style={styles.hintText}>
                点击头像可以拍照或从相册选择
            </Text>
        </View>
    );
};

const styles = StyleSheet.create({
    container: {
        alignItems: 'center',
        padding: 20,
    },
    avatarContainer: {
        backgroundColor: '#f0f0f0',
        justifyContent: 'center',
        alignItems: 'center',
        borderWidth: 3,
        borderColor: '#e0e0e0',
        position: 'relative',
        shadowColor: '#000',
        shadowOffset: {
        width: 0,
        height: 2,
        },
        shadowOpacity: 0.1,
        shadowRadius: 3.84,
        elevation: 5,
    },
    avatarImage: {

    },
    placeholder: {
        backgroundColor: '#e0e0e0',
