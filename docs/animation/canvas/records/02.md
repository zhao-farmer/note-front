# 日常使用图案总结

## 2.1 PS透明度背景图

1. 代码

    ```html
    <!DOCTYPE html>
    <html lang="zh-CN">

    <head>
      <meta charset="UTF-8">
      <title>纯Canvas棋盘格</title>
    </head>

    <body>
      <canvas id="b_canvas"></canvas>
      <script>
        (() => {
          // 背景 canvas
          let bCanvas = document.querySelector("#b_canvas");
          let bCtx = bCanvas.getContext('2d');

          // 画布尺寸（可改）
          let initialWidth = 800;
          let initialHeight = 300;

          bCanvas.width = initialWidth;
          bCanvas.height = initialHeight;

          // 小格子大小 8x8
          const w = 8;
          const h = 8;

          // 直接在 canvas 绘制！
          let wflag, hflag;
          for (let i = 0; i < Math.ceil(initialWidth / w); i++) {
            for (let j = 0; j < Math.ceil(initialHeight / h); j++) {
              wflag = i % 2 === 0;
              hflag = j % 2 === 0;

              if ((wflag && hflag) || (!wflag && !hflag)) {
                bCtx.fillStyle = '#fff';
              } else {
                bCtx.fillStyle = '#b3b3b3';
              }

              bCtx.fillRect(i * w, j * h, w, h);
            }
          }
        })();

      </script>
    </body>
    </html>
    ```

## 2.2 处理音频功能波动

1. 代码

    ```html
    <!DOCTYPE html>
    <html lang="zh-CN">
    <head>
      <meta charset="UTF-8">
      <title>音频波形图 - 分段颜色</title>
      <style>
        .wave-box {
          width: 600px;
          height: 73px;
          position: relative;
          overflow: hidden;
        }
        canvas {
          display: block;
          width: 100%;
          height: 100%;
          background: #0a3d91;
        }
      </style>
    </head>
    <body>
      <div class="wave-box">
        <canvas id="waveCanvas"></canvas>
      </div>

      <script>
        const AUDIO_URL = "./audio/01.mp3";

        const canvas = document.getElementById("waveCanvas");
        const ctx = canvas.getContext("2d");
        let audioContext;
        let audioBuffer;

        async function loadAudio() {
          try {
            const res = await fetch(AUDIO_URL);
            const buffer = await res.arrayBuffer();
            audioContext = new (window.AudioContext || window.webkitAudioContext)();
            audioBuffer = await audioContext.decodeAudioData(buffer);
            drawWave();
          } catch (err) {
            console.error("音频加载失败", err);
          }
        }

        function drawWave() {
          const width = canvas.width = 600;
          const height = canvas.height = 73;

          // 背景
          ctx.fillStyle = "#0a3d91";
          ctx.fillRect(0, 0, width, height);

          const data = audioBuffer.getChannelData(0);
          const step = Math.ceil(data.length / width);

          // 分界线：高度的 5/6（超过这部分就变橘色）
          const splitHeight = height * 0.85 * 5 / 6;

          for (let x = 0; x < width; x++) {
            let max = 0;
            for (let j = 0; j < step; j++) {
              const v = Math.abs(data[x * step + j]);
              if (v > max) max = v;
            }

            // 总高度
            const totalH = max * height * 0.85;

            // ----------------------
            // 第一段：蓝色（下半段）
            // ----------------------
            const blueH = Math.min(totalH, splitHeight);
            ctx.fillStyle = "#00b4ff";
            ctx.fillRect(x, height - blueH, 1, blueH);

            // ----------------------
            // 第二段：橘红色（超过的上半段）
            // ----------------------
            if (totalH > splitHeight) {
              const orangeH = totalH - splitHeight;
              ctx.fillStyle = "#ff5a00";
              ctx.fillRect(x, height - totalH, 1, orangeH);
            }
          }
        }

        loadAudio();
      </script>
    </body>
    </html>
    ```

## 2.3 斜角线图案展示

1. 代码

    ```html
    <!DOCTYPE html>
    <html lang="zh-CN">

    <head>
      <meta charset="UTF-8">
    </head>

    <body>

      <canvas id="demoCanvas" width="800" height="200"></canvas>

      <script>
        const canvas = document.getElementById('demoCanvas');
        const ctx = canvas.getContext('2d');

        // 生成纹理：对角线第一条线 → 向两侧平行偏移
        function createSolidLineTile(size) {
  
          const tile = document.createElement('canvas');
          const tCtx = tile.getContext('2d');

          tile.width = size;
          tile.height = size;

          // 背景
          tCtx.fillStyle = '#333';
          tCtx.fillRect(0, 0, size, size);

          // 线条样式
          tCtx.strokeStyle = '#fff';
          tCtx.lineWidth = 2;
          const spacing = 10; // 平行线间距 要是size能整除的数

          tCtx.beginPath();

          // 1. 先画【第一条线：主对角线】（最长线）
          tCtx.moveTo(0, 0);
          tCtx.lineTo(size, size);

          // 2. 向【左上】位移画线
          for (let offset = -spacing; offset > -size; offset -= spacing) {
            tCtx.moveTo(0 + offset, 0);
            tCtx.lineTo(size + offset, size);
          }

          // 3. 向【右下】位移画线
          for (let offset = spacing; offset < size; offset += spacing) {
            tCtx.moveTo(0 + offset, 0);
            tCtx.lineTo(size + offset, size);
          }

          tCtx.stroke();
          return tile;
        }

        // 填充重复纹理
        const tile = createSolidLineTile(200);
        const pattern = ctx.createPattern(tile, 'repeat');

        ctx.fillStyle = pattern;
        ctx.fillRect(0, 0, canvas.width, canvas.height);
      </script>

    </body>

    </html>
    ```
