## 1.1 功能按钮

1.  **`toolbox` 工具箱（功能按钮）**

    ```js
    toolbox: {
        feature: {
            saveAsImage: {},    // 保存图片，导出图表png
            dataView: {},       // 数据视图：弹窗查看/编辑原始数据
            restore: {},        // 重置：恢复图表到初始状态
            dataZoom: {},       // 区域缩放：框选放大图表视图
            magicType: {        // 动态切换图表类型
                type: ["bar", "line"],
            },
        },
    },
    ```

## 1.2 框选功能-brush

1. 使用brush

    `brush`：画出半透明选框，拿到**选中区域的数据 / 坐标**，用于筛选、统计，适合折线 / 散点

    ```js
    brush: {
      // 启用横向刷选（仅X轴区间选择，折线时序场景首选）
      toolbox: ["lineX", "clear"],
      xAxisIndex: 0, // 绑定第0个x轴
      brushStyle: {
        borderWidth: 1,
        color: "rgba(120,140,180,0.2)",
        borderColor: "rgba(120,140,180,0.8)",
      },
      transformable: true, // 选区支持拖拽、缩放
      brushLink: 'all' // 多系列折线，全部联动刷选
    },
    ```

2. 设置toolbox
    
    配套的 toolbox 刷选按钮配置（必须配合，不然工具栏没有框选按钮）
    
    ```js
    toolbox: {
      show: true,
      feature: {
        brush: {
          show: true,
          title: {
            lineX: "框选计算",
            clear: "关闭框选",
          },
        }
      },
      top: 0,
      right: 15
    },
    ```

3. 事件监听（单独的JS，用来捕获框选结果）
    
    ```js
    // 框选完成（鼠标松开后触发，推荐）
    myChart.on("brushEnd", function (params) {
      if (!params.areas || params.areas.length === 0) return;
      const area = params.areas[0];
      // area.range[0][0] 起始索引，area.range[0][1]结束索引
      const startIndex = area.range[0][0];
      const endIndex = area.range[0][1];
      console.log('选中索引区间：', startIndex, endIndex);
    });
    ```

4. 参数简要说明
    1. `toolbox: ["lineX"]`：横向框选（只沿X轴拉选区，适合时间折线）；换成`rect`就是矩形框选
    2. `xAxisIndex:0`：限定刷选作用在x轴，y方向不参与选择
    3. `brushLink:'all'`：多条折线一起参与框选，你双折线场景必加
    4. `transformable:true`：画完选区之后，可以拖动选区、拉伸选区大小

5. 具体示例

    ```html
    <!DOCTYPE html>
    <html lang="en">
    <head>
        <meta charset="UTF-8" />
        <title>Document</title>
        <script src="../js/echarts-5.5.js"></script>
    </head>
    <body>
        <div style="width: 800px; height: 600px" id="chart"></div>
        <div id="brushData"></div>
        <script>
            // 初始化 ECharts 实例
            const myChart = echarts.init(document.getElementById("chart"));
            const brushData = document.getElementById("brushData");

            let xData = ["2022-05-01", "2022-05-02", "2022-05-03", "2022-05-04", "2022-05-05", "2022-05-06",    "2022-05-07"]; // x轴数据
            option = {
                toolbox: {
                    //框选计算工具栏
                    show: true,
                    feature: {
                        brush: {
                            show: true,
                            title: {
                                lineX: "框选计算",
                                clear: "关闭框选",
                            },
                        },
                        saveAsImage: { show: true },
                    },
                    top: 0,
                    right: 15,
                    itemSize: 20, //工具栏 icon 的大小
                },
                brush: {
                    toolbox: ["lineX", "clear"],
                    xAxisIndex: 0,
                    brushStyle: {
                        borderWidth: 1,
                        color: "rgba(120,140,180,0.2)",
                        borderColor: "rgba(120,140,180,0.8)",
                    },
                },
                xAxis: {
                    type: "category",
                    data: xData,
                },
                yAxis: {
                    type: "value",
                },
                series: [
                    {
                        name: "销售量",
                        type: "line",
                        data: [150, 230, 224, 218, 135, 147, 260],
                    },
                ],
            };

            // 点击框选 brushEnd、brushSelected
            myChart.on("brushSelected", function (params) {
                console.log(params);
                if (params.batch.length < 1 || params.batch[0].areas.length < 1) {
                    brushData.innerHTML = "";
                    return;
                }
                var range = params.batch[0].areas[0].coordRange; //根据坐标获取选中区域数据
                range[range.length - 1] = range[range.length - 1] + 1;
                var brushed = [],
                    nameData = [],
                    sumData = [],
                    meanData = [];
                var brushComponent = params.batch[0];
                for (var sIdx = 0; sIdx < brushComponent.selected.length; sIdx++) {
                    var name = brushComponent.selected[sIdx].seriesName;
                    var values = myChart.getOption().series[sIdx].data.slice(...range);
                    var timeData = xData.slice(...range);
                    var timeRange = "";
                    if (values.length < 1 || timeData.length < 1) {
                        return;
                    }
                    if (timeData.length === 1) {
                        timeRange = timeData[0];
                    } else {
                        timeRange = timeData[0] + "至" + timeData[timeData.length - 1];
                    }
                    var sum = SumNum(values) ? SumNum(values) : 0;
                    var mean = AverageNum(values) ? AverageNum(values) : 0;
                    sumData.push(sum);
                    meanData.push(mean);
                    nameData.push(name);
                    var temp = `${name}：日均值：${mean.toFixed(0)}，总量：${sum}`;
                    brushed.push(temp);
                }
                var text = brushed.join("");
                var brushInfo = timeRange + "\n" + text;
                // $("#brushData").html(brushInfo);
                brushData.innerHTML = brushInfo;

                myChart.setOption({
                    title: {
                        backgroundColor: "blue",
                        text: brushInfo,
                        bottom: 0,
                        right: "10%",
                        width: 100,
                        textStyle: {
                            fontSize: 12,
                            color: "#fff",
                        },
                    },
                });
            });

            //均值
            function AverageNum(arr) {
                let sum = eval(arr.join("+"));
                let svg = ((sum / arr.length) * 100) / 100;
                return svg;
            }
            //求和
            function SumNum(arr) {
                let sum = eval(arr.join("+"));
                return sum;
            }

            myChart.setOption(option);
        </script>
    </body>
    </html>
    ```

## 1.3 十字准星

1. tooltip（悬浮提示框）

    ```js
    tooltip: {
      trigger: 'axis', // 按坐标轴触发，鼠标悬浮到轴，同时取同X轴所有系列数据
      axisPointer: {
        type: 'cross' // 十字准星：横向+纵向两条标线；可选 line(单线) / shadow(阴影) / none
      },
      backgroundColor: 'rgba(255, 255, 255, 0.8)', // 提示框背景（半透明白底）
      // 自定义提示框位置：在图表左右两侧自动切换，防止超出画布
      position: function (pos, params, el, elRect, size) {
        var obj = { top: 10 };
        obj[['left', 'right'][+(pos[0] < size.viewSize[0] / 2)]] = 30;
        return obj;
      },
      extraCssText: 'width: 170px' // 额外CSS，固定提示框宽度
    },
    ```

2. 全局 axisPointer（多图联动，适合你前面多grid上下双图表场景）

    ```js
    axisPointer: {
      link: { xAxisIndex: 'all' }, // 所有x轴的十字标线联动，鼠标移动，上下图表十字一起动
      label: {
        backgroundColor: '#777' // 坐标轴上数值标签背景色
      }
    },
    ```

    > 
    >  注意区分两处`axisPointer`
    > 
    > 
    > - `tooltip.axisPointer`：**当前图表内十字标线类型**，控制cross十字
    > - 顶层`axisPointer`：**全局坐标轴指示器配置**，主要用于多grid多图表之间标线联动，是独立配置

3. 常见踩坑点

    1. **`trigger:axis` 才会配合十字准星生效**，如果改成 `trigger:'item'`，只悬浮在数据点触发，十字准星表现不一样。
    2. **多grid（上下两个图表）**：`link:{xAxisIndex:'all'}` 就是专门用来做**跨图表十字同步**，正好适配你前面双grid需求。
    3. 十字准星样式自定义：在`tooltip.axisPointer`内增加样式
        
        ```js
        axisPointer: {
            type: 'cross',
            lineStyle: {
                color: '#999',
                width:1,
                type:'dashed'
            }
        }
        ```

    4. 顶层`axisPointer.label` 是X轴底部显示的刻度浮标，就是十字落在x轴上弹出的数值小标签。

4. 代码示例

    ```html
    <!DOCTYPE html>
    <html lang="zh-CN">
    <head>
      <meta charset="UTF-8">
      <title>十字准星 cross + tooltip</title>
      <script src="../js/echarts-5.5.js"></script>
    </head>
    <body>
      <div id="chart" style="width:800px;height:600px;"></div>
      <script>
        const myChart = echarts.init(document.getElementById('chart'));
        const option = {
          tooltip: {
            trigger: 'axis',
            axisPointer: {
              type: 'cross'
            },
            backgroundColor: 'rgba(255, 255, 255, 0.8)',
            position: function (pos, params, el, elRect, size) {
              var obj = { top: 10 };
              obj[['left', 'right'][+(pos[0] < size.viewSize[0] / 2)]] = 30;
              return obj;
            },
            extraCssText: 'width: 170px'
          },
          axisPointer: {
            link: { xAxisIndex: 'all' },
            label: {
              backgroundColor: '#777'
            }
          },
          grid: [
            {
              left: '80px',
              right: '80px',
              top: '60px',
              height: '300px'
            },
            {
              left: '80px',
              right: '80px',
              top: '380px',
              height: '180px'
            }
          ],
          xAxis: [
            {
              type: 'category',
              data: ['周一','周二','周三','周四','周五','周六','周日'],
              gridIndex:0
            },
            {
              type: 'category',
              data: ['周一','周二','周三','周四','周五','周六','周日'],
              gridIndex:1
            }
          ],
          yAxis:[
            {type:'value',gridIndex:0},
            {type:'value',gridIndex:1}
          ],
          series:[
            {
              name:'数据A',
              type:'line',
              xAxisIndex:0,
              yAxisIndex:0,
              data:[120,200,150,80,70,110,130]
            },
            {
              name:'数据B',
              type:'line',
              xAxisIndex:1,
              yAxisIndex:1,
              data:[20,40,30,15,10,28,35]
            }
          ]
        };
        myChart.setOption(option);
      </script>
    </body>
    </html>
    ```
