# 二、pixi属性补充

## 2.1 stroke属性

1. 参数表

    | 参数         | 类型           | 说明                       |
    |------------|--------------|--------------------------|
    | width      | number       | 线条宽度（必填）                 |
    | color      | number       | 线条颜色（十六进制，如 0xffffff）    |
    | alpha      | number       | 透明度 0 ~ 1                |
    | cap        | string       | 线端：round / butt / square |
    | join       | string       | 拐角：round / miter / bevel |
    | miterLimit | number       | 尖角限制（仅 join: 'miter'）    |
    | fill       | FillGradient | 渐变填充（优先级 > color）        |
    | texture    | Texture      | 线条纹理（图片 / 纯色纹理）          |

2. 代码展示

    ```html
    <!DOCTYPE html>
    <html lang="zh-CN">
    <head>
      <meta charset="UTF-8" />
      <title>PixiJS stroke 属性全演示</title>
    </head>
    <body>
    <script src="./js/pixi.js"></script>
    <script>
      async function init() {
        const app = new PIXI.Application();
        await app.init({
          width: 600,       // 调整画布宽度
          height: 400,      // 调整画布高度
          backgroundColor: 0x1a1a1a,
          antialias: true
        });
        document.body.appendChild(app.canvas);

        // ==============================================
        // 第一行：3个图形 自带的
        // ==============================================
        // 1. 线条宽度 width
        const g1 = new PIXI.Graphics();
        g1.x = 40;
        g1.y = 40;
        g1.rect(0, 0, 120, 60);
        g1.stroke({ width: 10, color: 0xffffff });
        app.stage.addChild(g1);

        // 2. 线条颜色 color
        const g2 = new PIXI.Graphics();
        g2.x = 220;
        g2.y = 40;
        g2.circle(60, 30, 30);
        g2.stroke({ width: 5, color: 0xff4444 });
        app.stage.addChild(g2);

        // 3. 透明度 alpha
        const g3 = new PIXI.Graphics();
        g3.x = 400;
        g3.y = 40;
        g3.rect(0, 0, 120, 60);
        g3.stroke({ width: 8, color: 0xffffff, alpha: 0.3 });
        app.stage.addChild(g3);

        // ==============================================
        // 第二行：3个图形 继承canvas
        // ==============================================
        // 4. 线端样式 cap
        const g4 = new PIXI.Graphics();
        g4.x = 40;
        g4.y = 140;
        g4.moveTo(10, 0);
        g4.lineTo(110, 0);
        g4.stroke({ width: 14, color: 0xffffff, cap: 'round' });
        app.stage.addChild(g4);

        // 5. 连接样式 join
        const g5 = new PIXI.Graphics();
        g5.x = 220;
        g5.y = 140;
        g5.moveTo(0, 0);
        g5.lineTo(100, 0);
        g5.lineTo(100, 60);
        g5.stroke({ width: 10, color: 0xffffff, join: 'round' });
        app.stage.addChild(g5);

        // 6. 尖角限制 miterLimit
        const g6 = new PIXI.Graphics();
        g6.x = 400;
        g6.y = 140;
        g6.moveTo(0, 40);
        g6.lineTo(60, 0);
        g6.lineTo(120, 40);
        g6.lineTo(0, 40);
        g6.stroke({ width: 8, color: 0xffffff, join: 'miter', miterLimit: 10 });
        app.stage.addChild(g6);

        // ==============================================
        // 第三行：复杂类型
        // ==============================================
    
        // 7. 线性渐变
        const g7 = new PIXI.Graphics();
        g7.x = 40;
        g7.y = 240;
        app.stage.addChild(g7);

        // 1. 创建线性渐变（x0,y0 → x1,y1 是相对于图形自身的局部坐标）
        const linearGrad = new PIXI.FillGradient({
          type: 'linear',          // 线性
          start: { x: 0, y: 0 },   // 起点（图形局部0~1）
          end: { x: 1, y: 0 },     // 终点（水平右）
          colorStops: [            // 色标
            { offset: 0, color: 0xff0000 }, // 红
            { offset: 0.5, color: 0xffff00 }, // 黄
            { offset: 1, color: 0x00ff00 }  // 绿
          ]
        });

        g7.tint = 0x44dd44;
        g7.rect(0, 0, 120, 60);
        g7.stroke({ width: 10, fill: linearGrad });

        // 8. 使用纹理描边
        // const tex = PIXI.Texture.WHITE; // 白色纹理
        await PIXI.Assets.load('./images/1.png');
        const tex = PIXI.Texture.from('./images/1.png');
        const g8 = new PIXI.Graphics();
        g8.x = 220;
        g8.y = 240;

        app.stage.addChild(g8);
        g8.rect(0, 0, 120, 60);
        g8.stroke({ width: 10, texture: tex, textureScale: 0.2 });


        // 9. 虚线矩形
        const g9 = new PIXI.Graphics();
        g9.x = 400;
        g9.y = 240;
        app.stage.addChild(g9);
        drawDashedRect(g9, 0, 0, 120, 60, 4, 0xffffff, 8, 4);
        // ==============================================
        // 工具函数：虚线
        // ==============================================
        function drawDashedLine(g, x1, y1, x2, y2, dash, gap) {
          let dx = x2 - x1;
          let dy = y2 - y1;
          let distance = Math.sqrt(dx * dx + dy * dy);
          let angle = Math.atan2(dy, dx);
          let drawDash = true;
          let current = 0;

          g.beginPath();
          while (current < distance) {
            let step = drawDash ? dash : gap;
            let end = Math.min(current + step, distance);
            let sx = x1 + Math.cos(angle) * current;
            let sy = y1 + Math.sin(angle) * current;
            let ex = x1 + Math.cos(angle) * end;
            let ey = y1 + Math.sin(angle) * end;

            if (drawDash) {
              g.moveTo(sx, sy);
              g.lineTo(ex, ey);
            }
            current = end;
            drawDash = !drawDash;
          }
          g.stroke();
        }

        function drawDashedRect(g, x, y, w, h, lineWidth, color, dash, gap) {
          g.stroke({ width: lineWidth, color: color });
          drawDashedLine(g, x, y, x + w, y, dash, gap);
          drawDashedLine(g, x + w, y, x + w, y + h, dash, gap);
          drawDashedLine(g, x + w, y + h, x, y + h, dash, gap);
          drawDashedLine(g, x, y + h, x, y, dash, gap);
        }
      }
      init();
    </script>
    </body>
    </html>
    ```

## 2.2 fill 属性

1. 参数表

    | 参数      | 类型           | 说明              |
    |---------|--------------|-----------------|
    | color   | number       | 填充颜色，十六进制格式     |
    | alpha   | number       | 透明度 0 ~ 1       |
    | fill    | FillGradient | 渐变填充（线性 / 径向）   |
    | texture | Texture      | 纹理填充（图片 / 纯色纹理） |

2. 代码展示

    ```html
    <!DOCTYPE html>
    <html lang="zh-CN">
    <head>
      <meta charset="UTF-8" />
      <title>PIXI8 fill 属性全演示</title>
    </head>
    <body>
    <script src="./js/pixi.js"></script>
    <script>
      async function init() {
        const app = new PIXI.Application();
        await app.init({
          width: 600,
          height: 400,
          backgroundColor: 0x1a1a1a,
          antialias: true
        });
        document.body.appendChild(app.canvas);

        // ==============================================
        // 第一行：纯色 / 透明 / 圆形填充
        // ==============================================
        // 1. 基础颜色填充
        const g1 = new PIXI.Graphics();
        g1.x = 40;
        g1.y = 40;
        g1.rect(0, 0, 120, 60);
        g1.fill({ color: 0xff4444 });
        app.stage.addChild(g1);

        // 2. 带透明度填充
        const g2 = new PIXI.Graphics();
        g2.x = 220;
        g2.y = 40;
        g2.rect(0, 0, 120, 60);
        g2.fill({ color: 0x44ddff, alpha: 0.4 });
        app.stage.addChild(g2);

        // 3. 圆形填充
        const g3 = new PIXI.Graphics();
        g3.x = 400;
        g3.y = 40;
        g3.circle(60, 30, 30);
        g3.fill({ color: 0xffff00 });
        app.stage.addChild(g3);

        // ==============================================
        // 第二行：线性渐变 / 径向渐变
        // ==============================================
        // 4. 线性渐变填充
        const g4 = new PIXI.Graphics();
        g4.x = 40;
        g4.y = 140;
        const linearGrad = new PIXI.FillGradient({
          type: 'linear',
          start: { x: 0, y: 0 },
          end: { x: 1, y: 0 },
          colorStops: [
            { offset: 0, color: 0xff0000 },
            { offset: 1, color: 0x00ff00 }
          ]
        });
        g4.rect(0, 0, 120, 60);
        g4.fill({ fill: linearGrad });
        app.stage.addChild(g4);

        // 5. 径向渐变填充
        const g5 = new PIXI.Graphics();
        g5.x = 220;
        g5.y = 140;
        const radialGrad = new PIXI.FillGradient({
          type: 'radial',
          start: { x: 0.5, y: 0.5, radius: 0 },
          end: { x: 0.5, y: 0.5, radius: 0.5 },
          colorStops: [
            { offset: 0, color: 0xffffff },
            { offset: 1, color: 0x0000ff }
          ]
        });
        g5.circle(60, 30, 30);
        g5.fill({ fill: radialGrad });
        app.stage.addChild(g5);

        // 6. 圆角矩形填充
        const g6 = new PIXI.Graphics();
        g6.x = 400;
        g6.y = 140;
        g6.roundRect(0, 0, 120, 60, 12);
        g6.fill({ color: 0xff00ff });
        app.stage.addChild(g6);

        // ==============================================
        // 第三行：纹理填充
        // ==============================================
        // 7. 纯色纹理填充（WHITE + tint）
        const g7 = new PIXI.Graphics();
        g7.x = 40;
        g7.y = 240;
        const tex = PIXI.Texture.WHITE;
        g7.tint = 0x44ddaa;
        g7.rect(0, 0, 120, 60);
        g7.fill({ texture: tex });
        app.stage.addChild(g7);

        // 8. 图片纹理填充
        await PIXI.Assets.load('./images/1.png');
        const imgTex = PIXI.Texture.from('./images/1.png');

        const g8 = new PIXI.Graphics();
        g8.x = 220;
        g8.y = 240;
        g8.rect(0, 0, 120, 60);
        g8.fill({ texture: imgTex });
        app.stage.addChild(g8);

        // 9. 填充 + 描边 组合
        const g9 = new PIXI.Graphics();
        g9.x = 400;
        g9.y = 240;
        g9.rect(0, 0, 120, 60);
        g9.fill({ color: 0x33aaee });
        g9.stroke({ width: 4, color: 0xffffff });
        app.stage.addChild(g9);
      }

      init();
    </script>
    </body>
    </html>
    ```

## 2.3 label 属性

1. 介绍

    所有 `Container`（Sprite/Graphics/Text/Mesh全部继承）上的实例标记属性。v7及更早版本叫 `.name`，**v8废弃 name，统一改用 `.label`**。

    ```js
    // 给任意显示对象打标签
    const sprite = new Sprite(texture);
    sprite.label = "video‑track‑0";

    const container = new Container();
    container.label = "timeline‑layer";

    const text = new Text({text:"test"});
    text.label = "subtitle‑text";
    ```

    配套查找API，如果找不到返回 `null`。

    ```js
    // 1. 查找【直接子节点】，匹配label
    const track = container.getChildByLabel("video‑track‑0");

    // 2. 深度递归查找整棵子树（包含孙子、曾孙）
    const subText = container.getDescendantByLabel("subtitle‑text");

    //3. 全部匹配，返回数组
    const allLayers = container.getDescendantsByLabel("timeline‑layer");
    ```

2. 核心特性

    1. **仅开发调试/检索标记，完全不参与渲染**
    `label` 只是一个字符串标记，**不会改变画面、宽高、渲染行为**，引擎内部完全不使用这个字段，100%留给业务代码使用。
    2. 所有显示对象都拥有：`Container、Sprite、Text、Graphics、BitmapText` 全部具备。
    3. 替代旧版 `.name`，v8中 `name` 只是别名，推荐全部写 `.label`。

3. 代码使用

    ```html
    <!DOCTYPE html>
    <html>

    <head>
      <meta charset="UTF-8">
      <script src="./js/pixi.js"></script>
    </head>

    <body>
      <script type="module">
        // 1. 初始化 Pixi8
        const app = new PIXI.Application();
        await app.init({ width: 800, height: 600, background: '#111' });
        document.body.appendChild(app.view);


        // 2. 创建容器
        const container = new PIXI.Container();
        app.stage.addChild(container);


        // 3. 创建组件 + 命名（label）
        const box1 = new PIXI.Graphics();
        box1.label = "box_red";      // 命名 
        box1.rect(0, 0, 100, 100).fill(0xff0000);
        box1.x = 100;
        box1.y = 100;
        container.addChild(box1);

        const box2 = new PIXI.Graphics();
        box2.label = "box_blue";     // 命名 
        box2.rect(0, 0, 100, 100).fill(0x0000ff);
        box2.x = 300;
        box2.y = 100;
        container.addChild(box2);


        // 4. 通过 label 查找组件（核心）
        const redBox = container.getChildByLabel("box_red");
        const blueBox = container.getChildByLabel("box_blue");


        // 5. 修改属性（任意属性都能改）
        redBox.x = 150;            // 改位置
        redBox.alpha = 0.5;        // 改透明度
        redBox.rotation = 0.3;     // 旋转
        redBox.scale.set(1.2);     // 缩放


        // 6. 移除组件（两种方法）

        // 方法A：通过找到的对象移除
        // container.removeChild(redBox);

        // 方法B：通过 label 找到后移除
        const toRemove = container.getChildByLabel("box_blue");
        if (toRemove) {
          container.removeChild(toRemove);
          console.log("已移除：box_blue");
        }

        // 7. 批量遍历容器内所有组件
        container.children.forEach((child, index) => {
          console.log(`子元素${index} → label: ${child.label}`);

          // 判断类型
          if (child instanceof PIXI.Graphics) {
            console.log("是图形组件");
          }
        });

      </script>
    </body>

    </html>
    ```

## 2.4 alpha 透明度

1. 属性介绍

    > `Container` 基类属性，**所有显示对象（Container/Sprite/Graphics/Text）全部拥有**。
    取值范围：`0 ~ 1`
    - `0`：完全透明，**不参与渲染，但对象仍然存在、可接收鼠标交互**
    - `1`：完全不透明（默认值）
    - `0.5`：半透明

    ```ts
    sprite.alpha = 0.3;
    container.alpha = 0;
    ```

2. 核心规则：层级继承

    `alpha` **会向下相乘继承**。父容器alpha × 子元素alpha = 最终屏幕透明度。

    ```ts
    const parent = new Container();
    parent.alpha = 0.5;

    const child = new Sprite(texture);
    child.alpha = 0.5;
    parent.addChild(child);
    // 屏幕实际透明度 = 0.5 * 0.5 = 0.25
    ```
    父容器设置 `alpha=0`，整棵子树全部看不见，但子对象不会销毁，事件依旧触发。

3. 代码示例

    ```html
    <!DOCTYPE html>
    <html>

    <head>
      <meta charset="UTF-8">
      <title>Pixi 线条 鼠标靠近显示</title>
      <script src="./js/pixi.js"></script>
    </head>
    <body>

      <script type="module">
        // 1. 创建应用
        const app = new PIXI.Application();
        await app.init({ width: 600, height: 400, background: '#111' });
        document.body.appendChild(app.view);
        // 2. 画线
        const line = new PIXI.Graphics();

        // 线条起点终点
        const x1 = 200, y1 = 200;
        const x2 = 400, y2 = 200;

        // 默认隐藏（alpha=0）
        line.alpha = 0;

        // 绘制线条（白色）
        line.moveTo(x1, y1);
        line.lineTo(x2, y2);
        line.stroke({ width: 10, color: 0xffffff });

        app.stage.addChild(line);

        // 3. 鼠标靠近检测（每帧判断）
        app.ticker.add(() => {
          const mouse = app.renderer.events.pointer.global;

          // 计算鼠标到线段的距离
          const dist = distanceToLineSegment(
            mouse.x, mouse.y,
            x1, y1,
            x2, y2
          );

          // 距离 < 30 像素 → 显示
          if (dist < 30) {
            line.alpha = 1;
          } else {
            line.alpha = 0;
          }
        });

        // 工具函数：计算点到线段的距离
        function distanceToLineSegment(px, py, x1, y1, x2, y2) {
          const A = px - x1;
          const B = py - y1;
          const C = x2 - x1;
          const D = y2 - y1;

          const dot = A * C + B * D;
          const lenSq = C * C + D * D;
          let param = -1;

          if (lenSq !== 0) param = dot / lenSq;

          let xx, yy;
          if (param < 0) { xx = x1; yy = y1; }
          else if (param > 1) { xx = x2; yy = y2; }
          else { xx = x1 + param * C; yy = y1 + param * D; }

          const dx = px - xx;
          const dy = py - yy;
          return Math.sqrt(dx * dx + dy * dy);
        }
      </script>
    </body>

    </html>
    ```
