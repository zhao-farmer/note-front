# 四、路由与导航

## 4.1 基础导航安装

1. RN中的路由是通过React-Navigation来完成的

    - React中通过 React-Router 实现路由
    - [https://reactnavigation.org/](https://reactnavigation.org/)

2. 基础组件

    - 安装

        ```sh
        # 基础包
        npm install @react-navigation/native
        # 必要插件
        npm install react-native-screens react-native-safe-area-context
        ```

    - 修改配置

        修改 MainActivity.kt 或 MainActivity.java 文件位于 android/app/src/main/java/下 。

        ```Kotlin{3-5}
        class MainActivity: ReactActivity() {
            // ...
            override fun onCreate(savedInstanceState: Bundle?) {
                super.onCreate(null)
            }
            // ...
        }
        ```
     
        文件顶部添加以下 import语句

        ```
        import android.os.Bundle;
        ```

## 4.2 StackNavigator 栈路由

1. 组件介绍

    - 简介
        - RN中默认没有类似浏览器的 history 对象
        - 在RN中跳转之前，先将路由声明在Stack中
    - 安装

        ```sh
        # 安装本机堆栈导航器库
        npm install @react-navigation/native-stack
        # 安装元素库
        npm install @react-navigation/elements
        ```
    - 使用

        ```js
        import { createNativeStackNavigator } from '@react-navigation/native-stack';
        const RootStack = createNativeStackNavigator({...})
        ```

2. 代码示例

    ```tsx
    import * as React from 'react';
    import { View, Text, StyleSheet } from 'react-native';
    import {
        createStaticNavigation,
        useNavigation,
    } from '@react-navigation/native';
    import { createNativeStackNavigator } from '@react-navigation/native-stack';
    import { Button } from '@react-navigation/elements';

    function HomeScreen() {
        const navigation = useNavigation();

        return (
            <View style={styles.container}>
                <Text>Home Screen</Text>
                <Button onPress={() => navigation.navigate('About')}>进入其他页</Button>
            </View>
        );
    }

    function AboutScreen() {
        const navigation = useNavigation();
        return (
            <View style={styles.container}>
                <Text>About Screen</Text>
                <Button onPress={() => navigation.navigate('Home')}>进入主页</Button>
                <Button onPress={() => navigation.goBack()}>返回上一页</Button>
            </View>
        );
    }

    // 3. 创建导航器
    const RootStack = createNativeStackNavigator({
        initialRouteName: 'Home', // 初始加载
        // 配置项
        screenOptions: {
            headerStyle: { backgroundColor: 'tomato' },
        },
        screens: {
            Home: HomeScreen,
            About: AboutScreen,
        },
    });

    const Navigation = createStaticNavigation(RootStack);

    export default function Index() {
        return <Navigation />;
    }

    const styles = StyleSheet.create({
        container: {
            flex: 1,
            justifyContent: 'center',
            alignItems: 'center',
            gap:10
        },
    });
    ```

## 4.3 BottomTabNavigator 底部导航

1. 使用组件

    - 介绍

        屏幕底部一个简单的标签栏，可以让你在不同的路线之间切换。路由是惰性初始化的——它们的屏幕组件在第一次聚焦之前不会被挂载。
    - 安装

        ```sh
        npm install @react-navigation/bottom-tabs
        ```
    - 地址：[https://reactnavigation.org/docs/bottom-tab-navigator](https://reactnavigation.org/docs/bottom-tab-navigator)

2. 代码示例

    ```tsx

        createStaticNavigation,

        const navigation = useNavigation();
        return (
            <View style={styles.container}>
                <Text style={styles.title}>首页</Text>
                <Text>欢迎来到首页</Text>
                <Button
                    title="跳转到个人资料"
                    onPress={() => navigation.navigate('Profile')}
                />
            </View>
        );
    }

    // 搜索页面
    function SearchScreen() {
        return (
            <View style={styles.container}>
                <Text style={styles.title}>搜索</Text>
                <Text>搜索页面内容</Text>
            </View>
        );
    }

    // 个人资料页面
    function ProfileScreen() {
        const navigation = useNavigation();
        return (
            <View style={styles.container}>
                <Text style={styles.title}>个人资料</Text>
                <Text>这是我的个人资料页面</Text>
                <Button title="返回首页" onPress={() => navigation.navigate('Home')} />

    }

    // 通知页面

        return (
            <View style={styles.container}>
                <Text style={styles.title}>通知</Text>
                <Text>您有 3 条新通知</Text>
            </View>
        );  
    }

    // 创建底部标签导航器（静态配置）

        initialRouteName: 'Home',
        screenOptions: {
            // 选装后颜色
            tabBarActiveTintColor: '#007AFF',
            // 未选中颜色
            tabBarInactiveTintColor: 'gray',
            // tabBar 样式
            tabBarStyle: {
                backgroundColor: 'white',
            },
            // 标头样式
            headerStyle: {
                backgroundColor: '#007AFF',
            },
            // 标头文字颜色
            headerTintColor: 'white',
        },
        screens: {
            Home: {
                screen: HomeScreen,

                    title: '首页',

                    tabBarIcon: ({ focused, color, size }: any) => (
                        <Button title={'首页'} color={color} />
                    ),
                },
            },
            Search: {
                screen: SearchScreen,
                options: {
                    title: '搜索',
                },
            },
            Notifications: {
                screen: NotificationsScreen,
                options: {
                    title: '通知',
                    tabBarBadge: 3, // 徽章
                },
            },
            Profile: {
                screen: ProfileScreen,

                    tabBarLabel: '我的',

    const Navigation = createStaticNavigation(Tab);

    export default function App() {
        return <Navigation />;
    }

    const styles = StyleSheet.create({
        container: {
            flex: 1,
            justifyContent: 'center',
            alignItems: 'center',
            backgroundColor: '#f5f5f5',
        },
        title: {
            fontSize: 24,
            fontWeight: 'bold',
            marginBottom: 20,
        },
    });
    ```

## 4.5 Drawer 抽屉导航

1. 介绍

    - 安装

        ```sh
        # 安装抽屉导航
        npm install @react-navigation/drawer
        # 安装图表库
        npm install @react-native-vector-icons/ionicons
        # 安装抽屉导航器所需的库
        npm install react-native-gesture-handler react-native-reanimated react-native-worklets
        ```
    - 配置
        - babel.config.js 中添加
            ```js
            plugins: [
                'react-native-worklets/plugin',
            ],        
            ```
        - index.js 获取 app.tsx 中添加

            ```js
            import 'react-native-gesture-handler';
            ```
    - 网站信息
        - Drawer 配置 [https://reactnavigation.org/docs/drawer-navigator](https://reactnavigation.org/docs/drawer-navigator)
        - 动画效果 [https://docs.swmansion.com/react-native-reanimated](https://docs.swmansion.com/react-native-reanimated)
        - github [https://github.com/software-mansion/react-native-reanimated](https://github.com/software-mansion/react-native-reanimated)

## 4.7 路由嵌套

1. 方法：可以将栈路由放入其他路由中

2. 代码示例

    ```tsx
    import {
        createStaticNavigation,
        useNavigation,
    } from '@react-navigation/native';
    import { createBottomTabNavigator } from '@react-navigation/bottom-tabs';
    import { createNativeStackNavigator } from '@react-navigation/native-stack';
    import { Text, View, Button, StyleSheet } from 'react-native';
    import { Ionicons } from '@react-native-vector-icons/ionicons';

    // ==================== Home Stack ====================
    function HomeScreen() {
        const navigation = useNavigation();
        return (
            <View style={[styles.container]}>
            <Text>主页</Text>
            <Button
                title="前往主页详情"

            </View>
        );

    function HomeDetailsScreen() {
        const navigation = useNavigation();
        return (

            <Button
                title="前往设置页面"
                onPress={() => navigation.navigate('SettingsTab')}

                        <Ionicons
                            name={focused ? 'person' : 'person-outline'}
    ```

## 4.8 路由传参

1. 以 `navigation.navigate(路由名称,参数)`  进行传值

2. 代码示例

    ```tsx
    import {
        createStaticNavigation,
        useNavigation,
        useRoute,
    } from '@react-navigation/native';
    import { createBottomTabNavigator } from '@react-navigation/bottom-tabs';
    import { createNativeStackNavigator } from '@react-navigation/native-stack';
    import { Text, View, Button, StyleSheet } from 'react-native';
    import { Ionicons } from '@react-native-vector-icons/ionicons';

    // ==================== Home Stack ====================
    function HomeScreen() {
        const navigation = useNavigation();

        const users = [
            { id: '1', name: '张三', age: 25, email: 'zhangsan@example.com' },
            { id: '2', name: '李四', age: 30, email: 'lisi@example.com' },
            { id: '3', name: '王五', age: 28, email: 'wangwu@example.com' },
        ];

        const products = [
            { id: 101, title: 'iPhone 15', price: 5999, category: '电子产品' },
            { id: 102, title: 'MacBook Pro', price: 12999, category: '电脑' },
            { id: 103, title: 'AirPods', price: 1299, category: '耳机' },
        ];

        return (
            <View style={[styles.container]}>
                <Text style={styles.title}>主页</Text>
                
                <View style={styles.section}>
                    <Text style={styles.sectionTitle}>用户列表</Text>
                    {users.map(user => (
                        <Button
                            key={user.id}
                            title={`查看 ${user.name} 的资料`}
                            onPress={() => navigation.navigate('HomeDetails', {
                                userId: user.id,
                                userName: user.name,
                                userAge: user.age,
                                userEmail: user.email,
                                type: 'user'
                            })}
                        />
                    ))}
                </View>

                <View style={styles.section}>
                    <Text style={styles.sectionTitle}>商品列表</Text>
                    {products.map(product => (
                        <Button
                            key={product.id}
                            title={`查看 ${product.title}`}
                            onPress={() => navigation.navigate('HomeDetails', {
                                productId: product.id,
                                productName: product.title,
                                productPrice: product.price,
                                productCategory: product.category,
                                type: 'product'
                            })}
                        />
                    ))}
                </View>

                <Button
                    title="前往设置页面（带参数）"
                    onPress={() => navigation.navigate('SettingsTab', {
                        screen: 'Settings',
                        params: { 
                            from: 'home',
                            timestamp: new Date().toISOString()
                        }
                    })}
                />
            </View>
        );
    }

    function HomeDetailsScreen() {
        const navigation = useNavigation();
        const route = useRoute();
        const params = route.params || {};

        return (
            <View style={[styles.container]}>
                <Text style={styles.title}>主页详情</Text>
                
                {params.type === 'user' ? (
                    <View style={styles.detailCard}>
                        <Text style={styles.detailTitle}>用户信息</Text>
                        <Text>用户ID: {params.userId}</Text>
                        <Text>姓名: {params.userName}</Text>
                        <Text>年龄: {params.userAge}岁</Text>
                        <Text>邮箱: {params.userEmail}</Text>
                    </View>
                ) : params.type === 'product' ? (
                    <View style={styles.detailCard}>
                        <Text style={styles.detailTitle}>商品信息</Text>
                        <Text>商品ID: {params.productId}</Text>
                        <Text>商品名称: {params.productName}</Text>
                        <Text>价格: ¥{params.productPrice}</Text>
                        <Text>分类: {params.productCategory}</Text>
                    </View>
                ) : (
                    <Text>暂无数据</Text>
                )}

                <Button
                    title="前往个人资料（带用户ID）"
                    onPress={() => navigation.navigate('ProfileTab', {
                        screen: 'Profile',
                        params: { 
                            userId: params.userId || 'default',
                            source: 'home_details'
                        }
                    })}
                />

                <Button
                    title="前往设置页面"
                    onPress={() => navigation.navigate('SettingsTab', {
                        screen: 'AdvancedSettings',
                        params: { 
                            feature: 'premium',
                            level: 'advanced'
                        }
                    })}
                />

                <Button
                    title="返回"
                    onPress={() => navigation.goBack()}
                />
            </View>
        );
    }

    const HomeStack = createNativeStackNavigator({
        screens: {
            Home: HomeScreen,
            HomeDetails: HomeDetailsScreen,
        },
    });

    // ==================== Profile Stack ====================
    function ProfileScreen() {
    ```
