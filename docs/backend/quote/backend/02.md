# 二、数据库

## 1.1 MySQL 连接

1. 安装插件

    ```bash
    npm install mysql2
    ```

2. 使用实例

    ```javascript
    // 连接配置
    const mysql = require('mysql2/promise');

    const dbConfig = {
      host: 'localhost',
      user: 'root',
      password: 'password',
      database: 'myapp',
      waitForConnections: true,
      connectionLimit: 10,
      queueLimit: 0
    };

    // 创建连接池（推荐生产环境使用）
    const pool = mysql.createPool(dbConfig);

    // 使用示例
    async function getUsers() {
      const [rows] = await pool.execute('SELECT * FROM users WHERE age > ?', [18]);
      return rows;
    }

    // 或者使用连接（不推荐生产环境）
    const connection = await mysql.createConnection(dbConfig);
    ```

## 1.2 PostgreSQL 连接

1. 安装插件

    ```bash
    npm install pg
    ```
2. 使用示例

    ```javascript
    const { Pool } = require('pg');

    const pool = new Pool({
      user: 'postgres',
      host: 'localhost',
      database: 'myapp',
      password: 'password',
      port: 5432,
      max: 20, // 最大连接数
      idleTimeoutMillis: 30000,
      connectionTimeoutMillis: 2000,
    });

    // 查询示例
    async function getUsers() {
      const result = await pool.query('SELECT * FROM users WHERE age > $1', [18]);
      return result.rows;
    }

    // 事务示例
    async function transferMoney(fromId, toId, amount) {
      const client = await pool.connect();
      try {
        await client.query('BEGIN');
        await client.query('UPDATE accounts SET balance = balance - $1 WHERE id = $2', [amount,     fromId]);
        await client.query('UPDATE accounts SET balance = balance + $1 WHERE id = $2', [amount,     toId]);
        await client.query('COMMIT');
      } catch (error) {
        await client.query('ROLLBACK');
        throw error;
      } finally {
        client.release();
      }
    }
    ```

## 1.3 Redis 连接

1. 安装插件

    ```bash
    npm install ioredis
    ```
2. 使用实例

    ```javascript
    const Redis = require('ioredis');

    // 基本连接
    const redis = new Redis({
      port: 6379,
      host: '127.0.0.1',
      password: 'your_password', // 如果有密码
      db: 0, // 数据库编号
    });

    // 连接池配置
    const redis = new Redis.Cluster([
      {
        port: 6379,
        host: '127.0.0.1',
      }
    ], {
      scaleReads: 'slave', // 读写分离
    });

    // 使用示例
    async function redisExample() {
      // 字符串操作
      await redis.set('key', 'value', 'EX', 3600); // 过期时间 1 小时
      const value = await redis.get('key');
    
      // 哈希操作
      await redis.hset('user:1', 'name', 'John', 'age', '30');
      const user = await redis.hgetall('user:1');
    
      // 列表操作
      await redis.lpush('messages', 'msg1', 'msg2');
      const messages = await redis.lrange('messages', 0, -1);
    
      // 发布订阅
      const subscriber = redis.duplicate();
      subscriber.subscribe('channel');
      subscriber.on('message', (channel, message) => {
        console.log(`收到消息: ${message} 来自频道: ${channel}`);
      });
    
      // 发布消息
      await redis.publish('channel', 'Hello Redis!');
    }
    ```

## 4. MongoDB 连接

1. 安装插件

    ```bash
    npm install mongoose
    ```

    ```javascript
    const mongoose = require('mongoose');

    // 连接字符串
    const MONGODB_URI = 'mongodb://username:password@localhost:27017/myapp?authSource=admin';

    // 连接配置
    mongoose.connect(MONGODB_URI, {
      useNewUrlParser: true,
      useUnifiedTopology: true,
      maxPoolSize: 10, // 连接池大小
      serverSelectionTimeoutMS: 5000,
      socketTimeoutMS: 45000,
    });

    // 连接事件
    mongoose.connection.on('connected', () => {
      console.log('MongoDB连接成功');
    });

    mongoose.connection.on('error', (err) => {
      console.error('MongoDB连接错误:', err);
    });

    mongoose.connection.on('disconnected', () => {
      console.log('MongoDB连接断开');
    });

    // 定义Schema和Model
    const userSchema = new mongoose.Schema({
      name: { type: String, required: true },
      email: { type: String, unique: true },
      age: Number,
      createdAt: { type: Date, default: Date.now }
    });

    const User = mongoose.model('User', userSchema);

    // 使用示例
    async function mongoExample() {
      // 创建用户
      const user = await User.create({
        name: 'John Doe',
        email: 'john@example.com',
        age: 30
      });
    
      // 查询
      const users = await User.find({ age: { $gt: 18 } });
    
      // 更新
      await User.updateOne({ _id: user._id }, { age: 31 });
    
      // 聚合查询
      const result = await User.aggregate([
        { $group: { _id: null, averageAge: { $avg: '$age' } } }
      ]);
    }
    ```


## 5. **多数据库整合配置**

1. **config/database.js**

    ```javascript
    require('dotenv').config();

    module.exports = {
      // MySQL 配置
      mysql: {
        host: process.env.MYSQL_HOST || 'localhost',
        user: process.env.MYSQL_USER || 'root',
        password: process.env.MYSQL_PASSWORD || '',
        database: process.env.MYSQL_DATABASE || 'myapp',
        pool: {
          max: 10,
          min: 0,
          acquire: 30000,
          idle: 10000
        }
      },
    
      // PostgreSQL 配置
      postgres: {
        host: process.env.PG_HOST || 'localhost',
        user: process.env.PG_USER || 'postgres',
        password: process.env.PG_PASSWORD || '',
        database: process.env.PG_DATABASE || 'myapp',
        port: process.env.PG_PORT || 5432,
        max: 20,
        idleTimeoutMillis: 30000
      },
    
      // Redis 配置
      redis: {
        host: process.env.REDIS_HOST || 'localhost',
        port: process.env.REDIS_PORT || 6379,
        password: process.env.REDIS_PASSWORD || '',
        db: process.env.REDIS_DB || 0
      },
    
      // MongoDB 配置
      mongodb: {
        uri: process.env.MONGODB_URI || 'mongodb://localhost:27017/myapp',
        options: {
          useNewUrlParser: true,
          useUnifiedTopology: true,
          maxPoolSize: 10
        }
      }
    };
    ```

2. **数据库连接管理器**



    ```javascript
    const mysql = require('mysql2/promise');
    const { Pool } = require('pg');
    const Redis = require('ioredis');
    const mongoose = require('mongoose');
    const config = require('./config/database');

    class DatabaseManager {
      constructor() {
        this.mysqlPool = null;
        this.pgPool = null;
        this.redis = null;
        this.mongoConnected = false;
      }
    
      // 连接 MySQL
      async connectMySQL() {
        this.mysqlPool = mysql.createPool(config.mysql);
        console.log('MySQL 连接池已创建');
      }
    
      // 连接 PostgreSQL
      async connectPostgreSQL() {
        this.pgPool = new Pool(config.postgres);
        console.log('PostgreSQL 连接池已创建');
      }
    
      // 连接 Redis
      async connectRedis() {
        this.redis = new Redis(config.redis);
        console.log('Redis 连接成功');
      }
    
      // 连接 MongoDB
      async connectMongoDB() {
        await mongoose.connect(config.mongodb.uri, config.mongodb.options);
        this.mongoConnected = true;
        console.log('MongoDB 连接成功');
      }
    
      // 关闭所有连接
      async closeAll() {
        if (this.mysqlPool) await this.mysqlPool.end();
        if (this.pgPool) await this.pgPool.end();
        if (this.redis) await this.redis.quit();
        if (this.mongoConnected) await mongoose.connection.close();
        console.log('所有数据库连接已关闭');
      }
    }

    module.exports = new DatabaseManager();
    ```

3. **使用示例**

    ```javascript
    const dbManager = require('./dbManager');

    async function startup() {
      try {
        await dbManager.connectMySQL();
        await dbManager.connectPostgreSQL();
        await dbManager.connectRedis();
        await dbManager.connectMongoDB();

        console.log('所有数据库连接成功');
      } catch (error) {
        console.error('数据库连接失败:', error);
        process.exit(1);
      }
    }

    // 优雅关闭
    process.on('SIGINT', async () => {
      await dbManager.closeAll();
      process.exit(0);
    });
    ```
