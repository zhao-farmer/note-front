# 九、列表与表单

## 9.1 SortableJS

当然可以！以下是 **SortableJS** 在 **原生 HTML、Vue、React、Svelte、SolidJS、Next.js、Nuxt** 中的完整使用示例，涵盖基础拖拽排序功能，并适配各框架的最佳实践（包括 SSR 兼容性处理）。

### 9.1.1 通用准备

先安装 SortableJS（若使用构建工具）：

```bash
npm install sortablejs
```

> 原生 HTML 可直接通过 CDN 引入。

```html
 <script src="https://cdn.jsdelivr.net/npm/sortablejs@1.15.2/Sortable.min.js"></script>
```

### 9.1.3 Vue 3（Composition API）

```vue
<template>
  <ul ref="listRef" class="list">
    <li v-for="item in items" :key="item.id">{{ item.name }}</li>
  </ul>
</template>

<script setup>
import { ref, onMounted } from 'vue'
import Sortable from 'sortablejs'

const items = ref([
  { id: 1, name: 'Apple' },
  { id: 2, name: 'Banana' },
  { id: 3, name: 'Cherry' }
])

const listRef = ref(null)

onMounted(() => {
  Sortable.create(listRef.value, {
    animation: 150,
    onEnd: (evt) => {
      const { oldIndex, newIndex } = evt
      const moved = items.value.splice(oldIndex, 1)[0]
      items.value.splice(newIndex, 0, moved)
    }
  })
})
</script>

<style scoped>
.list {
  list-style: none; padding: 0;
}
.list li {
  padding: 8px; margin: 4px 0;
  background: #f5f5f5; cursor: move;
}
</style>
```

### 9.1.4 React（函数组件 + useRef）

```jsx
import React, { useRef, useEffect, useState } from 'react'
import Sortable from 'sortablejs'

export default function SortableList() {
  const [items, setItems] = useState([
    { id: 1, name: 'React' },
    { id: 2, name: 'Vue' },
    { id: 3, name: 'Svelte' }
  ])

  const listRef = useRef(null)

  useEffect(() => {
    const sortable = new Sortable(listRef.current, {
      animation: 150,
      onEnd: ({ oldIndex, newIndex }) => {
        const newItems = [...items]
        const moved = newItems.splice(oldIndex, 1)[0]
        newItems.splice(newIndex, 0, moved)
        setItems(newItems)
      }
    })

    return () => sortable.destroy()
  }, [items])

  return (
    <ul ref={listRef} style={{ listStyle: 'none', padding: 0 }}>
      {items.map(item => (
        <li key={item.id} style={{
          padding: '8px', margin: '4px 0',
          background: '#f0f0f0', cursor: 'move'
        }}>
          {item.name}
        </li>
      ))}
    </ul>
  )
}
```

### 9.1.5  Svelte

```svelte
<!-- SortableList.svelte -->
<script>
  import { onMount } from 'svelte'
  import Sortable from 'sortablejs'

  let items = [
    { id: 1, name: 'Svelte' },
    { id: 2, name: 'Solid' },
    { id: 3, name: 'Qwik' }
  ]

  let listEl

  onMount(() => {
    const sortable = new Sortable(listEl, {
      animation: 150,
      onEnd: (evt) => {
        const { oldIndex, newIndex } = evt
        const moved = items.splice(oldIndex, 1)[0]
        items = [...items.slice(0, newIndex), moved, ...items.slice(newIndex)]
      }
    })

    return () => sortable.destroy()
  })
</script>

<ul bind:this={listEl} style="list-style: none; padding: 0;">
  {#each items as item (item.id)}
    <li style="padding: 8px; margin: 4px 0; background: #eee; cursor: move;">
      {item.name}
    </li>
  {/each}
</ul>
```

> svelte 的响应式赋值 `items = [...]` 会自动触发更新。

### 9.1.6 SolidJS

```jsx
import { createSignal, onMount } from 'solid-js'
import Sortable from 'sortablejs'

export default function App() {
  const [items, setItems] = createSignal([
    { id: 1, name: 'Solid' },
    { id: 2, name: 'Preact' },
    { id: 3, name: 'Alpine' }
  ])

  let listRef

  onMount(() => {
    Sortable.create(listRef, {
      animation: 150,
      onEnd: (evt) => {
        const { oldIndex, newIndex } = evt
        const current = items()
        const moved = current.splice(oldIndex, 1)[0]
        current.splice(newIndex, 0, moved)
        setItems([...current])
      }
    })
  })

  return (
    <ul ref={listRef!} style={{ "list-style": "none", "padding": "0" }}>
      {items().map(item => (
        <li style={{
          "padding": "8px",
          "margin": "4px 0",
          "background": "#f9f9f9",
          "cursor": "move"
        }}>
          {item.name}
        </li>
      ))}
    </ul>
  )
}
```

### 9.1.7 Next.js（App Router，React + SSR 安全）

```tsx
// app/sortable/page.tsx
'use client'

import { useEffect, useRef, useState } from 'react'

export default function SortablePage() {
  const [items, setItems] = useState([
    { id: 1, name: 'Next.js' },
    { id: 2, name: 'Remix' },
    { id: 3, name: 'Astro' }
  ])

  const listRef = useRef<HTMLUListElement>(null)

  useEffect(() => {
    // 动态导入避免 SSR 报错
    import('sortablejs').then(({ default: Sortable }) => {
      new Sortable(listRef.current!, {
        animation: 150,
        onEnd: ({ oldIndex, newIndex }) => {
          const newItems = [...items]
          const moved = newItems.splice(oldIndex, 1)[0]
          newItems.splice(newIndex, 0, moved)
          setItems(newItems)
        }
      })
    })
  }, [items])

  return (
    <ul ref={listRef} style={{ listStyle: 'none', padding: 0 }}>
      {items.map(item => (
        <li key={item.id} style={{
          padding: '8px', margin: '4px 0',
          background: '#fafafa', cursor: 'move'
        }}>
          {item.name}
        </li>
      ))}
    </ul>
  )
}
```

> 使用 `'use client'` + 动态 `import()` 确保仅在客户端运行。

### 9.1.8 Nuxt 3（Vue 3 + SSR 安全）

```vue
<!-- components/SortableList.vue -->
<template>
  <ul ref="listRef" class="list">
    <li v-for="item in items" :key="item.id">{{ item.name }}</li>
  </ul>
</template>

<script setup>
import { ref, onMounted } from 'vue'

const items = ref([
  { id: 1, name: 'Nuxt' },
  { id: 2, name: 'Pinia' },
  { id: 3, name: 'Nitro' }
])

const listRef = ref(null)

onMounted(async () => {
  const Sortable = (await import('sortablejs')).default
  Sortable.create(listRef.value, {
    animation: 150,
    onEnd: (evt) => {
      const { oldIndex, newIndex } = evt
      const moved = items.value.splice(oldIndex, 1)[0]
      items.value.splice(newIndex, 0, moved)
    }
  })
})
</script>

<style scoped>
.list {
  list-style: none; padding: 0;
}
.list li {
  padding: 8px; margin: 4px 0;
  background: #f0f0f0; cursor: move;
}
</style>
```

在页面中使用时（可选加 `<ClientOnly>`）：

```vue
<!-- pages/index.vue -->
<template>
  <div>
    <ClientOnly>
      <SortableList />
    </ClientOnly>
  </div>
</template>
```

> `<ClientOnly>` 是 Nuxt 3 内置组件，确保仅客户端渲染。
