# 五、路由

## 5.1 基础使用

1. 安装插件

    ```sh
    pnpm install @solidjs/router
    ```

2. 创建路由组件

    ```jsx
    // src/pages/Home.tsx
    export default function Home() {
      return <h1>首页</h1>;
    }

    // src/pages/About.tsx
    export default function About() {
      return <h1>关于我们</h1>;
    }
    ```
3. index.js 入口文件添加 路由

    ```js
    // src/index.js
    import { render } from 'solid-js/web';
    import 'solid-devtools';

    import App from './App';
    import { Router,Route } from '@solidjs/router';
    import Home from './pages/Home';
    import About from './pages/About';

    render(() =>
        <Router root={App} >
            <Route path="/" component={Home} />
            <Route path="/about" component={About} />  
        </Router>, 
        document.getElementById('root')
    );
    ```
4. App.jsx 中接受路由，并设置导航

    ```jsx
    import { A } from '@solidjs/router';

    const App = (props) => {
      return (
        <>
          <nav>
            <A href="/">首页</A>
            <A href="/about">关于</A>
          </nav>
          {props.children}
        </>
      );
    };

    export default App;
    ```

## 5.2 动态路由

1. 添加页面

    ```jsx
    // src/pages/User.jsx
    import { useParams } from '@solidjs/router';

    export default function User() {
      const params = useParams();
      return <h1>用户 ID: {params.id}</h1>;
    }
    ```
2. 配置路由index.js

    ```js
    import User from './pages/User';
    <Route path="/user/:id" component={User} />
    ```
3. 页面输入网址 `http://localhost:5173/user/123`

## 5.3 查询参数

1. 添加页面

    ```jsx
    // src/pages/Search.jsx
    import { useLocation } from '@solidjs/router';

    export default function Search() {
      const location = useLocation();
      const query = new URLSearchParams(location.search).get('name');
    
      return <div>搜索关键词: {query}</div>;
    }
    ```
2. 配置路由index.js

    ```js
    import Search from './pages/Search';
    <Route path="/search" component={Search} />
    ```
3. 页面输入网址 `http://localhost:5173/search?name=zhangsan&age=18`

## 5.4 编程式导航

1. home.jsx 修改

    ```jsx
    import { useNavigate } from '@solidjs/router';
    export default function Home() {

      const navigate = useNavigate();

      const userData = {
        id: 1,
        name: '张三',
        age: 18
      };

      // 直接跳转
      const handleAbout = async () => {
        navigate('/about', { replace: true }); // 替换当前历史记录
      };

      // 传递参数
      const handleSearch = async () => {
        // 序列化为查询字符串
        const queryParams = new URLSearchParams(userData).toString();
        console.log(queryParams);

        // 通过查询参数传递
        navigate(`/search?${queryParams}`);
      };


      // 传递对象
      const handleProfile = () => {
        // 通过 state 传递对象
        navigate('/profile', { state: userData });
      };



      return (
        <>
          <h1>首页</h1>
          <button onClick={handleAbout}>跳转到about页面</button>
          <button onClick={handleSearch}>跳转到Search页面</button>
          <button onClick={handleProfile}>跳转到Profile页面</button>
        </>

      );
    }
    ```
2. 新建 Profile.jsx 页面

    ```jsx
    import { useLocation } from '@solidjs/router';

    export default function Profile() {
      const location = useLocation();
      const userData = location.state; // 直接获取传递的对象

      // 确保数据存在
      if (!userData) {
        return <div>用户数据未找到</div>;
      }

      return (
        <div>
          <h1>用户资料</h1>
           <p>姓名: {userData.name}</p>
           <p>年龄: {userData.age}</p>
        </div>
      );
    }
    ```

## 5.5 嵌套路由

1. 新建页面

    ```jsx
    // src/pages/Path1/index.jsx
    export default function Path1() {
        return <h1>Path1页面</h1>
    }

    // src/pages/Path1/Path2.jsx
    export default function Path2() {
        return <h2>Path2页面</h2>;
    }
    ```
2. 路由配置

    ```js
    import Path1 from './pages/Path1';
    import Path2 from './pages/Path1/Path2';

    <Route path="/path1">
        <Route path="/" component={Path1} />
        <Route path="/path2" component={Path2} />
    </Route>
    ```

3. 浏览器输入

    - 父页面：`http://localhost:5173/path1`
    - 子页面：`http://localhost:5173/path1/path2`

> 注意：
>
> - 只有子节点`<Route>`（最里面的`<Route>`组件）才有路由。
> - 嵌套的父节点内容并不会展示到子节点路径中

## 5.6 懒加载

作用：减少首次加载包的体积，优化首屏性能

```jsx
import { lazy } from 'solid-js';
import { Route } from '@solidjs/router';

const About = lazy(() => import('./pages/About'));

// 在 Route 中直接使用
<Route path="/about" component={About} />
```

## 5.7 全局路由守卫

- 步骤 1：创建全局守卫组件


    ```jsx
    // guards/GlobalGuard.tsx
    import { createEffect } from 'solid-js';
    import { useLocation, useNavigate } from '@solidjs/router';
    import { isAuthenticated, getUserRole } from '../auth'; // 假设的认证逻辑

    export default function GlobalGuard(props: { children: any }) {
      const location = useLocation();
      const navigate = useNavigate();

      createEffect(() => {
        const currentPath = location.pathname;

        // 1. 检查是否需要认证
        const publicRoutes = ['/', '/login', '/register', '/about'];
        const isPublicRoute = publicRoutes.some(route => currentPath.startsWith(route));

        if (!isAuthenticated() && !isPublicRoute) {
          // 未登录且访问私有路由 → 重定向到登录页
          navigate(`/login?redirect=${encodeURIComponent(currentPath)}`, { replace:true });
          return;
        }

        // 2. 检查权限（例如：只有 admin 能访问 /admin）
        if (currentPath.startsWith('/admin') && getUserRole() !== 'admin') {
          navigate('/unauthorized', { replace: true });
          return;
        }

        // 3. 其他全局逻辑（如记录页面访问、设置页面标题等）
        console.log('监听页面跳转:', currentPath);
        document.title = `MyApp - ${currentPath}`;
      });

      // 渲染子组件
      return props.children;
    }
    ```

    **关键点**：

    - createEffect 会在 location.pathname 变化时自动重新运行
    - 使用 navigate(..., { replace: true }) 避免在历史记录中留下无效条目
    - 守卫逻辑写在 createEffect 中，天然响应路由变化


- 步骤 2：在 App 根部包裹守卫

    ```tsx
    // App.tsx
    import { Routes, Route } from '@solidjs/router';
    import GlobalGuard from './guards/GlobalGuard';
    import Home from './pages/Home';
    import Login from './pages/Login';
    import Dashboard from './pages/Dashboard';
    import Admin from './pages/Admin';

    export default function App() {
      return (
        <GlobalGuard>
          <Routes>
            <Route path="/" component={Home} />
            <Route path="/login" component={Login} />
            <Route path="/dashboard" component={Dashboard} />
            <Route path="/admin" component={Admin} />
          </Routes>
        </GlobalGuard>
      );
    }
    ```
