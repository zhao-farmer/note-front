    ```sql
    -- 创建数据库
    CREATE DATABASE IF NOT EXISTS solidstart_app CHARACTER SET utf8mb4 COLLATE  utf8mb4_unicode_ci;

    -- 使用数据库
    USE solidstart_app;

    -- 删除已存在的表（如果存在）
    DROP TABLE IF EXISTS users;

    -- 创建 users 表
    CREATE TABLE users (
        id INT AUTO_INCREMENT PRIMARY KEY,
        name VARCHAR(100) NOT NULL,
        email VARCHAR(255) NOT NULL UNIQUE,
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
        updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

        -- 索引优化
        INDEX idx_email (email),
        INDEX idx_name (name)
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

    -- 插入初始化数据
    INSERT INTO users (id, name, email) VALUES
    (1, '张三', 'zhangsan@example.com'),
    (2, '李四', 'lisi@example.com'),
    (3, '王五', 'wangwu@example.com'),
    (4, '赵六', 'zhaoliu@example.com'),
    (5, '孙七', 'sunqi@example.com');

    -- 设置自增起始值（确保与插入的最大ID一致）
    ALTER TABLE users AUTO_INCREMENT = 6;

    -- 验证数据
    SELECT * FROM users;
    ```

4. 数据库连接池

    ```ts
    // src/lib/db/config.ts
    import { createPool, Pool } from 'mysql2/promise';

    // 环境变量验证
    const requiredEnvVars = [
      'DATABASE_HOST',
      'DATABASE_PORT',
      'DATABASE_NAME',
      'DATABASE_USER',
      'DATABASE_PASSWORD'
    ] as const;

    for (const envVar of requiredEnvVars) {
      if (!process.env[envVar]) {
        throw new Error(`Missing environment variable: ${envVar}`);
      }
    }

    // 数据库配置
    export const dbConfig = {
      host: process.env.DATABASE_HOST!,
      port: parseInt(process.env.DATABASE_PORT!),
      database: process.env.DATABASE_NAME!,
      user: process.env.DATABASE_USER!,
      password: process.env.DATABASE_PASSWORD!,
      connectionLimit: parseInt(process.env.DATABASE_CONNECTION_LIMIT || '10'),
      charset: 'utf8mb4' as const,
      timezone: '+08:00',
      dateStrings: true, // 保持日期为字符串格式
      supportBigNumbers: true,
      bigNumberStrings: true,
    };

    // 创建连接池
    let pool: Pool;

    export function getPool(): Pool {
      if (!pool) {
        pool = createPool({
          ...dbConfig,
          waitForConnections: true,
          queueLimit: 0,
        });

        // 连接池事件监听（可选，用于调试）
        pool.on('connection', () => {
          console.log('MySQL connection established');
        });

        pool.on('error', (err) => {
          console.error('MySQL pool error:', err);
        });
      }
      return pool;
    }
    ```
5. 用户数据库操作文件

    ```ts
    // src/lib/db/users.ts
    import { getPool } from './config';

    export type User = {
      id: number;
      name: string;
      email: string;
      createdAt?: string;
      updatedAt?: string;
    };

    export type CreateUserInput = {
      name: string;
      email: string;
    };

    export type UpdateUserInput = Partial<Omit<User, 'id' | 'createdAt' | 'updatedAt'>>;

    // 用户数据库操作类
    export class UsersDB {
      private static instance: UsersDB;
    
      private constructor() {}
    
      public static getInstance(): UsersDB {
        if (!UsersDB.instance) {
          UsersDB.instance = new UsersDB();
        }
        return UsersDB.instance;
      }

      // 获取所有用户
      async getAll(): Promise<User[]> {
        try {
          const pool = getPool();
          const [rows] = await pool.execute<User[]>(
            'SELECT id, name, email, created_at as createdAt, updated_at as updatedAt FROM  users ORDER BY id ASC'
          );
          return rows;
        } catch (error) {
          console.error('Error fetching all users:', error);
          throw new Error('Failed to fetch users');
        }
      }

      // 根据ID获取用户
      async getById(id: number): Promise<User | null> {
        if (id <= 0) {
          throw new Error('Invalid user ID');
        }

        try {
          const pool = getPool();
          const [rows] = await pool.execute<User[]>(
            'SELECT id, name, email, created_at as createdAt, updated_at as updatedAt FROM  users WHERE id = ?',
            [id]
          );
          return rows[0] || null;
        } catch (error) {
          console.error(`Error fetching user by ID ${id}:`, error);
          throw new Error('Failed to fetch user');
        }
      }

      // 根据邮箱获取用户
      async getByEmail(email: string): Promise<User | null> {
        if (!email || !email.includes('@')) {
          throw new Error('Invalid email format');
        }

        try {
          const pool = getPool();
          const [rows] = await pool.execute<User[]>(
            'SELECT id, name, email, created_at as createdAt, updated_at as updatedAt FROM  users WHERE email = ?',
            [email]
          );
          return rows[0] || null;
        } catch (error) {
          console.error(`Error fetching user by email ${email}:`, error);
          throw new Error('Failed to fetch user');
        }
      }

      // 创建用户
      async create(input: CreateUserInput): Promise<User> {
        const { name, email } = input;

        // 基本验证
        if (!name?.trim()) {
          throw new Error('Name is required');
        }
        if (!email?.trim() || !email.includes('@')) {
          throw new Error('Valid email is required');
        }

        try {
          const pool = getPool();

          // 检查邮箱是否已存在
          const existingUser = await this.getByEmail(email);
          if (existingUser) {
            throw new Error('Email already exists');
          }

          const [result] = await pool.execute(
            'INSERT INTO users (name, email) VALUES (?, ?)',
            [name.trim(), email.trim()]
          );

          // 获取新创建的用户
          const newUserId = (result as any).insertId;
          const newUser = await this.getById(newUserId);

          if (!newUser) {
            throw new Error('Failed to retrieve created user');
          }

          return newUser;
        } catch (error) {
          console.error('Error creating user:', error);
          if (error instanceof Error && error.message.includes('Email already exists')) {
            throw error;
          }
          throw new Error('Failed to create user');
        }
      }

      // 更新用户
      async update(id: number, data: UpdateUserInput): Promise<User | null> {
        if (id <= 0) {
          throw new Error('Invalid user ID');
        }

        const { name, email } = data;
        const updates: string[] = [];
        const values: any[] = [];

        if (name !== undefined) {
          if (!name?.trim()) {
            throw new Error('Name cannot be empty');
          }
          updates.push('name = ?');
          values.push(name.trim());
        }

        if (email !== undefined) {
          if (!email?.trim() || !email.includes('@')) {
            throw new Error('Valid email is required');
          }
          // 检查邮箱是否已被其他用户使用
          const existingUser = await this.getByEmail(email);
          if (existingUser && existingUser.id !== id) {
            throw new Error('Email already exists');
          }
          updates.push('email = ?');
          values.push(email.trim());
        }

        if (updates.length === 0) {
          // 没有需要更新的字段
          return await this.getById(id);
        }

        try {
          const pool = getPool();
          values.push(id);

          await pool.execute(
            `UPDATE users SET ${updates.join(', ')} WHERE id = ?`,
            values
          );

          return await this.getById(id);
        } catch (error) {
          console.error(`Error updating user ${id}:`, error);
          if (error instanceof Error && error.message.includes('Email already exists')) {
            throw error;
          }
          throw new Error('Failed to update user');
        }
      }

      // 删除用户
      async delete(id: number): Promise<boolean> {
        if (id <= 0) {
          throw new Error('Invalid user ID');
        }

        try {
          const pool = getPool();
          const [result] = await pool.execute(
            'DELETE FROM users WHERE id = ?',
            [id]
          );

          const affectedRows = (result as any).affectedRows;
          return affectedRows > 0;
        } catch (error) {
          console.error(`Error deleting user ${id}:`, error);
          throw new Error('Failed to delete user');
        }
      }
    }

    // 导出单例实例
    export const usersDB = UsersDB.getInstance();

    // 为了向后兼容，也导出函数式接口
    export const db = {
      getAll: () => usersDB.getAll(),
      getById: (id: number) => usersDB.getById(id),
      getByEmail: (email: string) => usersDB.getByEmail(email),
      create: (name: string, email: string) => usersDB.create({ name, email }),
      update: (id: number, data: Partial<User>) => usersDB.update(id, data),
      delete: (id: number) => usersDB.delete(id),
    };
    ```


## 3.1 API接口
### 3.1.1 基础内容

1. 基本规则

    - 所有 API 路由必须放在 `src/routes/api/` 目录下  
    - 文件路径自动映射为 URL 路径  
    - 支持嵌套、动态段、通配符等

2. 示例结构

    ```
    src/routes/api/
    ├── hello.ts                → GET /api/hello
    ├── users.ts                → GET /api/users
    ├── users/[id].ts           → GET /api/users/:id
    ├── posts/[...slug].ts      → GET /api/posts/any/nested/path
    └── auth/
        ├── login.ts            → POST /api/auth/login
        └── logout.ts           → POST /api/auth/logout
    ```

    > 注意：SolidStart 会自动将 `.ts`/`.tsx` 文件编译为服务器端点，**无需额外配置**。


3. api上下文

    | 属性 | 类型 | 说明 |
    |------|------|------|
    | `request` | `Request` | 原始 Fetch API Request 对象 |
    | `params` | `Record<string, string \| string[]>` | 动态路由参数 |
    | `locals` | `Record<string, any>` | 中间件注入的本地数据（如用户信息） |
    | `url` | `URL` | 当前请求的完整 URL（`new URL(request.url)`） |


4. 文件内容

    
    在同一个文件中导出不同方法的处理函数：

    ```ts
    // src/routes/api/todos/[id].ts

    // GET /api/todos/123
    export async function GET({ params }) {
      // ...
    }

    // PUT /api/todos/123
    export async function PUT({ request, params }) {
      const data = await request.json();
      // 更新逻辑
    }

    // DELETE /api/todos/123
    export async function DELETE({ params }) {
      // 删除逻辑
    }

    // PATCH /api/todos/123
    export async function PATCH({ request, params }) {
      const updates = await request.json();
      // 部分更新
    }
    ```

    > 未实现的方法会自动返回 `405 Method Not Allowed`。

3. API接口

    ```ts
    // src/routes/api/users.ts
    import { db } from "~/lib/db/users";
    import type { APIEvent } from "@solidjs/start/server";

    export async function GET() {
      try {
        const users = await db.getAll();
        return new Response(JSON.stringify(users), {
          headers: { "Content-Type": "application/json" }
        });
      } catch (error) {
        const message = error instanceof Error ? error.message : "获取用户列表失败";
        return new Response(JSON.stringify({ error: message }), {
          status: 500,
          headers: { "Content-Type": "application/json" }
        });
      }
    }

    export async function POST({ request }: APIEvent) {
      try {
        const body = await request.json();
        const { name, email } = body;

        const newUser = await db.create(name, email);
        return new Response(JSON.stringify(newUser), {
          status: 201,
          headers: { "Content-Type": "application/json" }
        });
      } catch (error) {
        const message = error instanceof Error ? error.message : "创建用户失败";
        const status = message.includes("邮箱已存在") ? 409 : 400;
        return new Response(JSON.stringify({ error: message }), {
          status,
          headers: { "Content-Type": "application/json" }
        });
      }
    }

    export async function PUT({ request }: APIEvent) {
      try {
        const url = new URL(request.url);
        const idStr = url.searchParams.get("id");
        const id = Number(idStr);

        if (!idStr || isNaN(id)) {
          return new Response(JSON.stringify({ error: "缺少或无效的 id 参数" }), {
            status: 400,
            headers: { "Content-Type": "application/json" }
          });
        }

        const body = await request.json();
        const updatedUser = await db.update(id, body);

        if (!updatedUser) {
          return new Response(JSON.stringify({ error: "用户不存在" }), {
            status: 404,
            headers: { "Content-Type": "application/json" }
          });
        }

        return new Response(JSON.stringify(updatedUser), {
          headers: { "Content-Type": "application/json" }
        });
      } catch (error) {
        const message = error instanceof Error ? error.message : "更新用户失败";
        const status = message.includes("邮箱已存在") ? 409 : 400;
        return new Response(JSON.stringify({ error: message }), {
          status,
          headers: { "Content-Type": "application/json" }
        });
      }
    }

    export async function DELETE({ request }: APIEvent) {
      try {
        const url = new URL(request.url);
        const idStr = url.searchParams.get(" id");
        const id = Number(idStr);

        if (!idStr || isNaN(id)) {
          return new Response(JSON.stringify({ error: "缺少或无效的 id 参数" }), {
            status: 400,
            headers: { "Content-Type": "application/json" }
          });
        }

        const deleted = await db.delete(id);
        if (!deleted) {
          return new Response(JSON.stringify({ error: "用户不存在" }), {
            status: 404,
            headers: { "Content-Type": "application/json" }
          });
        }

        return new Response(null, { status: 204 });
      } catch (error) {
        const message = error instanceof Error ? error.message : "删除用户失败";
        return new Response(JSON.stringify({ error: message }), {
          status: 400,
          headers: { "Content-Type": "application/json" }
        });
      }
    }
    ```

    ```jsx
        } else {
    ```

## 3.2 server数据获取

> 直接使用API路由,`createResource`中 fetch 里要写全部名称，因为createResource是ssr渲染，
无法去组装路由，可以使用Server中query与action

1. 核心对比

    | 特性 | `query` / `action` | `/src/routes/api` 路由 |
    |------|-------------------|------------------------|
    | 设计理念 | 全栈一体化（服务端函数） | 传统 REST API |
    | 调用方式 | 直接函数调用（类型安全） | HTTP 请求（fetch/axios） |
    | 数据序列化 | 自动处理（无需 JSON.stringify/parse） | 手动处理（req/res） |
    | 类型安全 |  完整 TypeScript 支持 |  需要手动定义类型 |
    | 缓存 | Query 自动缓存 |  需要手动实现 |
    | 错误处理 |  统一错误格式 |  需要手动处理 |
    | 身份验证 |  直接访问请求上下文 |  需要解析 headers/cookies |
    | 开发体验 |  极佳（无缝集成） |  传统方式 |
    | 网络请求 |  内部优化（可能无网络请求） |  始终是网络请求 |
    | 适用场景 | 推荐用于 SolidStart 应用 | 兼容传统 API 或外部调用 |

2. 核心概念

    | 功能 | `query` | `action` |
    |------|---------|----------|
    | 用途 | 数据获取（GET） | 数据操作（POST/PUT/DELETE） |
    | HTTP 方法 | GET | POST（默认），可配置其他方法 |
    | 缓存 | 自动缓存 | 不缓存 |

3. 接口文件

    ```ts
    // src/routes/api/users.ts
    import { query,action } from "@solidjs/router";
    import { db } from "~/server/db/users";

    // 查询：获取所有用户
    export const getUsers = query(async () => {
      return await db.getAll();
    },"AllUser");

    // 操作：创建用户
    export const createUser = action(async (data: { name: string; email: string }) => {
      try {
        const user = await db.create(data.name, data.email);
        return { success: true, user };
      } catch (error) {
        const message = error instanceof Error ? error.message : "创建用户失败";
        return { success: false, error: message };
      }
    });

    // 操作：更新用户（通过 id + 数据）
    export const updateUser = action(async (payload: { id: number; name?: string; email?:   string }) => {
      try {
        const { id, ...updateData } = payload;
        const user = await db.update(id, updateData);
        if (!user) {
          return { success: false, error: "用户不存在" };
        }
        return { success: true, user };
      } catch (error) {
        const message = error instanceof Error ? error.message : "更新用户失败";
        return { success: false, error: message };
      }
    });

    // 操作：删除用户
    export const deleteUser = action(async (id: number) => {
      try {
        const deleted = await db.delete(id);
        if (!deleted) {
          return { success: false, error: "用户不存在" };
        }
        return { success: true };
      } catch (error) {
        const message = error instanceof Error ? error.message : "删除用户失败";
        return { success: false, error: message };
      }
    });
    ```

## 3.3 中间件


中间件拦截HTTP请求和响应，以执行身份验证、重定向、日志记录等任务。它还支持使用event.locals对象在应用程序中共享请求范围内的数据。


1. 新建配置文件

    ```ts
    // src/middleware/index.ts
    import { createMiddleware } from "@solidjs/start/middleware";

    export default createMiddleware({
      onRequest: (event) => {
        console.log("Request received:", event.request.url);

        event.locals.startTime = Date.now();
      },
      onBeforeResponse: (event) => {
        const endTime = Date.now();
        const duration = endTime - event.locals.startTime;
        console.log(`Request took ${duration}ms`);
      },
    });
    ```
2. 声明配置文件

    ```ts
    // app.config.ts
    import { defineConfig } from "@solidjs/start/config";

    export default defineConfig({
        middleware: "src/middleware/index.ts",
    });
    ```
