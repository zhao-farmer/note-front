### 3.1.2 回调函数（子→父通信）

1. 代码示例
    - Child.svelte

        ```svelte
        <script>
          // 接收回调函数
          let { 
            onButtonClick,
            onDataUpdate
          } = $props();

          let count = $state(0);

          function handleClick() {
            count++;

            // 调用父组件传递的回调
            onButtonClick?.(count);

            onDataUpdate?.({
              count,
              message: '按钮被点击了',
              timestamp: Date.now()
            });
          }
        </script>

        <button onclick={handleClick}>
          点击: {count}
        </button>
        ```
    - Parent.svelte

        ```svelte
        <script>
          import Child from './Child.svelte';

          let childClicks = 0;
          let lastEvent = null;

          // 传递给子组件的回调函数
          function handleChildClick(count) {
            console.log('子组件点击了', count, '次');
            childClicks = count;
          }

          function handleChildData(data) {
            console.log('收到子组件数据:', data);
            lastEvent = data;
          }
        </script>

        <Child
          onButtonClick={handleChildClick}
          onDataUpdate={handleChildData}
        />

        <p>子组件点击次数: {childClicks}</p>
        <pre>{JSON.stringify(lastEvent, null, 2)}</pre>
        ```

## 3.2 Context API（上下文）


>  **注意**：Svelte 4 及之前**没有官方 Context API**。  
> Svelte 5（2024 年发布）引入了 `setContext` / `getContext`。

- 祖先组件 `App.svelte`

    ```svelte
    <script>
      import { setContext } from 'svelte';
      setContext('theme', 'dark');
      import Parent from './lib/Parent.svelte'
    </script>

    <main>
        <Parent />
    </main>
    ```
- 中间组件 `Parent.svelte`

    ```svelte
    <script>
        import Child from './Child.svelte';
    </script>

    <Child />
    ```
- 后代组件 `Child.svelte`

    ```svelte
    <script>
        import { getContext } from 'svelte';
        const theme = getContext('theme'); // 'dark'
    </script>

    <span>主题颜色：{theme}</span>
    ```

## 3.3 Store 状态管理

### 3.3.1 Svelte 4 → Svelte 5 迁移对比

| 特性  | Svelte 4 Store | Svelte 5 Store | 优势  |
| --- | --- | --- | --- |
| **创建方式**​ | `writable()`, `readable()`, `derived()` | 直接使用 `$state()`, `$derived()` | 更简单，无额外 API |
| **访问方式**​ | `$store`语法糖 | 直接访问属性 | 更直观 |
| **更新方式**​ | `store.set()`, `store.update()` | 直接赋值 | 更像普通变量 |
| **订阅**​ | 自动订阅/手动订阅 | 无订阅概念 | 自动响应式 |
| **类型安全**​ | 中等  | 优秀  | 编译时检查 |
| **模块化**​ | 需要 import/export | 天然模块化 | 更灵活 |

> 注意：
>
> 1. `Svelte 5` 的 runes 只能用于 .svelte、.svelte.js或 .svelte.ts文件
> 2. `export let`与 `$state` 有冲突,不能写 `export let count = $state(0);`

### 3.3.2 示例

1. 共享响应式状态 `stores/todoStore.svelte.js`

    ```javascript
    // 创建可共享的响应式状态
	let todos = $state([]);
	let filter = $state('all');
	
	// 计算属性
	let filteredTodos = $derived({
	  get value() {
	    switch (filter) {
	      case 'active':
	        return todos.filter(t => !t.completed);
	      case 'completed':
	        return todos.filter(t => t.completed);
	      default:
	        return todos;
	    }
	  }
	});
	
	export function getTodos() {
	  return todos;
	}
	export function getFilteredTodos() {
	  return filteredTodos;
	}
	
	// 操作方法
	export function addTodo(text) {
	  todos.push({
	    id: Date.now(),
	    text,
	    completed: false
	  });
	}
	
	export function toggleTodo(id) {
	  const todo = todos.find(t => t.id === id);
	  if (todo) {
	    todo.completed = !todo.completed;
	  }
	}
    ```

2. 在组件中使用store 


	- TodoList.svelte

		```svelte
		<script>
		    import { getFilteredTodos, addTodo, toggleTodo } from "../stores/todoStore.svelte.js";
		</script>

		<input
		    type="text"
		    placeholder="添加新任务"
		    onkeydown={ e => {
		        if (e.key === "Enter" && e.currentTarget.value.trim()) {
		            addTodo(e.currentTarget.value);
		            e.currentTarget.value = "";
		        }
		    }}
		/>

		<ul>
		    {#each getFilteredTodos().value as todo (todo.id)}
		        <li>
		            <input type="checkbox" checked={todo.completed} onchange={() => toggleTodo		(todo.id)} />
		            {todo.text}
		        </li>
		    {/each}
		</ul>
		```
	- TodoStats.svelte

		```svelte
		<script>
		    import { getTodos } from '../stores/todoStore.svelte.js';
		    let todos = getTodos();
		    let total = $derived(todos.length);
		    let completed = $derived(todos.filter(t => t.completed).length);
		    let remaining = $derived(total - completed);
		</script>

		<p>总计: {total}</p>
		<p>已完成: {completed}</p>
		<p>进行中: {remaining}</p>
		```
    - App.svelte

		```svelte
		<script>  
		  import TodoList from "./lib/TodoList.svelte";
		  import TodoStats from "./lib/TodoStats.svelte";
		</script>

		<main>
		  <TodoList />
		  <TodoStats />
		</main>
		```

## 3.4 插槽通信

### 3.4.1 Svelte 4 → Svelte 5 迁移对比

| 特性  | Svelte 4 (旧语法) | Svelte 5 (新语法) |
| --- | --- | --- |
| **语法**​ | `<slot name="header" />` | `{@render header()}` |
| **作用域**​ | `<slot {data} />` | `{@render children(data)}` |
| **接收**​ | `let:data` | 通过函数参数传递 |
| **默认内容**​ | `<slot>默认内容</slot>` | 在 `$props`中设置默认值 |
| **类型安全**​ | 弱   | 强   |

### 3.4.2 基础插槽

1. 代码示例

	- Parent.svelte

		```svelte
		<script>
		  import Child from './Child.svelte';
		</script>

		<Child>
		  {#snippet header()}
		    <h2>自定义标题</h2>
		  {/snippet}

		  {#snippet children(data)}
		    <p>接收到的数据: {data}</p>
		  {/snippet}
		</Child>
		```
	- Child.svelte

		```svelte
		<script>
		  let { header, children } = $props();
		  let data = '这是数据';
		</script>

		<div>
		  <!-- 渲染命名插槽（带默认值） -->
		  {#if header}
		    {@render header()}
		  {:else}
		    <div>默认标题</div>
		  {/if}

		  <!-- 渲染作用域插槽 -->
		  {@render children?.(data)}
		</div>
		```

### 3.4.3 动态插槽

1. 代码示例

	- Parent.svelte

		```svelte
		<script>
		  import Child from './Child.svelte';
		</script>

		<Child>
		  {#snippet header()}
		    <header>网站头部</header>
		  {/snippet}

		  {#snippet sidebar()}
		    <aside>侧边栏</aside>
		  {/snippet}

		  {#snippet main()}
		    <main>主要内容</main>
		  {/snippet}

		  {#snippet footer()}
		    <footer>页脚</footer>
		  {/snippet}  
		</Child>
		```
	- Child.svelte

		```svelte
		<script>
		  // Svelte 5 会自动将所有 {#snippet} 转换为 $props() 中的函数
		  let { header, sidebar, main, footer } = $props();

		  // 或者更灵活地处理
  		  const areas = $derived([
  		    { name: 'header', content: header },
  		    { name: 'sidebar', content: sidebar },
  		    { name: 'main', content: main },
  		    { name: 'footer', content: footer }
  		  ]);
		</script>

		<div class="layout">
		  {#each areas as { name, content }}
		    {#if content}
		      <div class={`area-${name}`}>
		        {@render content()}
		      </div>
		    {/if}
		  {/each}
		</div>
		```

## 3.5 DOM 引用通信

1. 方式的不同
	- svelet4 `bind:this`
	- svelet5  `$state + 显式 ref`


2. 代码示例
	- Child.svelte

		```svelte
		<script>
		  let { ref } = $props(); // 接收父组件传来的 ref 对象

		  let value = $state('');

		  // 在 $effect 中暴露 API
		  $effect(() => {
		    if (ref) {
		      ref.current = {
		        reset: () => {
		          value = '';
		          console.log('子组件已重置');
		        },
		        getData: () => ({
		          value,
		          timestamp: Date.now()
		        })
		      };
		    }

		    // 清理：组件销毁时清空引用
		    return () => {
		      if (ref) ref.current = null;
		    };
		  });
		</script>

		<input 
		  type="text" 
		  value={value}
		  oninput={(e) => value = e.currentTarget.value}
		/>
		```

	- Parent.svelte

		```svelte
		<script>
		  import Child from './Child.svelte';
		  import { onMount } from 'svelte';

		  // 创建 ref 容器（必须是对象，以便按引用传递）
		  let childRef = $state({ current: null });
		  let childData = $state('');

		  onMount(() => {
		    if (childRef.current) {
		      console.log('子组件已挂载');
		      childRef.current.reset();
		    }
		  });

		  function getDataFromChild() {
		    if (childRef.current) {
		      const data = childRef.current.getData();
		      console.log('获取到子组件数据:', data);
		      childData = JSON.stringify(data, null, 2);
		    }
		  }
		</script>

		<!-- 将 ref 作为普通 prop 传入 -->
		<Child ref={childRef} />

		<button onclick={getDataFromChild}>
		  获取子组件数据
		</button>
		<button onclick={() => childRef.current?.reset()}>
		  重置子组件
		</button>

		<pre>{childData}</pre>
		```
