## 1.1 填充图案

### 1.1.1 canvas的createPattern处理

1. 代码示例

	```html
	<!DOCTYPE html>
	<html lang="zh-CN">

	<head>
	  <meta charset="UTF-8">
	</head>

	<body>

	  <canvas id="demoCanvas" width="600" height="200"></canvas>

	  <script>
	    const canvas = document.getElementById('demoCanvas');
	    const ctx = canvas.getContext('2d');

	    // 生成纹理：对角线第一条线 → 向两侧平行偏移
	    function createSolidLineTile() {
	      const size = 200;
	      const tile = document.createElement('canvas');
	      const tCtx = tile.getContext('2d');

	      tile.width = size;
	      tile.height = size;

	      // 背景
	      // tCtx.fillStyle = '#333';
	      tCtx.fillRect(0, 0, size, size);

	      // 线条样式
	      tCtx.strokeStyle = '#fff';
	      tCtx.lineWidth = 2;
	      const spacing = 10; // 平行线间距 要是size能整除的数

	      tCtx.beginPath();

	      // 1. 先画【第一条线：主对角线】（最长线）
	      tCtx.moveTo(0, 0);
	      tCtx.lineTo(size, size);

	      // 1. 向【左上】位移画线
	      for (let offset = -spacing; offset > -size; offset -= spacing) {
	        tCtx.moveTo(0 + offset, 0);
	        tCtx.lineTo(size + offset, size);
	      }

	      // 3. 向【右下】位移画线
	      for (let offset = spacing; offset < size; offset += spacing) {
	        tCtx.moveTo(0 + offset, 0);
	        tCtx.lineTo(size + offset, size);
	      }

	      tCtx.stroke();
	      return tile;
	    }

	    // 填充重复纹理
	    const tile = createSolidLineTile();
	    const pattern = ctx.createPattern(tile, 'repeat');

	    ctx.fillStyle = pattern;
	    ctx.fillRect(0, 0, canvas.width, canvas.height);
	  </script>

	</body>

	</html>
	```

### 1.1.3 使用TilingSprite处理

1. 代码示例

	```html
	<!DOCTYPE html>
	<html lang="zh-CN">

	<head>
	  <meta charset="UTF-8" />
	  <title>PIXI8 使用瓷砖精灵</title>
	  <script src="./js/pixi.js"></script>
	</head>

	<body>

	  <script>
	    (async () => {
	      // 1. 初始化PIXI应用
	      const app = new PIXI.Application();
	      await app.init({
	        width: 600,
	        height: 300,
	        backgroundColor: 0x1a1a1a,
	      });
	      document.body.appendChild(app.canvas);


	      // 1.生成绝对无缝的45°斜纹纹理
	      function createSolidLineTile() {
	        const size = 200;
	        const tile = document.createElement('canvas');
	        const tCtx = tile.getContext('2d');

	        tile.width = size;
	        tile.height = size;

	        //  不加背景,canvas默认会有黑色背景

	        // 线条样式
	        tCtx.strokeStyle = '#fff';
	        tCtx.lineWidth = 2;
	        const spacing = 10; // 平行线间距 要是size能整除的数

	        tCtx.beginPath();

	        // 1. 先画【第一条线：主对角线】（最长线）
	        tCtx.moveTo(0, 0);
	        tCtx.lineTo(size, size);

	        // 1. 向【左上】位移画线
	        for (let offset = -spacing; offset > -size; offset -= spacing) {
	          tCtx.moveTo(0 + offset, 0);
	          tCtx.lineTo(size + offset, size);
	        }

	        // 3. 向【右下】位移画线
	        for (let offset = spacing; offset < size; offset += spacing) {
	          tCtx.moveTo(0 + offset, 0);
	          tCtx.lineTo(size + offset, size);
	        }

	        tCtx.stroke();
	        return tile;
	      }

	      // 3. 正确加载纹理到PIXI
	      const tileCanvas = createSolidLineTile();
	      const texture = PIXI.Texture.from(tileCanvas);

	      // 3. 绘制图形
	      const graphics = new PIXI.TilingSprite({
	        texture,
	        width: 600,
	        height: 100,
	      });

	      graphics.y = 100
	      app.stage.addChild(graphics);
	    })();
	  </script>
	</body>

	</html>
	```

### 1.1.4 FillPattern图案填充

1. 代码示例

	```html
	<!DOCTYPE html>
	<html lang="zh-CN">

	<head>
	  <meta charset="UTF-8" />
	  <title>PIXI8 完美45°斜纹填充</title>
	</head>

	<body>
	  <script src="./js/pixi.js"></script>
	  <script>
	    (async () => {
		
	      const config = {
	        size: 150,  // 纹理大小
	        showWidth: 600, // 显示宽度
	        showHeight: 100, // 显示高度
	        spacing: 150 / 20, // 平行线间距 
	      }

	      // 1. 初始化PIXI应用
	      const app = new PIXI.Application();
	      await app.init({
	        width: 600,
	        height: 300,
	        backgroundColor: 0x1a1a1a,
	        antialias: false 
	      });
	      document.body.appendChild(app.canvas);

	      // 1.生成绝对无缝的45°斜纹纹理
	      function createSolidLineTile(size) {

	        const tile = document.createElement('canvas');
	        const tCtx = tile.getContext('2d');

	        tile.width = size;
	        tile.height = size;

	        // 不加背景,canvas默认会有黑色背景

	        // 线条样式 
	        tCtx.strokeStyle = '#fff';
	        tCtx.lineWidth = 1;
	        const spacing = config.spacing;

	        tCtx.beginPath();

	        // 1. 先画【第一条线：主对角线】（最长线）
	        tCtx.moveTo(0, 0);
	        tCtx.lineTo(size, size);

	        // 1. 向【左上】位移画线
	        for (let offset = -spacing; offset > -size; offset -= spacing) {
	          tCtx.moveTo(0 + offset, 0);
	          tCtx.lineTo(size + offset, size);
	        }

	        // 3. 向【右下】位移画线
	        for (let offset = spacing; offset < size; offset += spacing) {
	          tCtx.moveTo(0 + offset, 0);
	          tCtx.lineTo(size + offset, size);
	        }

	        tCtx.stroke();
	        return tile;
	      }

	      // 3. 正确加载纹理到PIXI
	      const tileCanvas = createSolidLineTile(config.size);
	      const texture = PIXI.Texture.from(tileCanvas);

	      // 4. 创建填充图案 
	      const fillPattern = new PIXI.FillPattern(texture, 'repeat');
	      // 配置放大，使纹理与显示区域等比例匹配
	      fillPattern.transform = new PIXI.Matrix().scale(config.size / config.	showWidth, config.size / config.showHeight);

	      // 5. 绘制图形
	      const graphics = new PIXI.Graphics();
	      graphics.rect(0, 100, config.showWidth, config.showHeight);
	      graphics.fill(fillPattern);
	      app.stage.addChild(graphics);
	    })();
	  </script>
	</body>

	</html>
	```

### 1.1.5 graphics 直接画

1. 代码示例

	```html
	<!DOCTYPE html>
	<html lang="zh-CN">

	<head>
	  <meta charset="UTF-8" />
	  <title>PIXI8 直接画全屏斜线</title>
	</head>

	<body>
	  <script src="./js/pixi.js"></script>
	  <script>
	    (async () => {
	      // 1. 初始化 PIXI 应用
	      const app = new PIXI.Application();
	      await app.init({
	        width: 600,
	        height: 300,
	        backgroundColor: 0x1a1a1a,
	      });
	      document.body.appendChild(app.canvas);


	      const config = {
	        step: 10,     // 线条间距
	        lineWidth: 2, // 线条粗细
	        color: 0xffffff, // 线条颜色
	        height: 100, // 线条高度
	        width: app.screen.width, // 线条宽度
	      }

	      // 1. 创建图形
	      const container = new PIXI.Container();
	      container.y = 100;
	      app.stage.addChild(container);



	      // ==========================================
	      // 核心：直接画全屏 45° 斜线（无纹理、无图片、不踩坑）
	      const graphics = new PIXI.Graphics();

	      // 铺满屏幕的斜线
	      for (let x = -config.height; x < config.width; x += config.step) {
	        graphics.moveTo(x, 0);
	        graphics.lineTo(x + config.height, config.height);
	      }
	      graphics.stroke({ width: config.lineWidth, color: config.color });

	      container.addChild(graphics);
	    })();
	  </script>
	</body>

	</html>
	```

## 1.2 容器拉伸

### 1.2.1 图片延申-基础

1. 代码示例

	```html
	<!DOCTYPE html>
	<html lang="zh-CN">

	<head>
	  <meta charset="UTF-8">
	  <title>PixiJS 8 图片重复填充 + 拉伸</title>
	  <script src="./js/pixi.js"></script>
	</head>

	<body>
	  <script type="module">
	    (async () => {
	      // 1. 初始化 Pixi8
	      const app = new PIXI.Application();
	      await app.init({ width: 800, height: 300, background: '#111' });
	      document.body.appendChild(app.canvas);

	      // 1. 新建容器
	      const container = new PIXI.Container();
	      app.stage.addChild(container);
	      container.y = 100;

	      // 初始宽度 & 拉伸状态
	      let maxWidth = app.screen.width - 200;
	      let isResizing = false;

	      // 3. 添加背景
	      const graphics = new PIXI.Graphics();
	      container.addChild(graphics);

	      // 4. 获取图片纹理
	      const texture = await PIXI.Assets.load('./img/1.png');

	      // 5. 创建 TilingSprite
	      let targetHeight = 60;
	      const tilingSprite = new PIXI.TilingSprite(texture, maxWidth, targetHeight);
	      tilingSprite.y = 20;
	      container.addChild(tilingSprite);

	      // 缩放到目标大小
	      const textureScale = targetHeight / texture.height;
	      tilingSprite.tileScale.set(textureScale);

	      // 6. 创建拉伸手柄并完成拉伸功能
	      const handle = new PIXI.Graphics();
	      container.addChild(handle);
	      handle.eventMode = 'static';
	      handle.cursor = 'ew-resize';

	      // 初始化渲染
	      update();

	      // 拖拽开始
	      handle.onpointerdown = () => {
	        isResizing = true;
	      };

	      // 拖拽结束（绑定到 canvas）
	      app.canvas.addEventListener('pointerup', () => {
	        isResizing = false;
	      });

	      // 拖拽移动（绑定到 canvas）
	      app.canvas.addEventListener('pointermove', (e) => {
	        if (!isResizing) return;

	        // 把鼠标坐标转成容器内的局部坐标
	        const rect = app.canvas.getBoundingClientRect();
	        const global = new PIXI.Point(
	          e.clientX - rect.left,
	          e.clientY - rect.top
	        );
	        const localX = container.toLocal(global).x;

	        // 更新宽度（限制最小100px）
	        maxWidth = Math.max(100, localX);
	        update();
	      });

	      // 更新视图
	      function update() {
	        // 1. 更新背景
	        graphics.clear();
	        graphics.rect(0, 0, maxWidth, 100);
	        graphics.fill(0xc34288);

	        // 1. 更新平铺宽度
	        tilingSprite.width = maxWidth;

	        // 3. 更新手柄位置（白色小方块）
	        handle.clear();
	        handle.rect(maxWidth - 10, 35, 10, 30);
	        handle.fill(0xffffff);
	      }
	    })();
	  </script>
	</body>

	</html>
	```

### 1.2.2 图片延申-调整比例

1. 代码示例

	```html
	<!DOCTYPE html>
	<html lang="zh-CN">
	
	<head>
	  <meta charset="UTF-8">
	  <title>PixiJS 8 图片重复填充 + 拉伸</title>
	  <script src="./js/pixi.js"></script>
	</head>
	
	<body>
	  <script type="module">
	    (async () => {
	      // 1. 初始化 Pixi8
	      const app = new PIXI.Application();
	      await app.init({ width: 800, height: 300, background: '#111' });
	      document.body.appendChild(app.canvas);
	
	      // 1. 新建容器
	      const container = new PIXI.Container();
	      app.stage.addChild(container);
	      container.y = 100;
	
	      // 初始宽度 & 拉伸状态
	      let maxWidth = app.screen.width - 200;
	      let isResizing = false;
	
	      // 3. 添加背景
	      const graphics = new PIXI.Graphics();
	      container.addChild(graphics);
	
	      // 4. 获取图片纹理
	      const texture = await PIXI.Assets.load('./img/1.png');
	
	      // 5. 创建 TilingSprite
	      let targetHeight = 60;
	      const tilingSprite = new PIXI.TilingSprite(texture, maxWidth, targetHeight);
	      tilingSprite.y = 20;
	      container.addChild(tilingSprite);
	
	      // 缩放到目标大小
	      setTilingSpriteCover(tilingSprite, 100, targetHeight);
	
	      // 6. 创建拉伸手柄并完成拉伸功能
	      const handle = new PIXI.Graphics();
	      container.addChild(handle);
	      handle.eventMode = 'static';
	      handle.cursor = 'ew-resize';
	
	      // 初始化渲染
	      update();
	
	      // 拖拽开始
	      handle.onpointerdown = () => {
	        isResizing = true;
	      };
	
	      // 拖拽结束（绑定到 canvas）
	      app.canvas.addEventListener('pointerup', () => {
	        isResizing = false;
	      });
	
	      // 拖拽移动（绑定到 canvas）
	      app.canvas.addEventListener('pointermove', (e) => {
	        if (!isResizing) return;
	
	        // 把鼠标坐标转成容器内的局部坐标
	        const rect = app.canvas.getBoundingClientRect();
	        const global = new PIXI.Point(
	          e.clientX - rect.left,
	          e.clientY - rect.top
	        );
	        const localX = container.toLocal(global).x;
	
	        // 更新宽度（限制最小100px）
	        maxWidth = Math.max(100, localX);
	        update();
	      });
	
	      // 更新视图
	      function update() {
	        // 1. 更新背景
	        graphics.clear();
	        graphics.rect(0, 0, maxWidth, 100);
	        graphics.fill(0xc34288);
	
	        // 1. 更新平铺宽度
	        tilingSprite.width = maxWidth;
	
	        // 3. 更新手柄位置（白色小方块）
	        handle.clear();
	        handle.rect(maxWidth - 10, 35, 10, 30);
	        handle.fill(0xffffff);
	      }
	      function setTilingSpriteCover(tilingSprite, containerW, containerH) {
	        // 原始纹理尺寸
	        const texture = tilingSprite.texture;
	        const imgW = texture.width;
	        const imgH = texture.height;
	
	        // 宽高比
	        const imgRatio = imgW / imgH;
	        const containerRatio = containerW / containerH;
	
	        // cover 核心：取更大的缩放比
	        let scale;
	        if (imgRatio > containerRatio) {
	          scale = containerH / imgH;
	        } else {
	          scale = containerW / imgW;
	        }
	
	        // 应用缩放（TilingSprite 专用）
	        tilingSprite.tileScale.set(scale);
	
	        // 计算居中偏移（和 CSS object-position: center 一致）
	        const offsetX = (containerW - imgW * scale) / 2;
	        const offsetY = (containerH - imgH * scale) / 2;
	        tilingSprite.tilePosition.set(offsetX, offsetY);
	      }
	    })();
	  </script>
	</body>
	
	</html>
	```
