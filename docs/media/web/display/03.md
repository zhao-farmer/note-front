## 3.1 视频纹理

### 3.1.1 基础使用

1. 代码示例

    ```html
    <!DOCTYPE html>
    <html>
    <head>
      <meta charset="UTF-8">
      <title>PIXI Video 基础使用</title>
    </head>
    <body>
      <script src="./js/pixi.js"></script>
      <script>
        (async () => {
          // 1. 初始化应用
          const app = new PIXI.Application();
          await app.init({
            width: 800,
            height: 600,
            backgroundColor: 0x222222,
            antialias: true
          });
          document.body.appendChild(app.canvas);

          // 2. 加载视频
          const videoTexture = await PIXI.Assets.load('./video/demo.mp4');
          // 这个video是HTML5的video元素
          const video = videoTexture.source.resource;

          // 3. 创建视频精灵
          const sprite = new PIXI.Sprite(videoTexture);
          sprite.anchor.set(0.5);
          sprite.x = app.screen.width / 2;
          sprite.y = app.screen.height / 2;
          sprite.scale.set(0.6);
          app.stage.addChild(sprite);

          // 3. 视频基础配置
          video.loop = true;          // 循环
          video.muted = true;        // 静音
          video.volume = 1;           // 音量
          video.playbackRate = 2;     // 播放速度
          video.autoplay = true;      // 自动播放

          // 5. 播放
          video.play();

        })();
      </script>
    </body>
    </html>
    ```

### 3.1.3 播放时点击抽帧

1. 代码示例

    ```html
    <!DOCTYPE html>
    <html>

    <head>
      <meta charset="UTF-8">
      <title>Pixi 视频抽帧</title>
      <script src="./js/pixi.js"></script>
    </head>

    <body>

      <script>
        (async () => {
          // 1. 初始化 Pixi 应用
          const app = new PIXI.Application();
          await app.init({ width: 800, height: 600, backgroundColor: 0x222222 });
          document.body.appendChild(app.canvas);

          // 2. 加载视频
          const videoTexture = await PIXI.Assets.load('./video/demo.mp4');
          const video = videoTexture.source.resource;
          video.loop = true;
          video.muted = true;
          video.play();

          // 3. 创建视频精灵
          const sprite = new PIXI.Sprite(videoTexture);
          sprite.anchor.set(0.5);
          sprite.x = app.screen.width * 0.5;
          sprite.y = app.screen.height * 0.5;
          sprite.scale.set(0.5);
          app.stage.addChild(sprite);


          function captureFrame() {
            const w = video.videoWidth;
            const h = video.videoHeight;

            // 1. 创建临时容器 + 临时精灵（不修改原精灵）
            const tempContainer = new PIXI.Container();
            const tempSprite = new PIXI.Sprite(videoTexture);
            tempContainer.addChild(tempSprite);

            // 2. 渲染到纹理
            const frameTex = PIXI.RenderTexture.create({ width: w, height: h });
            app.renderer.render(tempContainer, { renderTexture: frameTex });

            // 3. 清理临时对象
            tempSprite.destroy();
            tempContainer.destroy();

            console.log("成功截取一帧");
            return frameTex;
          }

          // 3.点击抽帧
          document.onclick = () => {
            const oneFrame = captureFrame();
            const frameSprite = new PIXI.Sprite(oneFrame);
            frameSprite.scale.set(0.15);
            frameSprite.x = 10;
            frameSprite.y = 10;
            app.stage.addChild(frameSprite);
          };

        })();
      </script>
    </body>

    </html>
    ```

### 3.1.4 抽帧列表展示

1. 代码示例

    ```html
    <!DOCTYPE html>
    <html>
    
    <head>
      <meta charset="UTF-8">
      <title>视频平均抽取18帧 + 排成一条线</title>
    </head>

    <body>
      <script src="./js/pixi.js"></script>
      <script>
        (async () => {
          // 1. 初始化应用
          const app = new PIXI.Application();
          await app.init({ width: 800, height: 300, backgroundColor: 0x222222 });
          document.body.appendChild(app.canvas);

          // 2. 加载视频
          const videoTexture = await PIXI.Assets.load('./video/demo.mp4');
          const video = videoTexture.source.resource;
          video.muted = true;

          // 3. 创建视频精灵（隐藏，不显示）
          const sprite = new PIXI.Sprite(videoTexture);
          sprite.alpha = 0;
          app.stage.addChild(sprite);

          // 抽帧（修复版：不改变位置）
          function captureFrame() {
            const w = video.videoWidth;
            const h = video.videoHeight;

            // 关键：用临时精灵渲染，不影响原精灵
            const tempContainer = new PIXI.Container();
            const tempSprite = new PIXI.Sprite(videoTexture);
            tempContainer.addChild(tempSprite);

            const frameTex = PIXI.RenderTexture.create({ width: w, height: h });
            app.renderer.render(tempContainer, { renderTexture: frameTex });

            tempSprite.destroy();
            tempContainer.destroy();
            return frameTex;
          }

          // 抽取帧的方法
          async function extractAverageFrames(total = 18) {
            const frames = [];
            const duration = video.duration;
            const step = duration / total;

            for (let i = 0; i < total; i++) {
              video.currentTime = i * step;

              await new Promise(resolve => {
                video.onseeked = resolve;
              });

              const frameTex = captureFrame();
              frames.push(frameTex);
              console.log(`已截取第 ${i + 1}/${total} 帧`);
            }

            console.log(` ${total}帧截取完成！`);
            return frames;
          }

          // 排成一条横线（整齐排列）
          function showFramesInLine(frames) {
            const startX = 15;       // 起始X
            const startY = 100;       // 高度Y（整条线统一）
            const scale = 0.08;      // 缩小倍数
            const spacing = 8;       // 图片间距

            frames.forEach((tex, i) => {
              const s = new PIXI.Sprite(tex);
              s.scale.set(scale);
              s.x = startX + i * (s.width + spacing);
              s.y = startY;
              app.stage.addChild(s);
            });
          }

          // 执行
          const t0 = performance.now();
          const frames = await extractAverageFrames(200);
          showFramesInLine(frames);
          const t1 = performance.now();
          const costSeconds = (t1 - t0) / 1000;
          console.log(`extractAndShowFrames 执行耗时：${costSeconds.toFixed(3)} 秒`);

        })();
      </script>
    </body>

    </html>
    ```

### 3.2.3 纹理复制解决

1. 代码示例

	```html
	<!DOCTYPE html>
	<html>

	<head>
	  <meta charset="UTF-8">
	  <title>VideoFrame 抽帧优化版 (RenderTexture 高性能版)</title>
	</head>

	<body>
	  <script src="./js/pixi.js"></script>

	  <!-- 隐藏 video 标签 -->
	  <video id="video" src="./video/demo.mp4" muted playsinline crossorigin="anonymous"
	    style="position:absolute;top:-1000px;left:-1000px;"></video>

	  <script>
	    (async () => {
	      // 1. 初始化 PIXI (v8)
	      const app = new PIXI.Application();
	      await app.init({
	        width: 800,
	        height: 300,
	        backgroundColor: 0x222222,
	        antialias: true
	      });
	      document.body.appendChild(app.canvas);

	      const video = document.getElementById('video');
	      await new Promise(resolve => { video.onloadedmetadata = resolve; });

	      // 视频原始尺寸
	      const videoWidth = video.videoWidth;
	      const videoHeight = video.videoHeight;

	      // 统一配置
	      const SPRITE_CONFIG = {
	        startX: 15,
	        startY: 100,
	        scale: 0.08,
	        spacing: 8
	      };

	      // 计算缩略图最终在舞台上占据的宽度（用于计算横向排版）
	      const thumbnailWidth = videoWidth * SPRITE_CONFIG.scale;

	      // 【核心优化】使用 RenderTexture 离屏烘焙
	      async function captureFrameToSprite(index) {
	        // 1. 根据当前视频状态创建 VideoFrame
	        const vf = new VideoFrame(video, {
	          timestamp: video.currentTime * 1e6
	        });

	        // 2. 用当前帧的 VideoFrame 创建临时纹理与临时精灵
	        const tempTexture = PIXI.Texture.from(vf);
	        const tempSprite = new PIXI.Sprite(tempTexture);

	        // 3. 创建一块等大、独立的离屏渲染纹理（RenderTexture）
	        const renderTexture = PIXI.RenderTexture.create({
	          width: videoWidth,
	          height: videoHeight
	        });

	        // 3. 将临时精灵渲染到 RenderTexture 中（数据在 GPU 内部完成拷贝）
	        app.renderer.render({
	          container: tempSprite,
	          target: renderTexture
	        });

	        // 5. 关键：解除引用并立即销毁临时纹理，同时关闭 VideoFrame 释放内存
	        tempSprite.destroy();
	        tempTexture.destroy(true); // 内部会解除对 vf 的显存占用
	        vf.close();                // 立即关闭，释放 WebCodecs 内存

	        // 6. 使用已经烘焙好图像数据的 renderTexture 创建最终要显示的精灵
	        const finalSprite = new PIXI.Sprite(renderTexture);

	        // 计算排版位置
	        finalSprite.x = SPRITE_CONFIG.startX + index * (thumbnailWidth + SPRITE_CONFIG.	spacing);
	        finalSprite.y = SPRITE_CONFIG.startY;
	        finalSprite.scale.set(SPRITE_CONFIG.scale);

	        app.stage.addChild(finalSprite);
	      }

	      // 抽帧并显示
	      async function extractAndShowFrames(total = 6) {
	        const duration = video.duration;
	        const step = duration / total;

	        for (let i = 0; i < total; i++) {
	          // 计算时间点（避免超出总时长）
	          video.currentTime = Math.min(i * step, duration - 0.1);
	```

	```html
	        startY: 100,
	        scale: 0.08,
	        spacing: 8
	      };

	      /**
	       * 【核心函数】将 VideoFrame 烘焙到 RenderTexture
	       * @param {VideoFrame} frame - 原始视频帧
	       * @param {number} index - 当前帧索引（用于计算位置）
	       */
	      function bakeAndCreateSprite(frame, index) {
	        // 1. 创建临时纹理和精灵用于“中转”
	        // 注意：这里必须使用 frame，PixiJS v8 支持直接从 VideoFrame 创建 TextureSource
	        const tempTexture = PIXI.Texture.from(frame);
	        const tempSprite = new PIXI.Sprite(tempTexture);

	        // 2. 创建目标 RenderTexture (显存画布)
	        // 尺寸应与视频帧实际尺寸一致，保证清晰度

	          width: frame.displayWidth,
	          height: frame.displayHeight
	        });

	        const finalSprite = new PIXI.Sprite(renderTexture);

	        // 计算位置 (使用预设的缩略图宽度逻辑，避免动态获取宽度的误差)
	        // 假设我们按原始比例缩放，这里简化计算逻辑
	        const thumbWidth = frame.displayWidth * SPRITE_CONFIG.scale;

	        finalSprite.x = SPRITE_CONFIG.startX + index * (thumbWidth + SPRITE_CONFIG.	spacing);
	        finalSprite.y = SPRITE_CONFIG.startY;
	        finalSprite.scale.set(SPRITE_CONFIG.scale);

	          bakeAndCreateSprite(videoFrame, i);
	```

	```html
	        const tempTexture = PIXI.Texture.from(frame);
	        const tempSprite = new PIXI.Sprite(tempTexture);

	        const renderTexture = PIXI.RenderTexture.create({
	          width: frame.displayWidth,
	          height: frame.displayHeight
	        });
	        app.renderer.render({
	          container: tempSprite,
	          target: renderTexture
	        });

	        tempSprite.destroy();
	        tempTexture.destroy(true);
	        return new PIXI.Sprite(renderTexture);
	      }


	      /**
	       * @param {number} total 总共划分多少等分帧
	       * @param {number} frameStart 起始下标包含
	       * @param {number} frameEnd 结束下标包含
	       */
	      async function extractFrames(total, frameStart, frameEnd) {
	        if(frameStart < 0) frameStart = 0;
	        if(frameEnd >= total) frameEnd = total -1;
	        if(frameStart > frameEnd){
	          throw new Error(`参数错误 frameStart(${frameStart}) > frameEnd(${frameEnd})`);

	        const decoderConfig = await videoTrack.getDecoderConfig();
	        if (!decoderConfig) throw new Error("获取decoderConfig失败");
	        const { supported } = await VideoDecoder.isConfigSupported(decoderConfig);
	        if (!supported) throw new Error("浏览器解码器配置不支持");


	        const indexQueue = [];
	        const sink = new EncodedPacketSink(videoTrack);

	        const decoder = new VideoDecoder({
	          output: (videoFrame) => {
	            const idx = indexQueue.shift();
	            if (idx !== -1) {
	              const spr = bakeFrameToSprite(videoFrame);
	              thumbSprites[idx] = spr;
	            }
	            videoFrame.close();
	          },
	          error: (err) => console.error("decode error", err)
	        });
	        decoder.configure(decoderConfig);


	        const stepTime = duration / total;
	        let targetIndex = frameStart;
	        let nextTargetTime = targetIndex * stepTime;
	        let gotKeyframe = false;


	        for await (const pkt of sink.packets()) {
	          const chunk = pkt.toEncodedVideoChunk();

	          if (!gotKeyframe) {
	            if (chunk.type === "key") {
	              decoder.decode(chunk);
	              indexQueue.push(-1);
	              gotKeyframe = true;
	            }
	            continue;
	          }

	          let useIndex = -1;
	          const ptsSec = chunk.timestamp / 1_000_000;
	          if (ptsSec >= nextTargetTime && targetIndex <= frameEnd) {
	            useIndex = targetIndex;
	            targetIndex += 1;
	            nextTargetTime = targetIndex * stepTime;
	          }

	          decoder.decode(chunk);
	          indexQueue.push(useIndex);

	          if (targetIndex > frameEnd) break;
	        }

	        await decoder.flush();
	        decoder.close();

	          const cellMask = new PIXI.Graphics();
	          cellMask.rect(0,0, cellW, cellImgH).fill(0xffffff);
	          cellContainer.addChild(cellMask);
	          cellContainer.mask = cellMask;

	          // 计算 cover 的scale
	          const texW = spr.texture.width;
	          const texH = spr.texture.height;
	          const texAspect = texW / texH;
	          const cellAspect = cellW / cellImgH;

	          let coverScale;
	          if(texAspect > cellAspect){
	            // 图片更宽：以高度对齐，左右裁切
	            coverScale = cellImgH / texH;
	          }else{
	            // 图片更高：以宽度对齐，上下裁切
	            coverScale = cellW / texW;
	          }
	          spr.scale.set(coverScale);

	          // 居中
	          spr.x = (cellW - spr.width) / 2;
	          spr.y = (cellImgH - spr.height) / 2;

	          cellContainer.addChild(spr);

	          // 单元格放到轨道上，垂直位置受paddingV控制
	          cellContainer.x = offsetX;
	          cellContainer.y = paddingV;

	        maskGraphics.rect(0, 0, viewWidth, trackHeight).fill(0xffffff);

	        handle.rect(viewWidth - 10, handleY, 10, handleH).fill(0xffffff);

	      });

	        const rect = app.canvas.getBoundingClientRect();
	        const global = new PIXI.Point(e.clientX - rect.left, e.clientY - rect.top);
	        const localX = clipContainer.toLocal(global).x;
	        viewWidth = Math.max(100, Math.min(totalThumbWidth, localX));
	```

### 3.4.6 pixi处理视频

1. 代码示例

	```html
	<!DOCTYPE html>
	<html>
	<head>
	  <meta charset="UTF-8">
	  <title>原生Video seek + Pixi cover模式缩略图</title>
	  <script src="./js/pixi.js"></script>
	</head>
	<body>
	  <script type="module">
	    (async () => {
	      const app = new PIXI.Application();
	      await app.init({
	        width: 800,
	        height: 220,
	        backgroundColor: 0x222222,
	        antialias: true
	      });
	      document.body.appendChild(app.canvas);

	      const clipContainer = new PIXI.Container();
	      clipContainer.y = 50;
	      app.stage.addChild(clipContainer);

	      let isResizing = false;
	      let viewWidth = 600;
	      let totalThumbWidth = 0;

	      const TRACK_CONFIG = {
	        trackHeight: 100,
	        trackWidth: 120,
	        spacing: 4,
	        paddingV: 8
	      };
	      const { trackWidth, trackHeight, spacing, paddingV } = TRACK_CONFIG;
	      const cellImgH = trackHeight - paddingV * 2;
	      const cellW = trackWidth;

	      const bgGraphics = new PIXI.Graphics();
	      clipContainer.addChild(bgGraphics);

	      const thumbContainer = new PIXI.Container();
	      clipContainer.addChild(thumbContainer);

	      const maskGraphics = new PIXI.Graphics();
	      maskGraphics.rect(0, 0, viewWidth, trackHeight).fill(0xffffff);
	      thumbContainer.addChild(maskGraphics);
	      thumbContainer.mask = maskGraphics;

	      const handle = new PIXI.Graphics();
	      clipContainer.addChild(handle);
	      handle.eventMode = 'static';
	      handle.cursor = 'ew-resize';

	      /** @type {PIXI.Sprite[]} */
	      const thumbSprites = [];

	      // -------- 复用示例里的 captureFrame --------
	      function captureFrame(video) {
	        const w = video.videoWidth;
	        const h = video.videoHeight;
	        const tempContainer = new PIXI.Container();
	        const tempSprite = new PIXI.Sprite(PIXI.Texture.from(video));
	        tempContainer.addChild(tempSprite);

	        const frameTex = PIXI.RenderTexture.create({ width: w, height: h });
	        app.renderer.render({
	          container: tempContainer,
	          target: frameTex
	        });

	        tempSprite.destroy();
	        tempContainer.destroy();
	        return frameTex;
	      }

	      /**
	       * @param {number} total 总共划分多少等分帧
	       * @param {number} frameStart 起始下标包含
	       * @param {number} frameEnd 结束下标包含
	       */
	      async function extractFrames(total, frameStart, frameEnd) {
	        if (frameStart < 0) frameStart = 0;
	        if (frameEnd >= total) frameEnd = total - 1;
	        if (frameStart > frameEnd) {
	          throw new Error(`参数错误 frameStart(${frameStart}) > frameEnd(${frameEnd})`);
	        }

	        // 原生video加载，替代MediaBunny
	        const videoTexture = await PIXI.Assets.load('./video/demo.mp4');
	        const video = videoTexture.source.resource;
	        video.muted = true;

	        const duration = video.duration;
	        const dw = video.videoWidth;
	        const dh = video.videoHeight;
	        console.log('视频原始比例 dw/dh', dw, dh);

	        const stepTime = duration / total;
	        thumbSprites.length = 0;

	        for (let i = frameStart; i <= frameEnd; i++) {
	          const targetTimeSec = i * stepTime;
	          video.currentTime = targetTimeSec;
	          await new Promise(resolve => {
	            video.onseeked = resolve;
	          });

	          const frameTex = captureFrame(video);
	          const spr = new PIXI.Sprite(frameTex);
	          thumbSprites[i] = spr;
	          console.log(`已截取 ${i}/${total}`);
	        }

	        // cover模式渲染逻辑完全保留
	        thumbContainer.removeChildren();
	        thumbContainer.addChild(maskGraphics);

	        let offsetX = 0;
	        for (let i = frameStart; i <= frameEnd; i++) {
	          const spr = thumbSprites[i];
	          if (!spr) continue;

	          const cellContainer = new PIXI.Container();
	          cellContainer.width = cellW;
	          cellContainer.height = cellImgH;

	          const cellMask = new PIXI.Graphics();
	          cellMask.rect(0, 0, cellW, cellImgH).fill(0xffffff);
	          cellContainer.addChild(cellMask);
	          cellContainer.mask = cellMask;

	          const texW = spr.texture.width;
	          const texH = spr.texture.height;
	          const texAspect = texW / texH;
	          const cellAspect = cellW / cellImgH;

	          let coverScale;
	          if (texAspect > cellAspect) {
	            coverScale = cellImgH / texH;
	          } else {
	            coverScale = cellW / texW;
	          }
	          spr.scale.set(coverScale);

	          spr.x = (cellW - spr.width) / 2;
	          spr.y = (cellImgH - spr.height) / 2;
	          cellContainer.addChild(spr);

	          cellContainer.x = offsetX;
	          cellContainer.y = paddingV;
	          thumbContainer.addChild(cellContainer);

	          offsetX += cellW + spacing;
	        }

	        totalThumbWidth = offsetX;
	        console.log(`extractFrames完成 [${frameStart}-${frameEnd}] totalWidth`, totalThumbWidth);
	      }

	      function updateView() {
	        bgGraphics.clear();
	        bgGraphics.rect(0, 0, viewWidth, trackHeight).fill(0x333355);

	        maskGraphics.clear();
	        maskGraphics.rect(0, 0, viewWidth, trackHeight).fill(0xffffff);

	        const handleH = 30;
	        const handleY = (trackHeight - handleH) / 2;
	        handle.clear();
	        handle.rect(viewWidth - 10, handleY, 10, handleH).fill(0xffffff);
	      }

	      handle.onpointerdown = () => {
	        isResizing = true;
	      };
	      app.canvas.addEventListener('pointerup', () => {
	        isResizing = false;
	      });
	      app.canvas.addEventListener('pointermove', (e) => {
	        if (!isResizing) return;
	        const rect = app.canvas.getBoundingClientRect();
	        const global = new PIXI.Point(e.clientX - rect.left, e.clientY - rect.top);
	        const localX = clipContainer.toLocal(global).x;
	        viewWidth = Math.max(100, Math.min(totalThumbWidth, localX));
	        updateView();
	      });

	      await extractFrames(20, 5, 15);
	      updateView();
	    })();
	  </script>
	</body>
	</html>

	```
