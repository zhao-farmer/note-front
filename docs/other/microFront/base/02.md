# 二、创建项目

>我使用pnpm安装项目，速度会比较快，如果未安装进行安装

```sh
npm install pnpm -g
```

## 2.1 创建vue2项目(webpack)

1. 安装vue2

    ```sh
    # 安装插件
    npm install -g @vue/cli
    # 创建项目
    vue create vue2-webpack
    # 选择vue2与pnpm安装
    ```
2. 添加端口配置 `vue.config.js`

    ```js{4-6}
    const { defineConfig } = require('@vue/cli-service')
    module.exports = defineConfig({
      transpileDependencies: true,
      devServer: {
        port: 6001
      }
    })
    ```
3. App.vue修改

    ```vue
    <template>
      <div id="app">
         <h1>Vue 2.6.14 + webpack 项目</h1>
         <p>这个项目用于测试微前端</p>
      </div>
    </template>

    <script>

    export default {
      name: 'App',
    }
    </script>

    <style>
    #app {
      display: flex;
      flex-direction: column;
      align-items: center;
      justify-content: center;  
    }
    h1 {
      color: #42b983;
      font-size: 24px;
    }
    p {
      font-size: 18px;
    }
    </style>
    ```
4. 启动项目

    ```sh
    npm run serve
    ```

    ![](/other/microFront/base/001.png)

## 2.3 创建react17项目(类组件+webpack)

1. 创建react17项目

    ```sh
    # 新建文件夹
    mkdir react17-webpack
    
    # 使用 pnpm 初始化项目
    cd react17-webpack  && npm init -y

    # 安装 React 17
    pnpm add react@17 react-dom@17

    # 安装类型定义（如果是 TypeScript 项目）
    pnpm add -D @types/react @types/react-dom

    # Webpack 核心
    pnpm add -D webpack webpack-cli webpack-dev-server

    # Babel 相关（用于 JSX/TS 转换）
    pnpm add -D @babel/core @babel/preset-env @babel/preset-react
    pnpm add -D babel-loader

    # CSS 和样式处理
    pnpm add -D style-loader css-loader mini-css-extract-plugin

    # 插件
    pnpm add -D html-webpack-plugin clean-webpack-plugin 
    ```

    **配置文件package.json**

    ```json
    {
     "name": "react-webpack",
     "version": "1.0.0",
     "description": "",
     "scripts": {
       "start": "webpack serve --mode development",
       "build": "webpack --mode production"
     },
     "dependencies": {
       "react": "^17.0.2",
       "react-dom": "^17.0.2"
     },
     "devDependencies": {
       "@babel/core": "^7.28.5",
       "@babel/preset-env": "^7.28.5",
       "@babel/preset-react": "^7.28.5",
       "@types/react": "^17.0.2",
       "@types/react-dom": "^17.0.2",
       "babel-loader": "^10.0.0",
       "clean-webpack-plugin": "^4.0.0",
       "css-loader": "^7.1.2",
       "html-webpack-plugin": "^5.6.5",
       "mini-css-extract-plugin": "^2.9.4",
       "style-loader": "^4.0.0",
       "webpack": "^5.104.1",
       "webpack-cli": "^6.0.1",
       "webpack-dev-server": "^5.2.2"
     }
    }
    ```

2. 创建配置文件

    - webpack.config.js

        ```js
        const path = require('path');
        const HtmlWebpackPlugin = require('html-webpack-plugin');
        const MiniCssExtractPlugin = require('mini-css-extract-plugin');

        module.exports = {
          // 入口文件
          entry: './src/index.js',

          // 输出
          output: {
            path: path.resolve(__dirname, 'dist'),
            filename: 'bundle.js',
          },

          // 开发服务器
          devServer: {
            static: {
              directory: path.join(__dirname, 'public'),
            },
            port: 6003,
            hot: true,
            open: true,
          },

          // 模块规则
          module: {
            rules: [
              {
                test: /\.(js|jsx)$/,
                exclude: /node_modules/,
                use: {
                  loader: 'babel-loader',
                },
              },
              {
                test: /\.css$/,
                use: [MiniCssExtractPlugin.loader, 'css-loader'],
              },
            ],
          },

          // 插件
          plugins: [
            new HtmlWebpackPlugin({
              template: './public/index.html',
            }),
            new MiniCssExtractPlugin({
              filename: 'bundle.css',
            }),
          ],

          // 解析扩展名
          resolve: {
            extensions: ['.js', '.jsx'],
          },

        };
        ```
    - babel.config.js

        ```js
        module.exports = {
          presets: [
            '@babel/preset-env',
            ['@babel/preset-react', { runtime: 'automatic' }]
          ],
        };
        ```
3. 新增文件

    - src/main.js

        ```js
        import React from 'react';
        import ReactDOM from 'react-dom';
        import App from './App';

        const rootElement = document.getElementById('root');
        ReactDOM.render(<App />, rootElement);
        ```
    - src/App.jsx

        ```jsx
        import { Component } from 'react'
        import './App.css'

        export default class App extends Component {
          render() {
            return (
              <div id='app'>
                <h1>React 17.0.2 + webpack 项目</h1>
                <p>这个项目用于测试微前端</p>
              </div>
            )
          }
        }
        ```
    - src/App.css

        ```css
        #app {
          display: flex;
          flex-direction: column;
          align-items: center;
          justify-content: center;  
        }
        h1 {
          color: #7f42b9;
          font-size: 24px;
        }
        p {
          font-size: 18px;
        }
        ```
    - public/index.html

        ```html
        <!DOCTYPE html>
        <html lang="zh-CN">
        <head>
          <meta charset="UTF-8">
          <meta name="viewport" content="width=device-width, initial-scale=1.0">
          <title>React 17 + Webpack + pnpm</title>
        </head>
        <body>
          <div id="root"></div>
          <!-- webpack 会自动插入打包的脚本 -->
        </body>
        </html>
        ```
4. 启动项目

    ```sh
    npm run start
    ```
