# 一、通用时间工具类

## 1. JS版本

```js
/**
 * 时间工具类 - 纯 JavaScript 实现
 */
export const TimeUtils = {
  /**
   * 格式化日期为字符串
   * 支持模板：yyyy, MM, dd, HH, mm, ss, SSS
   * @param {Date | number | string} date - 日期对象、时间戳或字符串
   * @param {string} format - 格式模板，默认 'yyyy-MM-dd'
   * @returns {string}
   */
  format(date, format = 'yyyy-MM-dd') {
    const d = date instanceof Date ? date : new Date(date);
    if (isNaN(d.getTime())) throw new Error('Invalid date');

    const pad = (num, width = 2) => String(num).padStart(width, '0');

    const year = d.getFullYear();
    const month = d.getMonth() + 1;
    const day = d.getDate();
    const hours = d.getHours();
    const minutes = d.getMinutes();
    const seconds = d.getSeconds();
    const milliseconds = d.getMilliseconds();

    return format
      .replace(/yyyy/g, String(year))
      .replace(/MM/g, pad(month))
      .replace(/dd/g, pad(day))
      .replace(/HH/g, pad(hours))
      .replace(/mm/g, pad(minutes))
      .replace(/ss/g, pad(seconds))
      .replace(/SSS/g, pad(milliseconds, 3));
  },

  /**
   * 解析日期字符串为 Date 对象
   * @param {string} str - 日期字符串（如 '2025-01-01', '2025/1/1'）
   * @returns {Date | null}
   */
  parse(str) {
    if (!str) return null;
    const iso = new Date(str);
    if (!isNaN(iso.getTime())) return iso;

    const match = str.match(/^(\d{4})[-/]?(\d{1,2})[-/]?(\d{1,2})$/);
    if (match) {
      const [, y, m, d] = match;
      const date = new Date(Number(y), Number(m) - 1, Number(d));
      return isNaN(date.getTime()) ? null : date;
    }
    return null;
  },

  /**
   * 判断两个日期是否是同一天
   */
  isSameDay(a, b) {
    const da = a instanceof Date ? a : new Date(a);
    const db = b instanceof Date ? b : new Date(b);
    return (
      da.getFullYear() === db.getFullYear() &&
      da.getMonth() === db.getMonth() &&
      da.getDate() === db.getDate()
    );
  },

  /**
   * 获取今天（时分秒归零）
   */
  today() {
    const now = new Date();
    return new Date(now.getFullYear(), now.getMonth(), now.getDate());
  },

  /**
   * 添加/减去时间
   * @param {Date | string | number} date - 基准日期
   * @param {number} amount - 数量（可为负）
   * @param {'day'|'month'|'year'|'hour'|'minute'} unit - 单位
   * @returns {Date}
   */
  add(date, amount, unit) {
    const d = date instanceof Date ? new Date(date) : new Date(date);
    switch (unit) {
      case 'year':
        d.setFullYear(d.getFullYear() + amount);
        break;
      case 'month':
        d.setMonth(d.getMonth() + amount);
        break;
      case 'day':
        d.setDate(d.getDate() + amount);
        break;
      case 'hour':
        d.setHours(d.getHours() + amount);
        break;
      case 'minute':
        d.setMinutes(d.getMinutes() + amount);
        break;
      default:
        throw new Error(`Unsupported unit: ${unit}`);
    }
    return d;
  },

  /**
   * 计算两个日期相差天数
   */
  diffInDays(from, to) {
    const fromDate = from instanceof Date ? from : new Date(from);
    const toDate = to instanceof Date ? to : new Date(to);
    const timeDiff = toDate.getTime() - fromDate.getTime();
    return Math.floor(timeDiff / (1000 * 60 * 60 * 24));
  },

  /**
   * 是否为今天
   */
  isToday(date) {
    return this.isSameDay(date, this.today());
  },

  /**
   * 是否为过去的时间
   */
  isPast(date) {
    return new Date(date).getTime() < Date.now();
  },

  /**
   * 是否为未来的时间
   */
  isFuture(date) {
    return new Date(date).getTime() > Date.now();
  },

  /**
   * 返回友好时间描述（如“3分钟前”、“昨天”）
   */
  fromNow(date) {
    const d = date instanceof Date ? date : new Date(date);
    const now = new Date();
    const diffMs = now.getTime() - d.getTime();

    if (diffMs < 0) return '未来';

    const seconds = Math.floor(diffMs / 1000);
    const minutes = Math.floor(seconds / 60);
    const hours = Math.floor(minutes / 60);
    const days = Math.floor(hours / 24);

    if (days > 7) return this.format(d, 'yyyy-MM-dd');
    if (days >= 2) return `${days}天前`;
    if (days === 1) return '昨天';
    if (hours >= 1) return `${hours}小时前`;
    if (minutes >= 1) return `${minutes}分钟前`;
    return '刚刚';
  },

  /**
   * 验证日期是否有效
   */
  isValid(date) {
    if (date instanceof Date) return !isNaN(date.getTime());
    if (typeof date === 'string' || typeof date === 'number') {
      const d = new Date(date);
      return !isNaN(d.getTime());
    }
    return false;
  }
};
```

## 2. TS版本

```ts
/**
 * 时间工具类 
 */
export class TimeUtils {
  /**
   * 格式化日期为指定字符串
   * @param date Date 对象或时间戳（ms）或 ISO 字符串
   * @param format 格式模板，支持：yyyy, MM, dd, HH, mm, ss, SSS
   * @example TimeUtils.format(new Date(), 'yyyy-MM-dd HH:mm:ss') → "2026-01-20 19:34:05"
   */
  static format(date: Date | number | string, format = 'yyyy-MM-dd'): string {
    const d = date instanceof Date ? date : new Date(date);
    if (isNaN(d.getTime())) throw new Error('Invalid date');

    const pad = (num: number, width = 2): string => String(num).padStart(width, '0');

    const year = d.getFullYear();
    const month = d.getMonth() + 1;
    const day = d.getDate();
    const hours = d.getHours();
    const minutes = d.getMinutes();
    const seconds = d.getSeconds();
    const milliseconds = d.getMilliseconds();

    return format
      .replace(/yyyy/g, String(year))
      .replace(/MM/g, pad(month))
      .replace(/dd/g, pad(day))
      .replace(/HH/g, pad(hours))
      .replace(/mm/g, pad(minutes))
      .replace(/ss/g, pad(seconds))
      .replace(/SSS/g, pad(milliseconds, 3));
  }

  /**
   * 解析日期字符串为 Date 对象
   * @param str 日期字符串（支持 ISO、'yyyy-MM-dd'、'yyyy/MM/dd' 等）
   * @returns Date | null（无效时返回 null）
   */
  static parse(str: string): Date | null {
    if (!str) return null;
    // 尝试 ISO 或标准格式
    const iso = new Date(str);
    if (!isNaN(iso.getTime())) return iso;

    // 尝试 yyyy-MM-dd 或 yyyy/MM/dd
    const match = str.match(/^(\d{4})[-/]?(\d{1,2})[-/]?(\d{1,2})$/);
    if (match) {
      const [, y, m, d] = match;
      const date = new Date(Number(y), Number(m) - 1, Number(d));
      return isNaN(date.getTime()) ? null : date;
    }
    return null;
  }

  /**
   * 判断两个日期是否是同一天
   */
  static isSameDay(a: Date | string | number, b: Date | string | number): boolean {
    const da = a instanceof Date ? a : new Date(a);
    const db = b instanceof Date ? b : new Date(b);
    return (
      da.getFullYear() === db.getFullYear() &&
      da.getMonth() === db.getMonth() &&
      da.getDate() === db.getDate()
    );
  }

  /**
   * 获取今天（Date 对象，时分秒归零）
   */
  static today(): Date {
    const now = new Date();
    return new Date(now.getFullYear(), now.getMonth(), now.getDate());
  }

  /**
   * 添加/减去时间（天、月、年等）
   * @param date 基准日期
   * @param amount 数量（正数为加，负数为减）
   * @param unit 单位：'day' | 'month' | 'year' | 'hour' | 'minute'
   */
  static add(
    date: Date | string | number,
    amount: number,
    unit: 'day' | 'month' | 'year' | 'hour' | 'minute'
  ): Date {
    const d = date instanceof Date ? new Date(date) : new Date(date);
    switch (unit) {
      case 'year':
        d.setFullYear(d.getFullYear() + amount);
        break;
      case 'month':
        d.setMonth(d.getMonth() + amount);
        break;
      case 'day':
        d.setDate(d.getDate() + amount);
        break;
      case 'hour':
        d.setHours(d.getHours() + amount);
        break;
      case 'minute':
        d.setMinutes(d.getMinutes() + amount);
        break;
      default:
        throw new Error(`Unsupported unit: ${unit}`);
    }
    return d;
  }

  /**
   * 计算两个日期之间的天数差
   */
  static diffInDays(from: Date | string | number, to: Date | string | number): number {
    const fromDate = from instanceof Date ? from : new Date(from);
    const toDate = to instanceof Date ? to : new Date(to);
    const timeDiff = toDate.getTime() - fromDate.getTime();
    return Math.floor(timeDiff / (1000 * 60 * 60 * 24));
  }

  /**
   * 判断是否为今天
   */
  static isToday(date: Date | string | number): boolean {
    return this.isSameDay(date, this.today());
  }

  /**
   * 判断是否为过去的时间
   */
  static isPast(date: Date | string | number): boolean {
    return new Date(date).getTime() < Date.now();
  }

  /**
   * 判断是否为未来的时间
   */
  static isFuture(date: Date | string | number): boolean {
    return new Date(date).getTime() > Date.now();
  }

  /**
   * 获取友好的相对时间描述（如“3分钟前”、“昨天”、“2025-12-01”）
   */
  static fromNow(date: Date | string | number): string {
    const d = date instanceof Date ? date : new Date(date);
    const now = new Date();
    const diffMs = now.getTime() - d.getTime();

    if (diffMs < 0) return '未来';

    const seconds = Math.floor(diffMs / 1000);
    const minutes = Math.floor(seconds / 60);
    const hours = Math.floor(minutes / 60);
    const days = Math.floor(hours / 24);

    if (days > 7) return this.format(d, 'yyyy-MM-dd');
    if (days >= 2) return `${days}天前`;
    if (days === 1) return '昨天';
    if (hours >= 1) return `${hours}小时前`;
    if (minutes >= 1) return `${minutes}分钟前`;
    return '刚刚';
  }

  /**
   * 验证日期字符串是否有效
   */
  static isValid(date: unknown): boolean {
    if (date instanceof Date) return !isNaN(date.getTime());
    if (typeof date === 'string' || typeof date === 'number') {
      const d = new Date(date);
      return !isNaN(d.getTime());
    }
    return false;
  }
}
```
