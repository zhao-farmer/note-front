# 二、Axios 拦截器

以下是一个 **完整、实用、可直接用于生产环境的 Axios 拦截器代码示例**，包含 **请求拦截器（** 和 **响应拦截器**，适用于 Vue、React、Angular 等主流前端框架。



功能涵盖：

- 自动携带 Token（从 localStorage / store 获取）
- 请求 loading 控制（可选）
- 请求重试（如 token 刷新后重发）
- 统一错误处理（401 未登录、500 服务器错误等）
- 响应数据解包（如 `{ data, code, msg }` 结构）


## 代码 `utils/request.js`

```js
import axios from 'axios';

// 创建独立 axios 实例（避免污染全局）
const request = axios.create({
  baseURL: process.env.VUE_APP_API_BASE_URL || '/api', // 根据你的项目配置
  timeout: 10000, // 10s 超时
  headers: {
    'Content-Type': 'application/json',
  },
});

// ======================
// 请求拦截器
// ======================
request.interceptors.request.use(
  (config) => {
    // 1. 添加认证 Token（例如从 localStorage 获取）
    const token = localStorage.getItem('access_token');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }

    // 2. 可选：显示全局 loading（如使用 Element Plus / Ant Design）
    // showLoading();

    // 3. 添加时间戳防缓存（可选）
    // if (config.method === 'get') {
    //   config.params = { ...config.params, _t: Date.now() };
    // }

    return config;
  },
  (error) => {
    // 请求发送前出错（如网络不通）
    console.error('请求拦截错误:', error);
    // hideLoading();
    return Promise.reject(error);
  }
);

// ======================
// 响应拦截器
// ======================
request.interceptors.response.use(
  (response) => {
    // hideLoading();

    // 假设后端返回格式为：{ code: 200, data: {...}, msg: "success" }
    const { code, data, msg } = response.data;

    if (code === 200) {
      return data; // 直接返回业务数据
    } else {
      // 业务逻辑错误（如参数错误、权限不足）
      console.warn('业务错误:', msg);
      // 可弹出 message 提示
      // ElMessage.warning(msg || '操作失败');
      return Promise.reject(new Error(msg || '请求失败'));
    }
  },
  async (error) => {
    // hideLoading();

    const { response, config } = error;

    if (!response) {
      // 网络错误 / 请求超时 / DNS 失败
      console.error('网络错误:', error.message);
      // ElMessage.error('网络异常，请检查连接');
      return Promise.reject(error);
    }

    const { status, data } = response;

    switch (status) {
      case 400:
        console.error('请求参数错误:', data);
        // ElMessage.error('请求参数错误');
        break;
      case 401:
        // 未授权：Token 过期或无效
        console.warn('Token 失效，跳转登录');
        localStorage.removeItem('access_token');
        // 可在此处跳转登录页
        window.location.href = '/login';
        break;
      case 403:
        // ElMessage.error('无权限访问');
        break;
      case 404:
        // ElMessage.error('接口不存在');
        break;
      case 500:
        // ElMessage.error('服务器内部错误');
        console.error('服务器错误:', data);
        break;
      default:
        // ElMessage.error(`请求失败 (${status})`);
    }

    return Promise.reject(error);
  }
);

export default request;
```

## 使用示例（在 API 模块中）

```js
// api/user.js
import request from '@/utils/request';

export function getUserInfo() {
  return request.get('/user/info');
}

export function updateUser(data) {
  return request.post('/user/update', data);
}
```

```vue
<!-- 在组件中 -->
<script setup>
import { getUserInfo } from '@/api/user';

getUserInfo().then(res => {
  console.log('用户信息:', res); // 已自动解包，res 就是 data
}).catch(err => {
  // 错误已在拦截器中处理，此处可选择是否额外处理
});
</script>
```

## 401 自动刷新 Token 并重试请求（可选）

如果你的系统支持 **refresh_token** 机制，可在 401 时自动刷新：

```js
// 在响应拦截器的 401 分支中替换为：

let isRefreshing = false;
let failedQueue = [];

const processQueue = (error, token = null) => {
  failedQueue.forEach((prom) => {
    if (error) {
      prom.reject(error);
    } else {
      prom.resolve(token);
    }
  });
  failedQueue = [];
};

if (status === 401) {
  if (isRefreshing) {
    // 正在刷新，将请求加入队列
    return new Promise((resolve, reject) => {
      failedQueue.push({ resolve, reject });
    }).then((token) => {
      config.headers.Authorization = `Bearer ${token}`;
      return request(config);
    }).catch(err => Promise.reject(err));
  }

  isRefreshing = true;
  const refreshToken = localStorage.getItem('refresh_token');

  if (refreshToken) {
    return request.post('/auth/refresh', { refreshToken })
      .then(res => {
        const { access_token, refresh_token } = res;
        localStorage.setItem('access_token', access_token);
        localStorage.setItem('refresh_token', refresh_token);

        config.headers.Authorization = `Bearer ${access_token}`;
        processQueue(null, access_token);
        return request(config); // 重试原请求
      })
      .catch(refreshError => {
        processQueue(refreshError, null);
        localStorage.clear();
        window.location.href = '/login';
        return Promise.reject(refreshError);
      })
      .finally(() => {
        isRefreshing = false;
      });
  } else {
    localStorage.clear();
    window.location.href = '/login';
    return Promise.reject(error);
  }
}
```
