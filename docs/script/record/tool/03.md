# 三、下载文件工具

## 1. 下载函数

```js
import axios from 'axios';

/**
 * 下载文件（POST 方式）
 * @param {string} url - 下载接口地址
 * @param {Object} data - 请求体参数（JSON 对象）
 * @param {string} [customFilename] - 可选：自定义文件名（优先级高于响应头）
 */
export function downloadFile(url, data = {}, customFilename = '') {
  // 安全获取 token
  let token = '';
  try {
    const userStr = localStorage.getItem('user');
    if (userStr) {
      const user = JSON.parse(userStr);
      token = user?.token || '';
    }
  } catch (e) {
    console.warn('Failed to parse user from localStorage', e);
  }

  return axios
    .post(url, data, {
      headers: {
        'Content-Type': 'application/json',
        ...(token ? { token } : {}),
      },
      responseType: 'blob',
      // withCredentials: true, // 如需携带 Cookie，取消注释
    })
    .then((res) => {
      // 检查是否是真正的文件（避免下载 HTML/JSON 错误页）
      const contentType = res.headers['content-type'];
      if (
        contentType?.includes('application/json') ||
        contentType?.includes('text/html')
      ) {
        // 尝试读取错误信息
        const reader = new FileReader();
        reader.readAsText(res.data);
        reader.onload = () => {
          try {
            const err = JSON.parse(reader.result);
            console.error('Download failed:', err);
            alert(err.msg || '文件导出失败，请重试');
          } catch {
            alert('文件导出失败，服务器返回异常');
          }
        };
        return Promise.reject(new Error('Invalid file response'));
      }

      // 1. 优先使用自定义文件名
      let filename = customFilename;

      // 2. 若无自定义名，从响应头解析
      if (!filename) {
        const disposition = res.headers['content-disposition'];
        if (disposition && disposition.indexOf('filename') !== -1) {
          // 支持 filename="xxx" 或 filename*=UTF-8''xxx
          const utf8Filename = disposition.match(/filename\*=UTF-8''(.+)/i);
          if (utf8Filename && utf8Filename[1]) {
            filename = decodeURIComponent(utf8Filename[1]);
          } else {
            const asciiFilename = disposition.match(/filename[^;=\n]*=((['"]).*?\2|[^;\n]*)/i);
            if (asciiFilename && asciiFilename[1]) {
              filename = asciiFilename[1].replace(/['"]/g, '');
            }
          }
        }
      }

      // 3. 最终 fallback
      if (!filename) {
        filename = 'downloaded-file';
      }

      // 创建 blob 并触发下载
      const blob = new Blob([res.data], { type: contentType });
      const urlBlob = URL.createObjectURL(blob);

      if ('download' in document.createElement('a')) {
        const a = document.createElement('a');
        a.href = urlBlob;
        a.download = filename;
        a.style.display = 'none';
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
      } else {
        // 兼容 IE
        navigator.msSaveBlob(blob, filename);
      }

      // 释放内存
      URL.revokeObjectURL(urlBlob);
    })
    .catch((error) => {
      console.error('Download error:', error);
      // 可在此处统一提示用户
      alert('文件下载失败，请稍后重试');
      throw error;
    });
}
```

## 2. 使用示例

```js
// 基础用法
downloadFile('/api/export/report', { userId: 123 });

// 自定义文件名
downloadFile('/api/export/data', { date: '2025-01-01' }, '月度报表.xlsx');
```
