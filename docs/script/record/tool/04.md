# 四、图片格式转换

## 1. `image-converter.js`

```js
/**
 * 图像格式转换工具类
 * 支持：URL ↔ Blob/File ↔ Base64 ↔ Object URL
 */
export const ImageConverter = {
  /**
   * 将图片 URL 转换为 File 对象（通过 fetch 或 XMLHttpRequest）
   * @param {string} url - 图片的绝对或相对 URL
   * @param {string} filename - 文件名（不含后缀，自动从 URL 或 Content-Type 推断）
   * @returns {Promise<File>}
   */
  async urlToFile(url, filename = 'image') {
    try {
      // 使用 fetch 更现代、简洁
      const response = await fetch(url, {
        headers: { Accept: 'image/*' }
      });

      if (!response.ok) {
        throw new Error(`Failed to fetch image: ${response.status} ${response.statusText}`);
      }

      const blob = await response.blob();
      const type = blob.type; // e.g., 'image/png'
      const ext = type.split('/')[1] || 'png';

      const file = new File([blob], `${filename}.${ext}`, { type });
      return file;
    } catch (error) {
      console.error('urlToFile error:', error);
      throw error;
    }
  },

  /**
   * 将图片 URL 转换为 Base64 字符串（支持跨域）
   * @param {string} url - 图片 URL
   * @returns {Promise<string>} - 返回 data URL，如 "data:image/png;base64,..."
   */
  urlToBase64(url) {
    return new Promise((resolve, reject) => {
      const img = new Image();
      img.crossOrigin = 'anonymous'; // 支持跨域
      img.src = url;

      img.onload = () => {
        const canvas = document.createElement('canvas');
        const ctx = canvas.getContext('2d');
        canvas.width = img.width;
        canvas.height = img.height;
        ctx.drawImage(img, 0, 0);

        // 自动推断格式
        let mimeType = 'image/png';
        if (url.endsWith('.jpg') || url.endsWith('.jpeg')) {
          mimeType = 'image/jpeg';
        } else if (url.endsWith('.webp')) {
          mimeType = 'image/webp';
        }

        const base64 = canvas.toDataURL(mimeType, 0.92); // JPEG 默认压缩
        resolve(base64);
      };

      img.onerror = (e) => {
        reject(new Error(`Failed to load image from URL: ${url}`));
      };
    });
  },

  /**
   * 将 Base64 字符串转换为 File 对象
   * @param {string} base64 - data URL 格式的 Base64 字符串
   * @param {string} filename - 文件名（不含后缀）
   * @returns {File}
   */
  base64ToFile(base64, filename = 'file') {
    try {
      const arr = base64.split(',');
      const mimeMatch = arr[0].match(/:(.*?);/);
      if (!mimeMatch) throw new Error('Invalid base64 format');

      const mime = mimeMatch[1];
      const suffix = mime.split('/')[1] || 'png';
      const bstr = atob(arr[1]);
      const u8arr = Uint8Array.from({ length: bstr.length }, (_, i) => bstr.charCodeAt(i));

      return new File([u8arr], `${filename}.${suffix}`, { type: mime });
    } catch (error) {
      console.error('base64ToFile error:', error);
      throw error;
    }
  },

  /**
   * 将 Base64 字符串转换为可访问的本地 URL（Object URL）
   * @param {string} base64 - data URL 格式的 Base64
   * @param {string} filename - 文件名（用于生成 File）
   * @returns {string} - 本地 Object URL，如 "blob:http://localhost:3000/..."
   */
  base64ToUrl(base64, filename = 'file') {
    const file = this.base64ToFile(base64, filename);
    return URL.createObjectURL(file);
  },

  /**
   * 【辅助】释放 Object URL 内存（使用后建议调用）
   * @param {string} objectUrl - 由 URL.createObjectURL 生成的 URL
   */
  revokeUrl(objectUrl) {
    if (objectUrl && objectUrl.startsWith('blob:')) {
      URL.revokeObjectURL(objectUrl);
    }
  }
};
```

## 2. 使用示例

1. 在 Vue / React 中使用：

    ```js
    import { ImageConverter } from '@/utils/image-converter';

    // 示例：上传前将网络图转为 File
    async function handleRemoteImageUpload(imageUrl) {
      try {
        const file = await ImageConverter.urlToFile(imageUrl, 'avatar');
        console.log('Converted file:', file);
        // 可直接 append 到 FormData
        const formData = new FormData();
        formData.append('file', file);
        await axios.post('/upload', formData);
      } catch (err) {
        console.error('转换失败', err);
      }
    }

    // 示例：预览 base64
    const base64 = await ImageConverter.urlToBase64('https://example.com/photo.jpg');
    document.getElementById('preview').src = base64;
    ```

2. 释放内存（重要！）：

    ```js
    const url = ImageConverter.base64ToUrl(base64Str);
    imgEl.src = url;

    // 组件卸载时（如 Vue onUnmounted / React useEffect cleanup）
    ImageConverter.revokeUrl(url);
    ```
