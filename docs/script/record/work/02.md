# 二、点击放大图片

## 1. 放大图片代码

```html
<!DOCTYPE html>
<html lang="zh-CN">
<head>
  <meta charset="UTF-8" />
  <meta name="viewport" content="width=device-width, initial-scale=1.0"/>
  <title>图片点击放大（纯HTML版）</title>
  <style>
    /* ========== 原 zoomImage.css 内容 ========== */
    .zoom-image-container {
      display: inline-block;
      cursor: zoom-in;
    }
    .zoom-image {
      width: 100%;
      height: auto;
      display: block;
      transition: transform 0.2s ease;
    }

    /* 放大视图样式 */
    .zoom-modal-overlay {
      position: fixed;
      top: 0;
      left: 0;
      width: 100%;
      height: 100%;
      background-color: rgba(0, 0, 0, 0.9);
      display: flex;
      justify-content: center;
      align-items: center;
      z-index: 9999;
      cursor: zoom-out;
    }
    .zoom-modal-image {
      max-width: 90%;
      max-height: 90%;
      object-fit: contain;
      transition: transform 0.1s ease;
      cursor: grab;
    }
    .zoom-modal-image:active {
      cursor: grabbing;
    }
    .zoom-modal-close {
      position: absolute;
      top: 20px;
      right: 30px;
      background: none;
      border: none;
      color: white;
      font-size: 40px;
      cursor: pointer;
      padding: 0;
      width: 40px;
      height: 40px;
      display: flex;
      align-items: center;
      justify-content: center;
      line-height: 1;
      transition: opacity 0.2s ease;
      z-index: 10000;
    }
    .zoom-modal-close:hover {
      opacity: 0.8;
    }

    /* ========== 示例页面样式 ========== */
    body {
      margin: 0;
      padding: 20px;
      background: #f5f5f5;
      font-family: Arial, sans-serif;
    }
    .gallery {
      display: flex;
      flex-wrap: wrap;
      gap: 20px;
      justify-content: center;
    }
    .zoom-image-container {
      width: 200px;
      border-radius: 8px;
      overflow: hidden;
      box-shadow: 0 2px 8px rgba(0,0,0,0.1);
    }
  </style>
</head>
<body>
  <h2>点击图片放大预览</h2>
  <div class="gallery">
    <!-- 示例图片（可替换为你自己的图片） -->
    <div class="zoom-image-container">
      <img src="https://picsum.photos/600/400?random=1" alt="示例图1" class="zoom-image">
    </div>
    <div class="zoom-image-container">
      <img src="https://picsum.photos/600/800?random=2" alt="示例图2" class="zoom-image">
    </div>
    <div class="zoom-image-container">
      <img src="https://picsum.photos/400/600?random=3" alt="示例图3" class="zoom-image">
    </div>
  </div>

  <!-- 放大模态框（初始隐藏） -->
  <div id="zoom-modal" class="zoom-modal-overlay" style="display: none;">
    <img id="zoom-modal-img" class="zoom-modal-image" src="" alt="">
    <button id="zoom-modal-close" class="zoom-modal-close" aria-label="关闭">×</button>
  </div>

  <script>
    // ========== 原 zoomImage.jsx 逻辑转为原生 JS ==========
    document.addEventListener('DOMContentLoaded', () => {
      const modal = document.getElementById('zoom-modal');
      const modalImg = document.getElementById('zoom-modal-img');
      const closeModalBtn = document.getElementById('zoom-modal-close');
      
      let transform = { scale: 1, translateX: 0, translateY: 0 };
      let isDragging = false;
      let lastMousePosition = { x: 0, y: 0 };
      let currentImgSrc = '';

      // 打开放大视图
      function openZoom(src) {
        currentImgSrc = src;
        modalImg.src = src;
        resetTransform();
        modal.style.display = 'flex';
        document.body.style.overflow = 'hidden'; // 禁止背景滚动
      }

      // 关闭放大视图
      function closeZoom() {
        modal.style.display = 'none';
        document.body.style.overflow = 'auto';
        isDragging = false;
      }

      // 重置变换状态
      function resetTransform() {
        transform = { scale: 1, translateX: 0, translateY: 0 };
        applyTransform();
      }

      // 应用变换到图片
      function applyTransform() {
        modalImg.style.transform = `scale(${transform.scale}) translate(${transform.translateX}px, ${transform.translateY}px)`;
        modalImg.style.cursor = isDragging ? 'grabbing' : 'grab';
      }

      // 滚轮缩放
      function handleWheel(e) {
        if (modal.style.display !== 'flex') return;
        e.preventDefault();
        
        // 兼容普通滚轮和 Ctrl+滚轮（macOS 触控板）
        const delta = e.deltaY > 0 ? -0.1 : 0.1;
        const newScale = Math.max(0.5, Math.min(5, transform.scale + delta));
        transform.scale = newScale;
        applyTransform();
      }

      // 鼠标按下（开始拖拽）
      function handleMouseDown(e) {
        if (e.target !== modalImg || transform.scale <= 1) return;
        e.preventDefault();
        isDragging = true;
        lastMousePosition.x = e.clientX;
        lastMousePosition.y = e.clientY;
        modalImg.style.cursor = 'grabbing';
      }

      // 鼠标移动（拖拽平移）
      function handleMouseMove(e) {
        if (!isDragging) return;
        const deltaX = e.clientX - lastMousePosition.x;
        const deltaY = e.clientY - lastMousePosition.y;
        transform.translateX += deltaX;
        transform.translateY += deltaY;
        applyTransform();
        lastMousePosition.x = e.clientX;
        lastMousePosition.y = e.clientY;
      }

      // 鼠标释放（结束拖拽）
      function handleMouseUp() {
        isDragging = false;
        if (modal.style.display === 'flex') {
          modalImg.style.cursor = 'grab';
        }
      }

      // 键盘事件（ESC 关闭）
      function handleKeyDown(e) {
        if (e.key === 'Escape' && modal.style.display === 'flex') {
          closeZoom();
        }
      }

      // 点击遮罩关闭
      function handleOverlayClick(e) {
        if (e.target === modal) {
          closeZoom();
        }
      }

      // ========== 事件绑定 ==========
      // 绑定所有缩略图点击事件
      document.querySelectorAll('.zoom-image').forEach(img => {
        img.addEventListener('click', () => {
          openZoom(img.src);
        });
      });

      // 模态框事件
      closeModalBtn.addEventListener('click', closeZoom);
      modal.addEventListener('click', handleOverlayClick);
      modalImg.addEventListener('mousedown', handleMouseDown);
      modalImg.addEventListener('dragstart', (e) => e.preventDefault()); // 禁用图片拖拽

      // 全局事件（需在 modal 打开时生效）
      document.addEventListener('wheel', handleWheel, { passive: false });
      document.addEventListener('mousemove', handleMouseMove);
      document.addEventListener('mouseup', handleMouseUp);
      document.addEventListener('keydown', handleKeyDown);
    });
  </script>
</body>
</html>
```

你提供的 HTML 文件实现了一个 **图片点击放大预览功能**，其核心是通过 **原生 JavaScript + CSS** 实现的模态框（Modal）式全屏查看器。下面我将从 **结构、样式、交互逻辑** 三个维度详细分析其实现原理。

## 2. 整体架构：模态框（Modal）模式

该功能采用经典的 **“缩略图 + 全屏模态框”** 架构：

- **缩略图区域**：页面中的 `.zoom-image-container` 容器，内含小图。
- **放大视图**：隐藏的 `#zoom-modal` 模态框，点击缩略图后显示，覆盖全屏。

这是一种轻量、通用且用户体验良好的图片查看方案。


## 3. 关键组成部分解析

1. 缩略图容器（Thumbnail）

    ```html
    <div class="zoom-image-container">
      <img src="..." class="zoom-image">
    </div>
    ```

    - **作用**：展示原始小图，作为触发入口。
    - **样式特点**：
      - `cursor: zoom-in`：鼠标悬停显示“放大镜”图标，暗示可点击。
      - 固定宽度（示例中为 200px），保持布局整齐。

    > 所有带 `.zoom-image` 的 `<img>` 都会被自动绑定点击事件。

2. 放大模态框（Modal Overlay）

    ```html
    <div id="zoom-modal" class="zoom-modal-overlay" style="display: none;">
      <img id="zoom-modal-img" class="zoom-modal-image" src="" alt="">
      <button id="zoom-modal-close">×</button>
    </div>
    ```

    - **初始状态**：`style="display: none"`，不可见。
    - **激活状态**：点击缩略图后设为 `display: flex`，居中显示大图。
    - **层级**：`z-index: 9999`，确保覆盖所有内容。

    **样式关键点**：

    | 类名 | 作用 |
    |------|------|
    | `.zoom-modal-overlay` | 全屏黑色半透明遮罩（`rgba(0,0,0,0.9)`） |
    | `.zoom-modal-image` | 大图本身，限制最大宽高（`max-width: 90%`），防止溢出 |
    | `.zoom-modal-close` | 右上角关闭按钮（×），高 `z-index` 确保可点击 |


3. 核心交互逻辑（JavaScript）

    所有交互由一段 `<script>` 实现，无任何框架依赖。

    - (1) 打开/关闭逻辑
        ```js
        // 打开
        function openZoom(src) {
          modalImg.src = src;
          resetTransform(); // 重置缩放/平移
          modal.style.display = 'flex';
          document.body.style.overflow = 'hidden'; // 锁定背景滚动
        }

        // 关闭
        function closeZoom() {
          modal.style.display = 'none';
          document.body.style.overflow = 'auto';
        }
        ```
        **为什么锁滚动？** 防止用户在放大状态下滚动页面，造成体验割裂。

    - (2) 缩放（Zoom）—— 滚轮控制
        ```js
        function handleWheel(e) {
          e.preventDefault();
          const delta = e.deltaY > 0 ? -0.1 : 0.1;
          transform.scale = Math.max(0.5, Math.min(5, transform.scale + delta));
          applyTransform();
        }
        ```
        - **缩放范围**：0.5x ~ 5x（防止无限缩小/放大）
        - **兼容性**：直接监听 `wheel` 事件，适用于鼠标滚轮和触控板。

    -  (3) 平移（Pan）—— 鼠标拖拽

        ```js
        // 按下
        handleMouseDown → isDragging = true
        // 移动
        handleMouseMove → 更新 translateX/Y
        // 松开
        handleMouseUp → isDragging = false
        ```
        - **仅在放大后启用**：`if (transform.scale <= 1) return;`
        - **光标反馈**：
          - 默认：`cursor: grab`
          - 拖拽中：`cursor: grabbing`

    - (4) 关闭方式（多入口）

        | 方式 | 实现 |
        |------|------|
        | 点击 × 按钮 | 绑定 `closeModalBtn.click` |
        | 点击遮罩背景 | `if (e.target === modal) closeZoom()` |
        | 按 ESC 键 | 监听 `keydown`，判断 `e.key === 'Escape'` |

        > 三重保障，符合用户直觉。


4. 动画与过渡效果

    - **缩略图 hover**：无（但可加 `transform: scale(1.02)` 增强反馈）
    - **大图加载**：无淡入（但可通过监听 `img.onload` 添加）
    - **变换过程**：
        ```css
        transition: transform 0.1s ease; /* 快速响应缩放/平移 */
        ```
    - 时间短（0.1s）→ 操作跟手，不拖沓

## 4. 技术亮点总结

| 特性 | 实现方式 | 优势 |
|------|--------|------|
| **零依赖** | 纯原生 JS/CSS | 轻量（<5KB），易集成 |
| **响应式** | `max-width: 90%` | 适配手机/PC |
| **无障碍** | `alt` 属性 + `aria-label` | 符合可访问性标准 |
| **性能友好** | 事件委托 + 节流（隐式） | 无内存泄漏风险 |
| **用户体验** | 多关闭方式 + 光标反馈 | 直观、容错率高 |


## 5. 如何复用到你的项目？

只需三步：

1. **复制 CSS**：将 `<style>` 中所有 `.zoom-*` 样式加入你的 CSS 文件。
2. **添加模态框 HTML**：在 `<body>` 底部加入 `#zoom-modal` 结构。
3. **初始化 JS**：将 `<script>` 内容放入你的 JS 文件，并确保在 DOM 加载后执行。

然后给任意图片加上：

```html
<div class="zoom-image-container">
  <img src="your-image.jpg" class="zoom-image">
</div>
```
即可自动获得点击放大功能！
