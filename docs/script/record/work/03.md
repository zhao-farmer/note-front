  <script>
    // ========== 状态管理 ==========
    const state = {
      croppingImage: null, // 当前裁剪的图片 DataURL
      crop: { x: 0, y: 0, width: 150, height: 150 },
      isDragging: false,
      dragStart: { x: 0, y: 0 },
      isResizing: false,
      resizeStart: { x: 0, y: 0, cropX: 0, cropY: 0, cropWidth: 0, cropHeight: 0, edge: '' },
      canvasDimensions: { width: 500, height: 500 }
    };

    // ========== DOM 引用 ==========
    const canvas = document.getElementById('canvas');
    const resultCanvas = document.getElementById('resultCanvas');
    const handelImageDiv = document.getElementById('handelImageDiv');
    const clipImgDiv = document.getElementById('clipImgDiv');
    const moveClipDiv = document.getElementById('moveClipDiv');
    const uploadTrigger = document.getElementById('uploadTrigger');
    const actionButtons = document.getElementById('actionButtons');
    const fileInput = document.getElementById('fileInput');
    const applyCropBtn = document.getElementById('applyCropBtn');
    const reuploadBtn = document.getElementById('reuploadBtn');
    const closeBtn = document.getElementById('closeBtn');

    // ========== 初始化 ==========
    function init() {
      setupCanvas();
      bindEvents();
      updateClipPosition();
    }

    // 设置 Canvas 尺寸
    function setupCanvas() {
      const rect = handelImageDiv.getBoundingClientRect();
      const size = Math.min(rect.width, 500);
      state.canvasDimensions = { width: size, height: size };
      canvas.width = size;
      canvas.height = size;
      updateInitialCrop();
    }

    // 更新初始裁剪区域（居中，30% 宽度）
    function updateInitialCrop() {
      const { width, height } = state.canvasDimensions;
      const cropSize = Math.min(150, width * 0.3);
      state.crop = {
        x: (width - cropSize) / 2,
        y: (height - cropSize) / 2,
        width: cropSize,
        height: cropSize
      };
      renderClip();
    }

    // 更新裁剪框位置和大小
    function renderClip() {
      moveClipDiv.style.left = `${state.crop.x}px`;
      moveClipDiv.style.top = `${state.crop.y}px`;
      moveClipDiv.style.width = `${state.crop.width}px`;
      moveClipDiv.style.height = `${state.crop.height}px`;
      updatePreview();
    }

    // 更新预览
    function updatePreview() {
      if (!state.croppingImage) return;
      const ctx = resultCanvas.getContext('2d');
      ctx.clearRect(0, 0, 200, 200);

      ctx.drawImage(
        canvas,
        state.crop.x,
        state.crop.y,
        state.crop.width,
        state.crop.height,
        0,
        0,
        200,
        200
      );
    }

    // 绘制图片到主 Canvas
    function drawImageToCanvas(src) {
      const ctx = canvas.getContext('2d');
      const img = new Image();
      img.onload = () => {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        const maxWidth = canvas.width;
        const maxHeight = canvas.height;
        let displayWidth = img.width;
        let displayHeight = img.height;
        if (displayWidth > maxWidth || displayHeight > maxHeight) {
          const scale = Math.min(maxWidth / displayWidth, maxHeight / displayHeight);
          displayWidth *= scale;
          displayHeight *= scale;
        }
        const x = (canvas.width - displayWidth) / 2;
        const y = (canvas.height - displayHeight) / 2;
        ctx.drawImage(img, x, y, displayWidth, displayHeight);
        updatePreview();
      };
      img.src = src;
    }

    // 更新裁剪遮罩位置
    function updateClipPosition() {
      const rect = handelImageDiv.getBoundingClientRect();
      const containerRect = document.getElementById('container').getBoundingClientRect();
      const left = rect.left - containerRect.left;
      const top = rect.top - containerRect.top;
      clipImgDiv.style.left = `${left}px`;
      clipImgDiv.style.top = `${top}px`;
      clipImgDiv.style.width = `${rect.width}px`;
      clipImgDiv.style.height = `${rect.height}px`;
    }

    // ========== 事件绑定 ==========
    function bindEvents() {
      // 上传触发
      uploadTrigger.addEventListener('click', triggerFileSelect);
      reuploadBtn.addEventListener('click', triggerFileSelect);

      // 应用裁剪
      applyCropBtn.addEventListener('click', applyCrop);

      // 关闭
      closeBtn.addEventListener('click', () => {
        window.location.reload(); // 简单重置
      });

      // 拖拽裁剪框
      document.getElementById('selectMoveDiv').addEventListener('mousedown', handleCropMouseDown);

      // 调整大小
      document.querySelectorAll('.avatar-cropper__option').forEach(handle => {
        handle.addEventListener('mousedown', (e) => {
          e.stopPropagation();
          const edge = e.target.dataset.edge;
          handleResizeMouseDown(e, edge);
        });
      });

      // 窗口变化
      window.addEventListener('resize', () => {
        setupCanvas();
        updateClipPosition();
        if (state.croppingImage) {
          drawImageToCanvas(state.croppingImage);
        }
      });
    }

    // 触发文件选择
    function triggerFileSelect() {
      fileInput.click();
    }

    fileInput.addEventListener('change', (e) => {
      const file = e.target.files[0];
      if (!file) return;
      if (!file.type.startsWith('image/')) {
        alert('请上传图片文件');
        return;
      }
      if (file.size > 2 * 1024 * 1024) {
        alert('图片大小不能超过2MB');
        return;
      }
      const reader = new FileReader();
      reader.onload = (e) => {
        state.croppingImage = e.target.result;
        drawImageToCanvas(state.croppingImage);
        clipImgDiv.style.display = 'block';
        actionButtons.style.display = 'flex';
        uploadTrigger.style.display = 'none';
        updateInitialCrop();
      };
      reader.readAsDataURL(file);
    });

    // 应用裁剪
    function applyCrop() {
      const link = document.createElement('a');
      resultCanvas.toBlob((blob) => {
        link.href = URL.createObjectURL(blob);
        link.download = 'cropped-avatar.jpg';
        link.click();
        URL.revokeObjectURL(link.href);
      }, 'image/jpeg', 0.9);
    }

    // ========== 拖拽与缩放逻辑 ==========
    function handleCropMouseDown(e) {
      state.isDragging = true;
      state.dragStart = { x: e.clientX - state.crop.x, y: e.clientY - state.crop.y };
      e.preventDefault();
    }

    function handleResizeMouseDown(e, edge) {
      state.isResizing = true;
      state.resizeStart = {
        x: e.clientX,
        y: e.clientY,
        cropX: state.crop.x,
        cropY: state.crop.y,
        cropWidth: state.crop.width,
        cropHeight: state.crop.height,
        edge: edge
      };
      e.preventDefault();
    }

    // 全局鼠标事件
    document.addEventListener('mousemove', (e) => {
      if (state.isDragging) {
        const newX = e.clientX - state.dragStart.x;
        const newY = e.clientY - state.dragStart.y;
        const maxX = state.canvasDimensions.width - state.crop.width;
        const maxY = state.canvasDimensions.height - state.crop.height;
        state.crop.x = Math.max(0, Math.min(newX, maxX));
        state.crop.y = Math.max(0, Math.min(newY, maxY));
        renderClip();
      }

      if (state.isResizing) {
        const deltaX = e.clientX - state.resizeStart.x;
        const deltaY = e.clientY - state.resizeStart.y;
        const aspectRatio = state.resizeStart.cropWidth / state.resizeStart.cropHeight;
        let newCrop = { ...state.crop };

        switch (state.resizeStart.edge) {
          case 'top-left':
            let newWidthTL = Math.max(100, state.resizeStart.cropWidth - deltaX);
            let newHeightTL = Math.max(100, state.resizeStart.cropHeight - deltaY);
            if (newWidthTL / aspectRatio <= newHeightTL) {
              newCrop.width = newWidthTL;
              newCrop.height = newWidthTL / aspectRatio;
            } else {
              newCrop.height = newHeightTL;
              newCrop.width = newHeightTL * aspectRatio;
            }
            newCrop.x = state.resizeStart.cropX + (state.resizeStart.cropWidth - newCrop.width);
            newCrop.y = state.resizeStart.cropY + (state.resizeStart.cropHeight - newCrop.height);
            break;
          case 'top-right':
            let newWidthTR = Math.max(100, state.resizeStart.cropWidth + deltaX);
            let newHeightTR = Math.max(100, state.resizeStart.cropHeight - deltaY);
            if (newWidthTR / aspectRatio <= newHeightTR) {
              newCrop.width = newWidthTR;
              newCrop.height = newWidthTR / aspectRatio;
            } else {
              newCrop.height = newHeightTR;
              newCrop.width = newHeightTR * aspectRatio;
            }
            newCrop.y = state.resizeStart.cropY + (state.resizeStart.cropHeight - newCrop.height);
            break;
          case 'bottom-left':
            let newWidthBL = Math.max(100, state.resizeStart.cropWidth - deltaX);
            let newHeightBL = Math.max(100, state.resizeStart.cropHeight + deltaY);
            if (newWidthBL / aspectRatio <= newHeightBL) {
              newCrop.width = newWidthBL;
              newCrop.height = newWidthBL / aspectRatio;
            } else {
              newCrop.height = newHeightBL;
              newCrop.width = newHeightBL * aspectRatio;
            }
            newCrop.x = state.resizeStart.cropX + (state.resizeStart.cropWidth - newCrop.width);
            break;
          case 'bottom-right':
            let newWidthBR = Math.max(100, state.resizeStart.cropWidth + deltaX);
            let newHeightBR = Math.max(100, state.resizeStart.cropHeight + deltaY);
            if (newWidthBR / aspectRatio <= newHeightBR) {
              newCrop.width = newWidthBR;
              newCrop.height = newWidthBR / aspectRatio;
            } else {
              newCrop.height = newHeightBR;
              newCrop.width = newHeightBR * aspectRatio;
            }
            break;
        }

        // 限制在画布内
        newCrop.x = Math.max(0, newCrop.x);
        newCrop.y = Math.max(0, newCrop.y);
        newCrop.width = Math.min(state.canvasDimensions.width - newCrop.x, newCrop.width);
        newCrop.height = Math.min(state.canvasDimensions.height - newCrop.y, newCrop.height);
        state.crop = newCrop;
        renderClip();
      }
    });

    document.addEventListener('mouseup', () => {
      state.isDragging = false;
      state.isResizing = false;
    });

    // 启动
    window.addEventListener('DOMContentLoaded', init);
  </script>
</body>

</html>

## 2. 整体架构：模态框 + 双画布设计

该工具采用经典的 **“上传 → 裁剪 → 预览 → 下载”** 流程，结构清晰：

```
[ 模态框 Overlay ]
  └── [ 左侧：主画布 + 裁剪框 ]
  └── [ 右侧：200×200 预览 + 操作按钮 ]
```

- **主画布（`#canvas`）**：显示用户上传的原始图片（居中缩放）
- **裁剪框（`.avatar-cropper__move-clip-div`）**：覆盖在主画布上的可拖拽/缩放区域
- **预览画布（`#resultCanvas`）**：实时显示裁剪结果（固定 200×200）


## 3. 核心组件解析

1. 🖼️ 图片上传与绘制

    - **触发方式**：点击 `.avatar-cropper__upload-trigger` → 触发隐藏 `<input type="file">`
    - **格式限制**：`accept="image/*"` + JS 校验 `file.type.startsWith('image/')`
    - **大小限制**：≤ 2MB（防止大图卡顿）
    - **绘制逻辑**：
        ```js
        // 将图片等比缩放并居中绘制到 canvas
        const scale = Math.min(maxWidth / img.width, maxHeight / img.height);
        ctx.drawImage(img, offsetX, offsetY, displayWidth, displayHeight);
        ```

2. 裁剪框（Crop Box）

    - **结构**：
        - 外层遮罩（半透明黑）：`.avatar-cropper__clip-img-div`
        - 裁剪区域（白色半透明）：`.avatar-cropper__move-clip-div`
        - 拖拽区（80% 中心）：`.avatar-cropper__select-move-div`
        - 四角缩放手柄：带 `data-edge` 属性（如 `top-left`）

    - **交互**：
        | 操作 | 实现 |
        |------|------|
        | 拖动移动 | 监听 `mousedown` → `mousemove` 更新 `crop.x/y` |
        | 拖动缩放 | 根据 `data-edge` 计算新尺寸，**保持宽高比** |
        | 边界限制 | 确保裁剪框不超出画布 |

3. 实时预览
    - **当前逻辑（有缺陷）**：

      ```js
      ctx.drawImage(
        canvas,               // ← 错误！应使用原始图片
        state.crop.x,
        state.crop.y,
        state.crop.width,
        state.crop.height,
        0, 0, 200, 200
      );
      ```

      - **问题**：直接从 **已缩放的 canvas** 裁剪，导致：
        - 分辨率损失（二次采样）
        - 坐标错位（未映射回原始图像）

4. 应用裁剪
    - **导出方式**：`resultCanvas.toBlob()` → 生成 JPEG 下载链接
    - **文件名**：固定为 `cropped-avatar.jpg`
