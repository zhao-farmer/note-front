# 一、Vant

Vant 是由 有赞团队 开发的轻量、高性能移动端组件库，支持 Vue 3 / Vue 2 / 微信小程序 / Taro / UniApp。本文以 Vue 3 + Vite 项目为例，展示如何快速集成并使用 Vant。

- 官网文档：[https://vant-ui.github.io/vant](https://vant-ui.github.io/vant)
- GitHub：[https://github.com/youzan/vant](https://github.com/youzan/vant)
- 在线示例：[https://vant-ui.github.io/vant/mobile.html](https://vant-ui.github.io/vant/mobile.html)

## 1.1 快速开始

1. 安装插件

    ```sh
    npm install vant@^4
    ```

2. 示例：按钮 + 表格 + 弹窗 + 下拉刷新

    ```vue
    <template>
      <div class="app">
        <!-- 顶部标题 -->
        <van-nav-bar title="Vant 示例" />
    
        <!-- 主要内容区 -->
        <div class="content">
          <!-- 按钮组 -->
          <van-button type="primary" @click="showDialog = true">打开弹窗</van-button>
          <van-button type="success" class="ml-2" @click="showToast">显示提示</van-button>
    
          <!-- 列表（带下拉刷新） -->
          <van-pull-refresh v-model="refreshing" @refresh="onRefresh">
            <van-list
              v-model:loading="loading"
              :finished="finished"
              finished-text="没有更多了"
              @load="onLoad"
            >
              <van-cell
                v-for="item in list"
                :key="item.id"
                :title="item.name"
                :label="item.role"
                is-link
                @click="viewDetail(item)"
              />
            </van-list>
          </van-pull-refresh>
        </div>
    
        <!-- 弹窗 -->
        <van-dialog
          v-model:show="showDialog"
          title="员工详情"
          show-cancel-button
          @confirm="handleConfirm"
        >
          <div v-if="selectedEmployee">
            <p>ID: {{ selectedEmployee.id }}</p>
            <p>姓名: {{ selectedEmployee.name }}</p>
            <p>职位: {{ selectedEmployee.role }}</p>
          </div>
        </van-dialog>
      </div>
    </template>
    
    <script setup lang="ts">
    import { ref } from 'vue'
    
    // 模拟数据
    const employees = [
      { id: 1, name: '张三', role: '前端工程师' },
      { id: 2, name: '李四', role: '后端工程师' },
      { id: 3, name: '王五', role: '产品经理' },
    ]
    
    // 状态管理
    const showDialog = ref(false)
    const selectedEmployee = ref<any>(null)
    const list = ref<any[]>([])
    const loading = ref(false)
    const finished = ref(false)
    const refreshing = ref(false)
    
    // 初始化数据
    const loadMore = () => {
      const start = list.value.length
      const end = start + 5
      list.value.push(...employees.slice(start % employees.length, end % employees.length))
      loading.value = false
      if (list.value.length >= 15) finished.value = true
    }
    
    // 加载更多
    const onLoad = () => {
      if (finished.value) return
      loading.value = true
      setTimeout(loadMore, 500)
    }
    
    // 下拉刷新
    const onRefresh = () => {
      setTimeout(() => {
        list.value = []
        finished.value = false
        loadMore()
        refreshing.value = false
      }, 1000)
    }
    
    // 显示 Toast
    const showToast = () => {
      // Vant 的 Toast 会自动按需引入
      Toast('操作成功！')
    }
    
    // 查看详情
    const viewDetail = (employee: any) => {
      selectedEmployee.value = employee
      showDialog.value = true
    }
    
    // 弹窗确认
    const handleConfirm = () => {
      console.log('确认操作')
    }
    </script>
    
    <style scoped>
    .app {
      min-height: 100vh;
      background-color: #f5f5f5;
    }
    .content {
      padding: 16px;
    }
    .ml-2 {
      margin-left: 8px;
    }
    </style>
    ```
3. 移动端适配

    1. 使用 postcss-px-to-viewport

        ```bash
        npm install -D postcss-px-to-viewport
        ```
    2. 配置 postcss.config.js：

        ```js
        module.exports = {
          plugins: {
            'postcss-px-to-viewport': {
              viewportWidth: 375, // 设计稿宽度
              unitPrecision: 5,
              viewportUnit: 'vw',
              selectorBlackList: [],
              minPixelValue: 1,
              mediaQuery: false,
            }
          }
        }
        ```
