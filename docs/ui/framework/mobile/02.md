# 二、Ant Design Mobile


Ant Design Mobile（简称 Antd Mobile）是由蚂蚁金服推出的 企业级移动端 React UI 组件库，适用于 H5、混合 App、小程序等场景。

- 官网文档：[https://mobile.ant.design](https://mobile.ant.design)
- GitHub：[https://github.com/ant-design/ant-design-mobile](https://github.com/ant-design/ant-design-mobile)
- 在线示例：[https://mobile.ant.design/zh/components/button](https://mobile.ant.design/zh/components/button)

## 2.1 快速使用

1. 安装插件

    ```sh
    npm install --save antd-mobile
    ```
2. 示例：按钮 + 列表 + 弹窗 + 下拉刷新

    ```jsx
    import React, { useState, useEffect } from 'react';
    import {
      Button,
      Dialog,
      Toast,
      PullToRefresh,
      List,
      Space,
      NavBar,
    } from 'antd-mobile';
    
    // 模拟员工数据
    const mockEmployees = [
      { id: 1, name: '张三', role: '前端工程师' },
      { id: 2, name: '李四', role: '后端工程师' },
      { id: 3, name: '王五', role: '产品经理' },
    ];
    
    export default function App() {
      // 状态管理
      const [showDialog, setShowDialog] = useState(false);
      const [selectedEmployee, setSelectedEmployee] = useState<any>(null);
      const [list, setList] = useState<any[]>([]);
      const [loading, setLoading] = useState(false);
      const [refreshing, setRefreshing] = useState(false);
      const [hasMore, setHasMore] = useState(true);
    
      // 初始化数据
      useEffect(() => {
        loadMore();
      }, []);
    
      // 加载更多
      const loadMore = () => {
        if (!hasMore) return;
        setLoading(true);
        setTimeout(() => {
          const start = list.length;
          const newData = mockEmployees.slice(start % mockEmployees.length, start %     mockEmployees.length + 5);
          setList([...list, ...newData]);
          setLoading(false);
          if (list.length + newData.length >= 15) {
            setHasMore(false);
          }
        }, 500);
      };
    
      // 下拉刷新
      const onRefresh = async () => {
        setRefreshing(true);
        await new Promise(resolve => setTimeout(resolve, 1000));
        setList([]);
        setHasMore(true);
        loadMore();
        setRefreshing(false);
      };
    
      // 显示 Toast
      const showToast = () => {
        Toast.show('操作成功！');
      };
    
      // 查看详情
      const viewDetail = (employee: any) => {
        setSelectedEmployee(employee);
        setShowDialog(true);
      };
    
      return (
        <div style={{ minHeight: '100vh', backgroundColor: '#f5f5f5' }}>
          {/* 顶部导航栏 */}
          <NavBar onBack={() => window.history.back()}>Antd Mobile 示例</NavBar>
    
          {/* 主要内容区 */}
          <div style={{ padding: '16px' }}>
            {/* 按钮组 */}
            <Space direction="horizontal" wrap>
              <Button color="primary" onClick={() => setShowDialog(true)}>
                打开弹窗
              </Button>
              <Button color="success" onClick={showToast}>
                显示提示
              </Button>
            </Space>
    
            {/* 列表（带下拉刷新） */}
            <PullToRefresh onRefresh={onRefresh} refreshing={refreshing}>
              <List>
                {list.map((item) => (
                  <List.Item
                    key={item.id}
                    clickable
                    onClick={() => viewDetail(item)}
                    extra={<span>查看详情</span>}
                  >
                    <List.Item.Meta
                      title={item.name}
                      description={item.role}
                    />
                  </List.Item>
                ))}
              </List>
              {loading && <div style={{ textAlign: 'center', padding: '16px' }}>加载中...</ div>}
              {!hasMore && <div style={{ textAlign: 'center', padding: '16px', color:   '#999' }}>没有更多了</div>}
            </PullToRefresh>
          </div>
    
          {/* 弹窗 */}
          <Dialog
            visible={showDialog}
            onClose={() => setShowDialog(false)}
            title="员工详情"
            content={
              selectedEmployee ? (
                <div>
                  <p>ID: {selectedEmployee.id}</p>
                  <p>姓名: {selectedEmployee.name}</p>
                  <p>职位: {selectedEmployee.role}</p>
                </div>
              ) : null
            }
            actions={[
              {
                key: 'cancel',
                text: '取消',
                onClick: () => setShowDialog(false),
              },
              {
                key: 'confirm',
                text: '确认',
                primary: true,
                onClick: () => {
                  console.log('确认操作');
                  setShowDialog(false);
                },
              },
            ]}
          />
        </div>
      );
    }
    ```
