# 一、Angular Material

Angular Material 是由 Angular 官方团队维护的一套 UI 组件库，遵循 **Google 的 Material Design 规范**，提供高质量、无障碍（a11y）、响应式的组件，非常适合构建现代化的 Angular 应用。

- 官网文档：[https://material.angular.io](https://material.angular.io)
- GitHub：[https://github.com/angular/components](https://github.com/angular/components)
- 组件列表：[https://material.angular.io/components/categories](https://material.angular.io/components/categories)


## 1.1 快速开始

1. 安装插件

    ```bash
    ng add @angular/material
    ```

    > 这会自动：
    > - 安装 `@angular/material`、`@angular/cdk`、`@angular/animations`
    > - 提示你选择 **主题**（如 Indigo/Pink）
    > - 询问是否启用 **全局 CSS 重置**（推荐启用）
    > - 自动导入 `BrowserAnimationsModule`

2. 示例：按钮 + 表格 + 弹窗


    - 步骤 1：创建弹窗组件

        ```bash
        ng generate component employee-detail-dialog
        ```

        **`employee-detail-dialog.component.ts`**
        ```ts
        import { Component, Inject } from '@angular/core';
        import { MAT_DIALOG_DATA, MatDialogRef } from '@angular/material/dialog';

        export interface Employee {
          id: number;
          name: string;
          role: string;
        }

        @Component({
          selector: 'app-employee-detail-dialog',
          template: `
            <h2 mat-dialog-title>员工详情</h2>
            <mat-dialog-content>
              <p><strong>ID:</strong> {{ data.id }}</p>
              <p><strong>姓名:</strong> {{ data.name }}</p>
              <p><strong>职位:</strong> {{ data.role }}</p>
            </mat-dialog-content>
            <mat-dialog-actions align="end">
              <button mat-button mat-dialog-close>关闭</button>
            </mat-dialog-actions>
          `,
        })
        export class EmployeeDetailDialogComponent {
          constructor(
            public dialogRef: MatDialogRef<EmployeeDetailDialogComponent>,
            @Inject(MAT_DIALOG_DATA) public data: Employee
          ) {}
        }
        ```

    - 步骤 2：在主模块中导入所需 Material 模块

        **`app.module.ts`**

        ```ts
        import { NgModule } from '@angular/core';
        import { BrowserModule } from '@angular/platform-browser';
        import { BrowserAnimationsModule } from '@angular/platform-browser/animations';

        // Material 模块
        import { MatButtonModule } from '@angular/material/button';
        import { MatTableModule } from '@angular/material/table';
        import { MatDialogModule } from '@angular/material/dialog';
        import { MatCardModule } from '@angular/material/card';

        import { AppComponent } from './app.component';
        import { EmployeeDetailDialogComponent } from './employee-detail-dialog/        employee-detail-dialog.component';

        @NgModule({
          declarations: [
            AppComponent,
            EmployeeDetailDialogComponent
          ],
          imports: [
            BrowserModule,
            BrowserAnimationsModule,
            MatButtonModule,
            MatTableModule,
            MatDialogModule,
            MatCardModule,
          ],
          providers: [],
          bootstrap: [AppComponent]
        })
        export class AppModule { }
        ```


    - 步骤 3：主组件实现（`app.component.ts`）

        **`app.component.ts`**

        ```ts
        import { Component } from '@angular/core';
        import { MatDialog } from '@angular/material/dialog';
        import { Employee, EmployeeDetailDialogComponent } from './ employee-detail-dialog/employee-detail-dialog.component';

        @Component({
          selector: 'app-root',
          template: `
            <div style="padding: 20px; max-width: 800px; margin: 0 auto;">
              <button mat-raised-button color="primary" (click)="onAdd()">
                新增员工
              </button>

              <table mat-table [dataSource]="employees" style="width: 100%; margin-top:     20px;"  class="mat-elevation-z8">
                <!-- 姓名列 -->
                <ng-container matColumnDef="name">
                  <th mat-header-cell *matHeaderCellDef>姓名</th>
                  <td mat-cell *matCellDef="let emp">{{ emp.name }}</td>
                </ng-container>

                <!-- 职位列 -->
                <ng-container matColumnDef="role">
                  <th mat-header-cell *matHeaderCellDef>职位</th>
                  <td mat-cell *matCellDef="let emp">{{ emp.role }}</td>
                </ng-container>

                <!-- 操作列 -->
                <ng-container matColumnDef="action">
                  <th mat-header-cell *matHeaderCellDef>操作</th>
                  <td mat-cell *matCellDef="let emp">
                    <button mat-button color="primary" (click)="viewDetail(emp)">
                      查看详情
                    </button>
                  </td>
                </ng-container>

                <tr mat-header-row *matHeaderRowDef="displayedColumns"></tr>
                <tr mat-row *matRowDef="let row; columns: displayedColumns;"></tr>
              </table>
            </div>
          `,
          styles: [`
            table { width: 100%; }
            th, td { padding: 12px 8px; }
          `]
        })
        export class AppComponent {
          displayedColumns = ['name', 'role', 'action'];

          employees: Employee[] = [
            { id: 1, name: '张三', role: '前端工程师' },
            { id: 2, name: '李四', role: '后端工程师' },
            { id: 3, name: '王五', role: '产品经理' }
          ];

          constructor(public dialog: MatDialog) {}

          viewDetail(employee: Employee) {
            this.dialog.open(EmployeeDetailDialogComponent, {
              width: '400px',
              data: employee
            });
          }

          onAdd() {
            alert('新增功能待实现');
          }
        }
        ```
