# 三、Naive UI

Naive UI 是一个基于 Vue 3 的现代化、高性能、TypeScript 编写的 UI 组件库，由 [TuSimple](https://github.com/TuSimple) 团队开发。它具有以下特点：


- 官网文档：[https://www.naiveui.com](https://www.naiveui.com)
- GitHub：[https://github.com/tusen-ai/naive-ui](https://github.com/tusen-ai/naive-ui)
- 组件列表：[https://www.naiveui.com/zh-CN/os-theme/components](https://www.naiveui.com/zh-CN/os-theme/components)

## 3.1 快速使用

1. 安装插件

    ```sh
    npm i -D naive-ui
    ```
2. 全局引入 main.ts

    ```js
    import { createApp } from 'vue'
    import App from './App.vue'

    // 引入 Naive UI 全局组件和样式
    import naive from 'naive-ui'

    const app = createApp(App)
    app.use(naive)
    app.mount('#app')
    ```
3. 示例 按钮+表格+弹窗

    ```vue
    <template>
      <n-space vertical>
        <!-- 按钮：打开弹窗 -->
        <n-button type="primary" @click="showModal = true">
          添加新用户
        </n-button>
    
        <!-- 表格 -->
        <n-data-table
          :columns="columns"
          :data="tableData"
          :pagination="pagination"
        />
    
        <!-- 弹窗 -->
        <n-modal v-model:show="showModal" preset="dialog" title="添加用户"    @positive-click="handleAddUser">
          <n-form :model="formValue" :rules="rules" ref="formRef">
            <n-form-item label="姓名" path="name">
              <n-input v-model:value="formValue.name" placeholder="请输入姓名" />
            </n-form-item>
            <n-form-item label="邮箱" path="email">
              <n-input v-model:value="formValue.email" placeholder="请输入邮箱" />
            </n-form-item>
          </n-form>
        </n-modal>
      </n-space>
    </template>
    
    <script setup>
    import { ref, reactive } from 'vue'
    import { useMessage } from 'naive-ui'
    
    const message = useMessage()
    
    // 表格数据
    const tableData = ref([
      { id: 1, name: '张三', email: 'zhangsan@example.com' },
      { id: 2, name: '李四', email: 'lisi@example.com' }
    ])
    
    // 表格列配置
    const columns = [
      { title: 'ID', key: 'id' },
      { title: '姓名', key: 'name' },
      { title: '邮箱', key: 'email' }
    ]
    
    // 分页配置（可选）
    const pagination = {
      pageSize: 10
    }
    
    // 弹窗控制
    const showModal = ref(false)
    
    // 表单数据
    const formValue = reactive({
      name: '',
      email: ''
    })
    
    // 表单规则
    const rules = {
      name: { required: true, message: '请输入姓名', trigger: 'blur' },
      email: [
        { required: true, message: '请输入邮箱', trigger: 'blur' },
        { type: 'email', message: '邮箱格式不正确', trigger: 'blur' }
      ]
    }
    
    const formRef = ref(null)
    
    // 提交表单
    const handleAddUser = (e) => {
      e.preventDefault()
      formRef.value?.validate((errors) => {
        if (!errors) {
          // 添加新用户
          const newId = Math.max(...tableData.value.map(item => item.id), 0) + 1
          tableData.value.push({
            id: newId,
            name: formValue.name,
            email: formValue.email
          })
          message.success('用户添加成功！')
          showModal.value = false
          // 重置表单
          formValue.name = ''
          formValue.email = ''
        } else {
          message.error('请检查输入内容')
        }
      })
    }
    </script>
    ```
