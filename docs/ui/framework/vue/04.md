# 四、ant-design-vue

ant-design-vue 是对 Ant Design 的 Vue 实现，开发和服务于企业级后台产品。

官方文档：[https://www.antdv.com/](持续更新，支持中英文）)

github地址：[https://github.com/vueComponent/ant-design-vue](https://github.com/vueComponent/ant-design-vue)

## 4.1 快速开始

1. 安装插件

    ```sh
    npm install ant-design-vue
    ```
2. 全局注册 main.ts

    ```ts
    import { createApp } from 'vue';
    import Antd from 'ant-design-vue';
    import App from './App';
    import 'ant-design-vue/dist/reset.css';

    const app = createApp(App);

    app.use(Antd).mount('#app');
    ```

3. 示例代码

    ```vue
    <template>
      <div style="padding: 20px">
        <!-- 按钮：打开弹窗 -->
        <a-button type="primary" @click="showModal = true" style="margin-bottom: 16px">
          添加新用户
        </a-button>
    
        <!-- 表格 -->
        <a-table
          :columns="columns"
          :data-source="tableData"
          :row-key="record => record.id"
          :pagination="false"
        />
    
        <!-- 弹窗 -->
        <a-modal
          v-model:open="showModal"
          title="添加用户"
          @ok="handleOk"
          @cancel="handleCancel"
          :confirm-loading="confirmLoading"
        >
          <a-form
            ref="formRef"
            :model="formState"
            :rules="rules"
            layout="vertical"
          >
            <a-form-item label="姓名" name="name">
              <a-input v-model:value="formState.name" placeholder="请输入姓名" />
            </a-form-item>
            <a-form-item label="邮箱" name="email">
              <a-input v-model:value="formState.email" placeholder="请输入邮箱" />
            </a-form-item>
          </a-form>
        </a-modal>
      </div>
    </template>
    
    <script setup>
    import { ref, reactive } from 'vue'
    
    // 表格数据
    const tableData = ref([
      { id: 1, name: '张三', email: 'zhangsan@example.com' },
      { id: 2, name: '李四', email: 'lisi@example.com' }
    ])
    
    // 表格列定义
    const columns = [
      { title: 'ID', dataIndex: 'id', key: 'id' },
      { title: '姓名', dataIndex: 'name', key: 'name' },
      { title: '邮箱', dataIndex: 'email', key: 'email' }
    ]
    
    // 弹窗控制
    const showModal = ref(false)
    const confirmLoading = ref(false)
    
    // 表单状态
    const formState = reactive({
      name: '',
      email: ''
    })
    
    // 表单规则
    const rules = {
      name: [
        { required: true, message: '请输入姓名', trigger: 'blur' }
      ],
      email: [
        { required: true, message: '请输入邮箱', trigger: 'blur' },
        { type: 'email', message: '邮箱格式不正确', trigger: 'blur' }
      ]
    }
    
    const formRef = ref()
    
    // 点击确定
    const handleOk = async () => {
      try {
        const values = await formRef.value.validateFields()
        confirmLoading.value = true
    
        // 模拟异步提交
        setTimeout(() => {
          const newId = Math.max(...tableData.value.map(item => item.id), 0) + 1
          tableData.value.push({
            id: newId,
            name: values.name,
            email: values.email
          })
          confirmLoading.value = false
          showModal.value = false
    
          // 重置表单
          formRef.value.resetFields()
        }, 800)
      } catch (error) {
        console.log('Validation failed:', error)
      }
    }
    
    // 点击取消
    const handleCancel = () => {
      showModal.value = false
      formRef.value?.resetFields()
    }
    </script>
    ```
