5.1 基础 Hooks (React 16.8+)

5.1.1 useState

作用: 在函数组件中添加状态管理
使用方式: const [state, setState] = useState(initialState)
参数作用:

  • initialState: 状态的初始值，可以是值或函数（惰性初始化）
import { useState } from 'react';

function Counter() {
  // 基本用法
  const [count, setCount] = useState(0);
  
  // 函数式初始值（性能优化）
  const [todos, setTodos] = useState(() => {
    const savedTodos = localStorage.getItem('todos');
    return savedTodos ? JSON.parse(savedTodos) : [];
  });
  
  // 对象状态更新
  const [user, setUser] = useState({ name: 'John', age: 25 });

  const increment = () => {
    setCount(prevCount => prevCount + 1); // 函数式更新
  };

  const updateName = (name) => {
    setUser(prevUser => ({ ...prevUser, name })); // 合并对象
  };

  return (
    <div>
      <p>Count: {count}</p>
      <button onClick={increment}>+1</button>
      <input
        value={user.name}
        onChange={(e) => updateName(e.target.value)}
      />
    </div>
  );
}

5.1.2 useEffect

作用: 处理副作用（数据获取、订阅、DOM操作）
使用方式: useEffect(callback, dependencies?)
参数作用:

  • callback: 包含副作用逻辑的函数，可返回清理函数

  • dependencies: 依赖数组，控制何时重新执行

import { useState, useEffect } from 'react';

function UserProfile({ userId }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  // 1. 无依赖数组 - 每次渲染都执行
  useEffect(() => {
    document.title = 'My App';
  });

  // 2. 空依赖数组 - 仅在挂载时执行
  useEffect(() => {
    console.log('Component mounted');
    
    return () => {
      console.log('Component unmounted'); // 清理函数
    };
  }, []);

  // 3. 有依赖数组 - 依赖变化时执行
  useEffect(() => {
    if (!userId) return;

    const fetchUser = async () => {
      setLoading(true);
      try {
        const response = await fetch(`/api/users/${userId}`);
        const userData = await response.json();
        setUser(userData);
      } catch (error) {
        console.error('Failed to fetch user:', error);
      } finally {
        setLoading(false);
      }
    };
    
    fetchUser();
  }, [userId]); // userId 变化时重新执行

  // 4. 订阅示例
  useEffect(() => {
    const timer = setInterval(() => {
      console.log('Timer tick');
    }, 1000);

    return () => clearInterval(timer); // 清理函数
  }, []);

  if (loading) return <div>Loading...</div>;
  return <div>Hello, {user?.name}</div>;
}

使用方式: const value = useContext(MyContext)
参数作用:

import { createContext, useContext, useState } from 'react';

// 创建 Context
const ThemeContext = createContext();
const UserContext = createContext();

function App() {
  const [theme, setTheme] = useState('light');

        <Main />

}

5.1.4 useReducer

作用: 复杂状态逻辑管理，类似于 Redux
使用方式: const [state, dispatch] = useReducer(reducer, initialState)
参数作用:

  • reducer: 状态更新函数 (state, action) => newState

  • initialState: 初始状态

  • init: 可选，惰性初始化函数

import { useReducer } from 'react';

// Reducer 函数
function todoReducer(state, action) {
  switch (action.type) {
    case 'ADD_TODO':
      return {
        ...state,
        todos: [...state.todos, {
          id: Date.now(),
          text: action.text,
          completed: false
        }]
      };
    case 'TOGGLE_TODO':
      return {
        ...state,
        todos: state.todos.map(todo =>
          todo.id === action.id
            ? { ...todo, completed: !todo.completed }
            : todo
        )
      };
    case 'DELETE_TODO':
      return {
        ...state,
        todos: state.todos.filter(todo => todo.id !== action.id)
      };
    case 'SET_FILTER':
      return {
        ...state,
        filter: action.filter
      };
    default:
      return state;
  }
}

const initialState = {
  todos: [],
  filter: 'all'
};

function TodoApp() {
  const [state, dispatch] = useReducer(todoReducer, initialState);

  const addTodo = (text) => {
    dispatch({ type: 'ADD_TODO', text });
  };

  const toggleTodo = (id) => {
    dispatch({ type: 'TOGGLE_TODO', id });
  };

  const deleteTodo = (id) => {
    dispatch({ type: 'DELETE_TODO', id });
  };

  const setFilter = (filter) => {
    dispatch({ type: 'SET_FILTER', filter });
  };

  const filteredTodos = state.todos.filter(todo => {
    if (state.filter === 'completed') return todo.completed;
    if (state.filter === 'active') return !todo.completed;
    return true;
  });

  return (
    <div>
      <h1>Todo List ({state.todos.length})</h1>
      
      <input
        placeholder="Add todo..."
        onKeyPress={(e) => {
          if (e.key === 'Enter' && e.target.value.trim()) {
            addTodo(e.target.value.trim());
            e.target.value = '';
          }
        }}
      />
      
      <div>
        <button onClick={() => setFilter('all')}>All</button>
        <button onClick={() => setFilter('active')}>Active</button>
        <button onClick={() => setFilter('completed')}>Completed</button>
      </div>
      
      <ul>
        {filteredTodos.map(todo => (
          <li key={todo.id} style={{
            textDecoration: todo.completed ? 'line-through' : 'none'
          }}>
            <span onClick={() => toggleTodo(todo.id)}>{todo.text}</span>
            <button onClick={() => deleteTodo(todo.id)}>Delete</button>
          </li>
        ))}
      </ul>
    </div>
  );
}

5.2 性能优化 Hooks

5.2.1 useMemo

作用: 缓存计算结果，避免不必要的重复计算
使用方式: const memoizedValue = useMemo(() => computeExpensiveValue(a, b), [a, b])
参数作用:

  • create: 计算函数，返回要缓存的值

  • dependencies: 依赖数组，变化时重新计算

import { useState, useMemo } from 'react';

function ExpensiveCalculation({ numbers }) {
  // 缓存昂贵的计算结果
  const sum = useMemo(() => {
    console.log('Calculating sum...');
    return numbers.reduce((acc, num) => acc + num, 0);
  }, [numbers]); // 只有当 numbers 变化时重新计算

  const [count, setCount] = useState(0);

  return (
    <div>
      <p>Numbers: {numbers.join(', ')}</p>
      <p>Sum: {sum}</p>
      <p>Count: {count}</p>
      <button onClick={() => setCount(count + 1)}>
        Re-render (won't recalculate sum)
      </button>
    </div>
  );
}

// 使用组件
function App() {
  const [numbers, setNumbers] = useState([1, 2, 3, 4, 5]);
  
  return (
    <div>
      <ExpensiveCalculation numbers={numbers} />
      <button onClick={() => setNumbers([...numbers, numbers.length + 1])}>
        Add Number (will recalculate sum)
      </button>
    </div>
  );
}

5.2.2 useCallback

作用: 缓存函数，避免不必要的函数重建
使用方式: const memoizedCallback = useCallback(() => { doSomething(a, b); }, [a, b]);
参数作用:

  • callback: 要缓存的函数

  • dependencies: 依赖数组，变化时重新创建函数

import { useState, useCallback, memo } from 'react';

const ChildButton = memo(({ onClick, children }) => {
  console.log('ChildButton rendered:', children);
  return <button onClick={onClick}>{children}</button>;
});

function ParentComponent() {
  const [count, setCount] = useState(0);
  const [text, setText] = useState('');

  // 没有 useCallback - 每次渲染都会创建新函数
  // const handleIncrement = () => setCount(count + 1);

  // 使用 useCallback - 依赖变化时才重新创建
  const handleIncrement = useCallback(() => {
    setCount(prevCount => prevCount + 1);
  }, []); // 空依赖，函数只创建一次

  const handleReset = useCallback(() => {
    setCount(0);
  }, []);

  // 依赖 text 的函数
  const handleAlert = useCallback(() => {
    alert(`Current text: ${text}`);
  }, [text]); // text 变化时重新创建

  return (
    <div>
      <p>Count: {count}</p>
      <input
        value={text}
        onChange={(e) => setText(e.target.value)}
        placeholder="Type something..."
      />
      
      {/* 这些按钮不会因为父组件重渲染而重新渲染 */}
      <ChildButton onClick={handleIncrement}>Increment</ChildButton>
      <ChildButton onClick={handleReset}>Reset</ChildButton>
      <ChildButton onClick={handleAlert}>Show Text</ChildButton>
    </div>
  );
}

5.3 Ref & DOM Hooks

5.3.1 useRef

作用: 创建可变引用，访问 DOM 元素或存储任意可变值
使用方式: const ref = useRef(initialValue)
参数作用:

  • initialValue: ref 对象的初始值
import { useRef, useState, useEffect } from 'react';

function RefExamples() {
  // 1. 访问 DOM 元素
  const inputRef = useRef(null);
  const buttonRef = useRef(null);
  
  // 2. 存储可变值（不会触发重渲染）
  const renderCount = useRef(0);
  const previousValue = useRef('');

  const [value, setValue] = useState('');

  useEffect(() => {
    renderCount.current += 1;
    previousValue.current = value;
  });

  const focusInput = () => {
    inputRef.current.focus();
    inputRef.current.select();
  };

  const changeButtonColor = () => {
    buttonRef.current.style.backgroundColor = 
      buttonRef.current.style.backgroundColor === 'red' ? 'blue' : 'red';
  };

  return (
    <div>
      <h2>useRef Examples</h2>
      
      <div>
        <input
          ref={inputRef}
          value={value}
          onChange={(e) => setValue(e.target.value)}
          placeholder="Type something..."
        />
        <button onClick={focusInput}>Focus Input</button>
      </div>
      
      <div>
        <button ref={buttonRef} onClick={changeButtonColor}>
          Change My Color
        </button>
      </div>
      
      <div>
        <p>Current Value: {value}</p>
        <p>Previous Value: {previousValue.current}</p>
        <p>Component rendered {renderCount.current} times</p>
      </div>

      {/* 3. 实现上一个值跟踪 */}
      <PreviousValueTracker currentValue={value} />
    </div>
  );
}

// 自定义 Hook：跟踪上一个值
function usePrevious(value) {
  const ref = useRef();
  useEffect(() => {
    ref.current = value;
  });
  return ref.current;
}

function PreviousValueTracker({ currentValue }) {
  const previousValue = usePrevious(currentValue);
  
  return (
    <div>
      <h3>Previous Value Tracker</h3>
      <p>Current: {currentValue}</p>
      <p>Previous: {previousValue}</p>
    </div>
  );
}

5.3.2 useImperativeHandle

作用: 自定义暴露给父组件的 ref 实例
使用方式: useImperativeHandle(ref, createHandle, [deps])
参数作用:

  • ref: 从 forwardRef 接收的 ref

  • createHandle: 返回要暴露的句柄的函数

  • deps: 依赖数组

import { forwardRef, useImperativeHandle, useRef } from 'react';

const FancyInput = forwardRef((props, ref) => {
  const inputRef = useRef();
  const buttonRef = useRef();

  // 自定义暴露给父组件的方法
  useImperativeHandle(ref, () => ({
    focus: () => {
      inputRef.current.focus();
    },
    shake: () => {
      inputRef.current.style.transform = 'translateX(10px)';
      setTimeout(() => {
        inputRef.current.style.transform = 'translateX(0)';
      }, 300);
    },
    getValue: () => inputRef.current.value,
    setValue: (value) => {
      inputRef.current.value = value;
    },
    clickButton: () => {
      buttonRef.current.click();
    }
  }));

  return (
    <div>
      <input
        ref={inputRef}
        placeholder="Fancy input..."
        style={{ transition: 'transform 0.3s' }}
      />
      <button ref={buttonRef} onClick={() => alert('Button clicked!')}>
        Hidden Button
      </button>
    </div>
  );
});

function ParentComponent() {
  const fancyInputRef = useRef();

  const handleFocus = () => {
    fancyInputRef.current.focus();
  };

  const handleShake = () => {
    fancyInputRef.current.shake();
  };

  const handleGetValue = () => {
    alert(`Input value: ${fancyInputRef.current.getValue()}`);
  };

  const handleSetValue = () => {
    fancyInputRef.current.setValue('Hello World!');
  };

  const handleClickButton = () => {
    fancyInputRef.current.clickButton();
  };

  return (
    <div>
      <FancyInput ref={fancyInputRef} />
      <div style={{ marginTop: '1rem' }}>
        <button onClick={handleFocus}>Focus Input</button>
        <button onClick={handleShake}>Shake Input</button>
        <button onClick={handleGetValue}>Get Value</button>
        <button onClick={handleSetValue}>Set Value</button>
        <button onClick={handleClickButton}>Click Hidden Button</button>
      </div>
    </div>
  );
}

5.4 React 18+ Concurrent Hooks

5.4.1 useTransition

作用: 将状态更新标记为非紧急，提高用户体验
使用方式: const [isPending, startTransition] = useTransition()
参数作用: 无参数

import { useState, useTransition, memo } from 'react';

const SlowList = memo(({ query }) => {
  const items = [];
  for (let i = 0; i < 500; i++) {
    items.push(<ListItem key={i} query={query} index={i} />);
  }
  
  return <ul>{items}</ul>;
});

const ListItem = ({ query, index }) => {
  const startTime = performance.now();
  while (performance.now() - startTime < 1) {
    // 模拟计算密集型操作
  }
  
  return <li>Result #{index} for "{query}"</li>;
};

function SearchBox() {
  const [query, setQuery] = useState('');
  const [deferredQuery, setDeferredQuery] = useState('');
  
  // 使用 useTransition 标记非紧急更新
  const [isPending, startTransition] = useTransition();

  const handleChange = (e) => {
    const value = e.target.value;
    setQuery(value); // 紧急更新：立即更新输入框
    
    // 非紧急更新：延迟更新列表
    startTransition(() => {
      setDeferredQuery(value);
    });
  };

  return (
    <div>
      <input
        type="text"
        value={query}
        onChange={handleChange}
        placeholder="Search..."
      />
      
      {/* 显示加载状态 */}
      {isPending && <div>Updating results...</div>}
      
      {/* 延迟更新的列表 */}
      <SlowList query={deferredQuery} />
    </div>
  );
}

5.4.2 useDeferredValue

作用: 延迟更新某个值，类似于防抖但更适合 React 并发特性
使用方式: const deferredValue = useDeferredValue(value)
参数作用:

  • value: 要延迟的值
import { useState, useDeferredValue, memo } from 'react';

const SearchResults = memo(({ query }) => {
  if (!query) return <div>Enter a search term</div>;
  
  // 模拟大量结果
  const results = [];
  for (let i = 0; i < 100; i++) {
    results.push(
      <div key={i} style={{ padding: '5px', borderBottom: '1px solid #ccc' }}>
        Result #{i + 1} for "{query}"
      </div>
    );
  }
  
  return <div>{results}</div>;
});

function SearchApp() {
  const [query, setQuery] = useState('');
  
  // 延迟查询值，保持输入响应式
  const deferredQuery = useDeferredValue(query);
  
  const isStale = query !== deferredQuery;

  return (
    <div>
      <h1>Search Demo</h1>
      
      <input
        type="text"
        value={query}
        onChange={(e) => setQuery(e.target.value)}
        placeholder="Search..."
        style={{
          width: '300px',
          padding: '8px',
          fontSize: '16px'
        }}
      />
      
      {/* 显示陈旧数据指示器 */}
      {isStale && (
        <div style={{ color: 'orange', fontSize: '14px' }}>
          Updating results...
        </div>
      )}
      
      <div style={{ 
        opacity: isStale ? 0.5 : 1,
        transition: 'opacity 0.2s',
        marginTop: '20px'
      }}>
        <SearchResults query={deferredQuery} />
      </div>
    </div>
  );
}

5.5 React 19 新 Hooks

5.5.1 use

作用: 读取 Context 或 Promise 的值，可在条件语句和循环中使用
使用方式: const value = use(resource)
参数作用:

  • resource: Context 对象或 Promise
import { createContext, use, Suspense, useState } from 'react';

// 1. 使用 Context
const ThemeContext = createContext();
const UserContext = createContext();

function ThemedButton() {
  // 可以在条件语句中使用！
  const theme = use(ThemeContext);
  const user = use(UserContext);
  
  return (
    <button style={{ 
      background: theme.background, 
      color: theme.foreground 
    }}>
      Hello, {user.name}
    </button>
  );
}

// 2. 使用 Promise（异步数据）
function fetchUserData(userId) {
  return new Promise((resolve) => {
    setTimeout(() => {
      resolve({
        id: userId,
        name: 'John Doe',
        email: 'john@example.com'
      });
    }, 2000);
  });
}

function UserProfile({ userId }) {
  // 直接使用 Promise，React 会暂停直到 Promise 解决
  const userData = use(fetchUserData(userId));
  
  return (
    <div>
      <h2>{userData.name}</h2>
      <p>Email: {userData.email}</p>
    </div>
  );
}

function App() {
  const [userId, setUserId] = useState(1);
  
  return (
    <ThemeContext.Provider value={{ background: 'blue', foreground: 'white' }}>
      <UserContext.Provider value={{ name: 'John Doe' }}>
        <div>
          <ThemedButton />
          
          <div>
            <button onClick={() => setUserId(1)}>User 1</button>
            <button onClick={() => setUserId(2)}>User 2</button>
          </div>
          
          <Suspense fallback={<div>Loading user data...</div>}>
            <UserProfile userId={userId} />
          </Suspense>
        </div>
      </UserContext.Provider>
    </ThemeContext.Provider>
  );
}

5.5.2 useOptimistic

作用: 实现乐观更新，在异步操作完成前立即更新 UI
使用方式: const [optimisticState, addOptimistic] = useOptimistic(state, updateFn)
参数作用:

  • state: 实际状态

  • updateFn: (currentState, optimisticValue) => newState

import { useOptimistic, useState, useRef } from 'react';

function MessageList({ messages, sendMessage }) {
  const formRef = useRef();
  
  const [optimisticMessages, addOptimisticMessage] = useOptimistic(
    messages,
    (state, newMessage) => [
      ...state,
      {
        ...newMessage,
        id: Math.random(), // 临时 ID
        sending: true,     // 乐观标记
        timestamp: new Date().toISOString()
      }
    ]
  );

  const formAction = async (formData) => {
    const message = formData.get('message');
    
    if (!message.trim()) return;
    
    // 立即乐观更新 UI
    addOptimisticMessage({
      text: message,
      id: Date.now() // 临时 ID
    });
    
    formRef.current.reset();
    
    // 执行真正的异步操作
    try {
      await sendMessage(message);
      // 成功后，实际的 messages prop 会更新
      // useOptimistic 会自动同步到最新状态
    } catch (error) {
      console.error('Failed to send message:', error);
      // 错误处理：可以显示错误提示
      // useOptimistic 会自动回滚到实际状态
    }
  };

  return (
    <div>
      <h2>Chat Messages</h2>
      
      <div style={{ 
        height: '300px', 
        border: '1px solid #ccc', 
        overflowY: 'auto',
        marginBottom: '1rem',
        padding: '1rem'
      }}>
        {optimisticMessages.map((message) => (
          <div 
            key={message.id}
            style={{
              padding: '0.5rem',
              margin: '0.5rem 0',
              background: message.sending ? '#e3f2fd' : '#f5f5f5',
              borderRadius: '8px',
              opacity: message.sending ? 0.7 : 1
            }}
          >
            <div>{message.text}</div>
            <div style={{ fontSize: '0.8rem', color: '#666' }}>
              {new Date(message.timestamp).toLocaleTimeString()}
              {message.sending && ' (Sending...)'}
            </div>
          </div>
        ))}
      </div>
      
      <form action={formAction} ref={formRef}>
        <input
          type="text"
          name="message"
          placeholder="Type a message..."
          style={{
            width: '200px',
            padding: '0.5rem',
            marginRight: '0.5rem'
          }}
        />
        <button type="submit">Send</button>
      </form>
    </div>
  );
}

// 使用示例
function ChatApp() {
  const [messages, setMessages] = useState([]);
  
  const sendMessage = async (messageText) => {
    // 模拟 API 调用延迟
    await new Promise(resolve => setTimeout(resolve, 1000));
    
    // 模拟 10% 的失败率
    if (Math.random() < 0.1) {
      throw new Error('Failed to send message');
    }
    
    const newMessage = {
      id: Date.now(),
      text: messageText,
      timestamp: new Date().toISOString()
    };
    
    setMessages(prev => [...prev, newMessage]);
  };
  
  return (
    <div>
      <MessageList messages={messages} sendMessage={sendMessage} />
    </div>
  );
}

5.6.2 useSyncExternalStore

作用: 订阅外部 store（用于第三方状态管理库集成）
使用方式: const state = useSyncExternalStore(subscribe, getSnapshot, getServerSnapshot?)
参数作用:

  • subscribe: 订阅函数

  • getSnapshot: 获取当前快照的函数

  • getServerSnapshot: 服务端渲染时获取快照的函数

import { useSyncExternalStore } from 'react';

// 外部 store
class CounterStore {
  constructor() {
    this.listeners = [];
    this.state = { count: 0 };
  }

  subscribe(listener) {
    this.listeners.push(listener);
    return () => {
      this.listeners = this.listeners.filter(l => l !== listener);
    };
  }

  getSnapshot() {
    return this.state;
  }

  increment() {
    this.state.count += 1;
    this.emitChange();
  }

  decrement() {
    this.state.count -= 1;
    this.emitChange();
  }

  emitChange() {
    this.listeners.forEach(listener => listener());
  }
}

const store = new CounterStore();

function ExternalStoreCounter() {
  const state = useSyncExternalStore(
    (callback) => store.subscribe(callback),
    () => store.getSnapshot()
  );

  return (
    <div>
      <p>Count: {state.count}</p>
      <button onClick={() => store.increment()}>+</button>
      <button onClick={() => store.decrement()}>-</button>
    </div>
  );
}

5.7 自定义 Hooks 示例

5.7.1 常用自定义 Hooks

import { useState, useEffect, useCallback } from 'react';

// 1. 本地存储 Hook
function useLocalStorage(key, initialValue) {
  const [storedValue, setStoredValue] = useState(() => {
    try {
      const item = window.localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch (error) {
      console.error(`Error reading localStorage key "${key}":`, error);
      return initialValue;
    }
  });

  const setValue = useCallback((value) => {
    try {
      setStoredValue(value);
      window.localStorage.setItem(key, JSON.stringify(value));
    } catch (error) {
      console.error(`Error setting localStorage key "${key}":`, error);
    }
  }, [key]);

  return [storedValue, setValue];
}

// 2. 网络请求 Hook
function useFetch(url) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    const fetchData = async () => {
      try {
        setLoading(true);
        const response = await fetch(url);
        if (!response.ok) throw new Error('Network response was not ok');
        const result = await response.json();
        setData(result);
      } catch (err) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    };

    fetchData();
  }, [url]);

  return { data, loading, error };
}

// 3. 防抖 Hook
function useDebounce(value, delay) {
  const [debouncedValue, setDebouncedValue] = useState(value);

  useEffect(() => {
    const handler = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);

    return () => {
      clearTimeout(handler);
    };
  }, [value, delay]);

  return debouncedValue;
}

// 使用示例
function ExampleApp() {
  const [name, setName] = useLocalStorage('name', '');
  const debouncedName = useDebounce(name, 500);
  const { data, loading, error } = useFetch(
    debouncedName ? `/api/users?search=${debouncedName}` : null
  );

  return (
    <div>
      <input
        value={name}
        onChange={(e) => setName(e.target.value)}
        placeholder="Search users..."
      />
      
      {loading && <div>Loading...</div>}
      {error && <div>Error: {error}</div>}
      {data && (
        <ul>
          {data.users.map(user => (
            <li key={user.id}>{user.name}</li>
          ))}
        </ul>
      )}
    </div>
  );
}
