四、其他功能

4.1 生命周期函数

1. onMount

import { onMount } from 'solid-js';

export default function Test() {
  onMount(() => {
    console.log('组件已挂载到 DOM');
    // 通常用于：
    // - 初始化第三方库（如地图、图表）
    // - 添加全局事件监听（需配合 onCleanup 移除）
    // - 启动定时器
  });

  return <div>Hello</div>;
};

特点：

  • 在组件首次渲染完成后执行（DOM 已存在）
  • 只运行一次
  • 类似 React 的 useEffect(..., [])

2. onCleanup

import { onMount, onCleanup } from 'solid-js';

const TimerComponent = () => {
  let intervalId: number;

  onMount(() => {
    intervalId = setInterval(() => {
      console.log('Tick');
    }, 1000);
  });

  onCleanup(() => {
    console.log('🧹 组件即将卸载');
    clearInterval(intervalId); // 清理副作用
  });

  return <div>计时器运行中...</div>;
};

关键点：

  • onCleanup 必须在响应式上下文内调用（如组件函数、effect 内）
  • 它会自动关联到当前作用域（组件或 effect）
  • 即使组件未挂载（如条件渲染被跳过），也会在作用域销毁时调用

3. createEffect

  1. 正常使用

    import { createSignal, createEffect } from 'solid-js';

export default function Test() {
 const [userId, setUserId] = createSignal(1);

  // 当 userId 变化时自动执行
  createEffect(() => {
    console.log('userId 变为:', userId());
  });

  return (
    <div>
      <button onClick={() => setUserId(u => u + 1)}>切换用户</button>
    </div>
  );
};

    行为：

    • 初始执行一次
    • 每当依赖的响应式值变化时重新运行
    • 自动追踪依赖（无需手动写 deps 数组）
    • 类似 React 的 useEffect（但更智能）

  2. 跳过首次

    Solid 没有直接提供“跳过首次”的 effect，但可轻松实现：

    import { createSignal, createEffect } from 'solid-js';

export default function Test() {
 const [count, setCount] = createSignal(0);
  let isFirst = true;

  createEffect(() => {
    if (isFirst) {
      isFirst = false;
      return;
    }
    console.log('非首次更新:', count());
  });

  return <button onClick={() => setCount(c => c + 1)}>+{count()}</button>;
};

4.2 createResource

在 Solid.js 中，createResource 是用于声明式管理异步数据加载（如 API 请求、数据库查询等）的核心响应式原语。它将 加载状态（loading）、错误处理（error）和最终数据（data） 统一整合在一个响应式结构中，非常适合与 UI 无缝结合。

4.2.1 参数与返回值

  • 用途: 通用的 异步资源管理器，适用于任何组件（路由或非路由）。

  • 函数签名

    function createResource<T, S>(
  source: S | (() => S),
  fetcher: (k: S, prev: T | undefined) => Promise<T>,
  options?: ResourceOptions<T>
): [Resource<T>, ResourceActions<T>];

  • 参数说明

    参数类型必填说明
    sourceS | (() => S)是依赖源• 可以是原始值、信号、或函数• 当其变化时触发fetcher
    fetcher(key: S, prev: T | undefined) => Promise<T>是数据获取函数• 第一个参数是 source 的当前值• 第二个参数是上一次成功的结果（可用于优化）
    optionsResourceOptions<T>否高级配置（见下表）

    • options 可选配置：

      interface ResourceOptions<T> {
  initialValue?: T;         // 初始值（SSR 时很有用）
  name?: string;            // 调试名称
  deferStream?: boolean;
  storage?: {               // 自定义存储（如 localStorage 缓存）
    read: (k: string) => T | undefined;
    write: (k: string, v: T) => void;
    serialize?: (v: T) => string;
    deserialize?: (v: string) => T;
  };
  maxAge?: number;          // 缓存过期时间（ms）
}

    • 示例：

      // 基础用法
const [user] = createResource(userId, fetchUser);

// 带初始值（用于 SSR）
const [data] = createResource(
  () => routeParams.id,
  fetchData,
  { initialValue: initialDataFromServer }
);

  • 返回值：元组 [Resource<T>, ResourceActions<T>]

    • Resource<T>（第一个元素）

      • 是一个 信号函数：data() 获取值
      • 值类型：T | undefined | initialValue

    • ResourceActions<T>（第二个元素）

      interface ResourceActions<T> {
  readonly loading: boolean;
  readonly error?: Error;
  mutate: (v: T | ((prev: T | undefined) => T)) => void; // 手动修改数据
  refetch: (info?: { value?: S; refetching?: boolean }) => Promise<T | undefined>; //重新加载
}

    • 使用方式：

      const [user, { loading, error, refetch }] = createResource(userId, fetchUser);

return (
  <div>
    {loading && <p>加载中...</p>}
    {error && <p>错误：{error.message}</p>}
    {user() && <div>{user().name}</div>}
    <button onClick={() => refetch()}>刷新</button>
  </div>
);

4.2.2 示例用法

  1. 基础用法（无依赖项）

    import { createResource } from 'solid-js';

const fetchUser = async () => {
  await new Promise(r => setTimeout(r, 1500)); // 模拟延迟
  return { name: "张三", email: "zhangsan@example.com" };
};

export default function Test() {

  // 第一个参数是 false 或 undefined 表示无依赖
  // 使用 createResource 异步获取数据
  const [user, { loading, error }] = createResource(fetchUser);

   // loading 和 error 需要访问返回的资源对象
  if (loading) return <div>加载中...</div>;
  if (error) return <div>错误: {error.message}</div>;

  return (
    <div>
      <h1>欢迎, {user()?.name}!</h1>
      <p>邮箱: {user()?.email}</p>
    </div>
  );
};

    注意：user() 可能为 undefined，所以要用 可选链 ?.

  2. 动态加载（带依赖项）

    import { createResource,createSignal } from 'solid-js';

const fetchUser = async (userId) => {
  await new Promise(r => setTimeout(r, 1500));
  return {name: `张三-${userId}`, email: "zhangsan@example.com" };
};

export default function Test() {
  // 创建依赖源
  const [userId,setUserId] = createSignal(1);

  // 传入依赖源
  const [user] = createResource(userId, fetchUser);

  return (
    <div>
      {/* 通过返回数据直接获取状态 */}
      {user.loading && <div>加载中...</div>}
      {user.error && <div>错误: {user.error.message}</div>}
      {user() && (
        <>
          <h1>欢迎, {user().name}!</h1>
          <p>邮箱: {user().email}</p>
        </>
      )}
      <button onClick={() => setUserId(c => c + 1)}>修改用户id</button>
    </div>
  );
}

  3. 配置options

    1. 配置项

      createResource(source, fetcher, {
    initialValue: null,     // 初始值（避免 undefined）
    ssrLoadFrom: "initial", // SSR 行为（"initial" | "server" | "never"）
    storage: customStorage, // 自定义缓存策略（高级）
    deferStream: true       // 流式响应（实验性）
});

    2. 示例代码

      import { createResource,createSignal } from 'solid-js';

const fetchUser = async (userId) => {
  await new Promise(r => setTimeout(r, 1500));
  return {name: `张三-${userId}`, email: "zhangsan@example.com" };
};

export default function Test() {

  const [userId,setUserId] = createSignal(1);

  // 添加初始化数据
  const [user] = createResource(
    userId, 
    fetchUser, 
    { initialValue: { name: '未知用户', email: '空' } }
  );

  return (
    <div>
      {user.loading && <div>加载中...</div>}
      {user.error && <div>错误: {user.error.message}</div>}
      {user() && (
        <>
          <h1>欢迎, {user().name}!</h1>
          <p>邮箱: {user().email}</p>
        </>
      )}
      <button onClick={() => setUserId(c => c + 1)}>修改用户id</button>
    </div>
  );
}

  4. 手动控制

    createResource 返回的第二个元素还包含 mutate 和 refetch 方法：

    import { createResource } from 'solid-js';

let count = 0;

const fetchUser = async () => {
  count++;
  await new Promise(r => setTimeout(r, 1500));
  return {name: `张三-${count}`, email: "zhangsan@example.com" };
};

export default function Test() {
  const [user,{ refetch, mutate }] = createResource(fetchUser);

  return (
    <div>
      {user.loading && <div>加载中...</div>}
      {user.error && <div>错误: {user.error.message}</div>}
      {user() && (
        <>
          <h1>欢迎, {user().name}!</h1>
          <p>邮箱: {user().email}</p>
        </>
      )}

      {/* 强制重新加载（忽略缓存） */}
      <button onClick={() => refetch()}>刷新</button>

      {/* 乐观更新：立即修改 UI */}
      <button onClick={() => {
        if (user()) {
          // 乐观更新：立即修改本地数据
          mutate({ ...user(), name: `张三-${count+2}` });
          // 可选：稍后同步到服务器
          // refetch();
        }
      }}>
        count + 2
      </button>
    </div>
  );
}

    mutate(updater)：直接修改本地数据（不触发网络请求）

4.3 <Suspense>标签

在 SolidJS 中，<Suspense> 是一个内置组件，用于 优雅地处理异步依赖（如数据加载、动态导入）期间的加载状态，避免 UI 卡顿或空白，提升用户体验。

它的核心作用是：当子组件中发生异步操作（如 await 一个 Promise）时，暂停渲染该子树，并显示 fallback 内容，直到异步操作完成。

  1. 基本语法

    <Suspense fallback={<div>加载中...</div>}>
  <AsyncComponent />
</Suspense>
    • fallback：异步操作进行时显示的内容（可以是 JSX、字符串等）。
    • 子组件中只要使用了 资源（Resource） 或 顶层 await（在 Solid 1.7+ 支持），就会触 发 Suspense。

  2. 核心使用场景：配合 createResource

    <Suspense> 最常与 createResource 配合使用，后者用于声明式地加载异步数据。

    示例：加载用户信息

    • 父组件

      import { Suspense } from 'solid-js';
import Child from './Child';

export default function Parent() {
 return (
    <div>
      <h1>用户资料</h1>
      {/* 包裹可能 suspend 的组件 */}
      <Suspense fallback={<p>正在加载用户信息，请稍候...</p>}>
        <Child userId={123} />
      </Suspense>
    </div>
  );
}

    • 子组件

      import { createResource } from 'solid-js';

// 模拟 API 调用
async function fetchUser(id) {
  await new Promise(r => setTimeout(r, 1500)); // 模拟延迟
  return { id, name: "张三", email: "zhangsan@example.com" };
}

export default function Child({ userId }) {
  // 创建资源（自动触发加载）
  const [user] = createResource(() => userId, fetchUser);

  // 注意：这里直接访问 user()，如果还在加载会 throw Promise！
  // 但被 Suspense 捕获，所以不会崩溃，而是显示 fallback
  return (
    <div>
      <h2>用户: {user()?.name}</h2>
      <p>邮箱: {user()?.email}</p>
    </div>
  );
}

    执行流程：

    1. Child 渲染，调用 user()。
    2. 此时数据未就绪，user() 抛出一个 Promise（这是 Solid Resource 的内部机制）。
    3. <Suspense> 捕获这个 Promise，暂停渲染 Child，转而显示 fallback。
    4. 1.5 秒后，Promise resolve，<Suspense> 自动切换回 Child 并渲染真实内容。

  3. 重要注意事项

    1. 必须包裹会“suspend”的组件

      如果子组件中没有触发异步挂起（如未使用 createResource 或未 await Promise），<Suspense> 不会生效。

    2. 不要手动 try/catch Resource

      错误应由 <ErrorBoundary> 处理，而不是在组件内 catch：

    3. Fallback 必须是轻量的

      避免在 fallback 中放入复杂逻辑或状态，它应该是一个静态或简单响应式组件。
