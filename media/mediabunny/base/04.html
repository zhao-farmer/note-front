四、转换合成

4.1 基础合成

4.1.1 快速转换（Conversion）

  1. 用法总结

    1. 核心概念

      1. Conversion：媒体转换/合成的核心实例，负责整个转码流程
      2. Input：输入源（文件、URL、流等）
      3. Output：输出配置（格式 + 输出目标）
      4. execute()：执行转换
      5. BufferTarget：输出到内存 Buffer（最常用）

    2. 最简标准流程（对应你的代码）

      1. 导入模块
2. 创建 Input（输入媒体）
3. 创建 Output（指定格式+输出目标）
4. await Conversion.init({ input, output })
5. await conversion.execute()  执行转换
6. 从 output.target.buffer 获取结果

    3. 重点规则（必须记住）

      1. Output 必须是全新的

        • 不能添加过轨道、不能设置过标签
        • 必须是 pending 状态（未开始）
        • 不能复用

      2. Conversion 会自动处理

        • 能复制轨道就直接复制（最快）
        • 不能复制就自动转码
        • 自动丢弃不支持的轨道

      3. 建议判断 isValid（官方推荐）

        if (!conversion.isValid) {
    console.log(conversion.discardedTracks);
    return;
}

  2. 代码

    <!DOCTYPE html>
<html lang="zh-CN">
    <head>
        <meta charset="UTF-8" />
        <title>MediaBunny 快速转换</title>
    </head>

    <body>
        <video id="player" controls height="300" style="margin-top:20px"></video>

        <pre id="output"></pre>

        <script type="module">
            import { Input, Output, WebMOutputFormat, BufferTarget, Conversion, UrlSource, ALL_FORMATS } from "../../js/mediabunny.1.42.mjs";

            const player = document.getElementById("player");
            // 日志
            const outputEl = document.getElementById("output");
            function log(msg) {
                outputEl.textContent += msg + "\n";
            }

            const VIDEO_URL = "../../video/04.mp4";
            (async () => {
                try {
                    log("1. 创建输入源...");
                    const input = new Input({
                        formats: ALL_FORMATS,
                        source: new UrlSource(VIDEO_URL),
                    });

                    log("2. 创建输出（WebM）...");
                    const output = new Output({
                        format: new WebMOutputFormat(),
                        target: new BufferTarget(),
                    });

                    log("3. 初始化转换任务...");
                    const conversion = await Conversion.init({ input, output });
                    if (!conversion.isValid) {
                        log("转换无效：" + conversion.discardedTracks.length);
                        return;
                    }

                    log("4. 正在转换...");
                    await conversion.execute();

                    log("5. 加载webm效果...");
                    const blob = new Blob([output.target.buffer], { type: "video/webm" });
                    player.src = URL.createObjectURL(blob);
                } catch (err) {
                    log("转换失败：" + err.message);
                    console.error(err);
                }
            })();
        </script>
    </body>
</html>

4.1.2 常用扩展功能

  1. 扩展功能分类

    1. 进度监听（onProgress）

      conversion.onProgress = progress => {
    // 0 ~ 1
    const percent = (progress * 100).toFixed(1);

    2. 取消转换

      await conversion.cancel();
      • 会抛出 ConversionCanceledError
      • 自动释放资源

    3. 监听输出文件大小

      output.target.on("write", ({ end }) => {
    const size = end;
});

  2. 代码实例

    <!DOCTYPE html>
<html lang="zh-CN">
    <head>
        <meta charset="UTF-8" />
        <title>MediaBunny 转换 + 进度条 + 取消 + 文件大小</title>
        <style>
            #progressBar {
                width: 230px;
                height: 15px;
                background: #eee;
                margin: 10px 0;
                display: none;
            }
            #progressBar .fill {
                height: 100%;
                background: #42b983;
                width: 0%;
                transition: width 0.1s;
            }
            button {
                padding: 8px 16px;
                margin: 5px;
                cursor: pointer;
            }
        </style>
    </head>

    <body>
        <video id="player" controls height="300" style="margin-top:20px"></video>

        <div>
            <button id="startBtn">开始转换</button>
            <button id="cancelBtn" disabled>取消转换</button>
        </div>
        <div id="progressBar">
            <div class="fill"></div>
        </div>

        <pre id="output"></pre>

        <script type="module">
            import { Input, Output, WebMOutputFormat, BufferTarget, Conversion, ConversionCanceledError, UrlSource, ALL_FORMATS } from "../../js/mediabunny.1.42.mjs";

            // 页面元素
            const outputEl = document.getElementById("output");
            const player = document.getElementById("player");
            const startBtn = document.getElementById("startBtn");
            const cancelBtn = document.getElementById("cancelBtn");
            const progressBar = document.getElementById("progressBar");
            const progressFill = progressBar.querySelector(".fill");

            // 视频源
            const VIDEO_URL = "../../video/04.mp4";

            // 全局实例
            let currentConversion = null;
            let lastProgressLine = -1;
            let currentFileSize = 0; // 文件大小

            // 清空日志
            function clearLog() {
                outputEl.textContent = "";
                lastProgressLine = -1;
                currentFileSize = 0;
            }

            // 普通日志
            function log(msg) {
                outputEl.textContent += msg + "\n";
            }

            // 统一更新：进度 + 文件大小
            function updateProgress(p) {
                const mb = (currentFileSize / 1024 / 1024).toFixed(2);
                const text = outputEl.textContent;
                const lines = text.split("\n");

                const info = `转换进度：${p}%  |  已输出：${mb} MB`;

                if (lastProgressLine === -1) {
                    lastProgressLine = lines.length;
                    lines.push(info);
                } else {
                    lines[lastProgressLine] = info;
                }

                outputEl.textContent = lines.join("\n");
                progressFill.style.width = p + "%";
            }

            // 重置 UI
            function resetUI() {
                startBtn.disabled = false;
                cancelBtn.disabled = true;
                progressBar.style.display = "none";
                progressFill.style.width = "0%";
                currentConversion = null;
            }

            // 开始转换
            async function startConversion() {
                player.src = "";
                clearLog();
                resetUI();

                startBtn.disabled = true;
                cancelBtn.disabled = false;
                progressBar.style.display = "block";
                log("=== 开始媒体转换 ===");

                try {
                    log("1. 创建输入源...");
                    const input = new Input({
                        formats: ALL_FORMATS,
                        source: new UrlSource(VIDEO_URL),
                    });

                    log("2. 创建输出（WebM）...");
                    const output = new Output({
                        format: new WebMOutputFormat(),
                        target: new BufferTarget(),
                    });

                    // ========== 【新增】监听写入字节大小 ==========
                    output.target.on("write", ({ end }) => {
                        currentFileSize = Math.max(currentFileSize, end);
                    });

                    log("3. 初始化转换任务...");
                    const conversion = await Conversion.init({ input, output });
                    currentConversion = conversion;

                    if (!conversion.isValid) {
                        log("转换无效：" + conversion.discardedTracks.length);
                        resetUI();
                        return;
                    }

                    log("4. 正在转换...");
                    conversion.onProgress = progress => {
                        const p = (progress * 100).toFixed(1);
                        updateProgress(p);
                    };

                    await conversion.execute();
                    updateProgress("100");
                    log("转换完成！");

                    const blob = new Blob([output.target.buffer], { type: "video/webm" });
                    player.src = URL.createObjectURL(blob);
                } catch (err) {
                    if (err instanceof ConversionCanceledError) {
                        log("转换已取消");
                    } else {
                        log("转换失败：" + err.message);
                        console.error(err);
                    }
                } finally {
                    resetUI();
                }
            }

            // ==============================================
            // 取消转换
            // ==============================================
            async function cancelConversion() {
                if (currentConversion) {
                    await currentConversion.cancel();
                    progressFill.style.width = "0%";
                    updateProgress(0);
                }
            }

            // 绑定事件
            startBtn.addEventListener("click", startConversion);
            cancelBtn.addEventListener("click", cancelConversion);

            log("准备就绪，点击【开始转换】");
        </script>
    </body>
</html>

4.2 配置信息

4.2.1 视频配置

1. 配置选项

ConversionVideoOptions 是 MediaBunny Conversion 转换实例的视频全局配置，用于统一控制所有输入视频轨道的丢弃、缩放、旋转、裁剪、帧率、编码、码率、硬件加速、自定义帧处理等行为。

  1. 完整类型定义

    type ConversionVideoOptions = {
    // 轨道控制
    discard?: boolean;

    // 尺寸与缩放
    width?: number;
    height?: number;
    fit?: "fill" | "contain" | "cover";

    // 旋转
    rotate?: 0 | 90 | 180 | 270;
    allowRotationMetadata?: boolean;

    // 裁剪
    crop?: { left: number; top: number; width: number; height: number };

    // 帧率
    frameRate?: number;

    // 编码与码率
    codec?: VideoCodec;
    bitrate?: number | Quality;
    alpha?: "discard" | "keep"; // 默认 discard
    hardwareAcceleration?: "no-preference" | "prefer-hardware" | "prefer-software";
    keyFrameInterval?: number;
    forceTranscode?: boolean;

    // 自定义帧处理
    process?: (sample: VideoSample) => MaybePromise<CanvasImageSource | VideoSample | (CanvasImageSource | VideoSample)[] | null>;
    processedWidth?: number;
    processedHeight?: number;

    // 轨道分组
    group?: OutputTrackGroup | OutputTrackGroup[];
};

  2. 配置项详细说明

    • 2.1 轨道控制

      配置项类型说明
      discardboolean丢弃所有视频轨道，只保留音频

    • 2.2 视频缩放（尺寸适配）

      配置项类型说明
      widthnumber输出视频宽度（px）
      heightnumber输出视频高度（px）
      fitfill / contain / cover宽高比不一致时的适配模式
      • fill：拉伸填满（变形）
      • contain：等比包含（黑边）
      • cover：等比覆盖（裁剪）
      • 只传宽/高 → 自动等比计算

    • 2.3 视频旋转

      配置项类型说明
      rotate0/90/180/270顺时针旋转角度
      allowRotationMetadataboolean是否允许使用旋转元数据（默认 true）

      执行顺序：旋转 → 裁剪 → 缩放

    • 2.4 视频裁剪

      配置项类型说明
      crop{left,top,width,height}旋转后、缩放前裁剪区域

    • 2.5 帧率控制

      配置项类型说明
      frameRatenumber强制输出帧率（Hz），不设置则使用原视频

    • 2.6 编码与转码控制

      配置项类型说明
      codecVideoCodec输出编码：avc/hevc/vp8/vp9/av1
      bitratenumber / Quality码率（bps）或画质等级
      alphadiscard / keep透明通道处理（默认丢弃）
      hardwareAccelerationno-preference / prefer-hardware / prefer-software硬件加速策略
      keyFrameIntervalnumber关键帧最大间隔（秒）
      forceTranscodeboolean强制转码，禁止直接流复制

      Quality 画质枚举

      • QUALITY_VERY_LOW
      • QUALITY_LOW
      • QUALITY_MEDIUM
      • QUALITY_HIGH
      • QUALITY_VERY_HIGH

    • 2.7 自定义帧处理（核心：水印/滤镜/帧修改）

      配置项类型说明
      process函数逐帧处理：Canvas 绘制、水印、滤镜、灰度、裁剪等
      processedWidthnumber处理后帧宽度（告诉编码器）
      processedHeightnumber处理后帧高度（告诉编码器）

      process 可返回：

      • Canvas / VideoSample
      • 数组（生成多帧）
      • null（丢弃该帧）

    • 2.8 轨道分组

      配置项类型说明
      groupOutputTrackGroup输出轨道分组，用于多音轨/多视频轨道管理

  3. 执行顺序（固定）

    丢弃检查 → 旋转 → 裁剪 → 缩放/适配 → 帧率调整 → 自定义 process → 编码/转码

  4. 最常用标准配置示例

    • 4.1 缩放为 720p（等比包含）

      video: {
    width: 1280,
    height: 720,
    fit: 'contain'
}

    • 4.2 强制转码 + 高质量

      video: {
    codec: 'avc',
    bitrate: QUALITY_HIGH,
    forceTranscode: true
}

    • 4.3 旋转 90 度 + 裁剪

      video: {
    rotate: 90,
    crop: { left: 100, top: 100, width: 800, height: 800 }
}

    • 4.4 自定义帧处理（灰度滤镜）

      video: {
    process: sample => {
        const canvas = new OffscreenCanvas(sample.displayWidth, sample.displayHeight);
        const ctx = canvas.getContext("2d");
        ctx.filter = "saturate(0)";
        sample.draw(ctx, 0, 0);
        return canvas;
    };
}

  5. 核心注意事项

    • 配置会作用于所有视频轨道
    • 需要单独轨道配置 → 使用 track-specific options
    • forceTranscode: true 可避免直接流复制，强制走编码流程
    • process 是实现水印、滤镜、美颜、文字叠加的唯一入口
    • 旋转、裁剪、缩放有固定执行顺序

  6. 与 VideoEncodingConfig 的区别

    特性ConversionVideoOptionsVideoEncodingConfig
    作用位置Conversion 转换级别VideoSampleSource 编码源级别
    使用方式Conversion.init({ video })new VideoSampleSource(config)
    控制范围全局所有视频轨道单条编码轨道
    功能缩放、旋转、裁剪、process、丢弃编码参数、码率、编码器配置

				await conversion.execute();
				updateProgress("100");

4.2.2 音频配置

1. 配置选项

ConversionAudioOptions 是 MediaBunny Conversion 转换实例的音频全局配置，用于统一控制所有输入音频轨道的丢弃、声道、采样率、编码、码率、强制转码、自定义采样处理等行为。

  1. 完整类型定义

    type ConversionAudioOptions = {
    // 轨道控制
    discard?: boolean;

    // 编码与转码
    codec?: AudioCodec;
    bitrate?: number | Quality;
    forceTranscode?: boolean;

    // 音频重采样与声道
    numberOfChannels?: number;
    sampleRate?: number;
    sampleFormat?: "u8" | "s16" | "s32" | "f32";

    // 自定义音频采样处理
    process?: (sample: AudioSample) => MaybePromise<AudioSample | AudioSample[] | null>;
    processedNumberOfChannels?: number;
    processedSampleRate?: number;
};

type MaybePromise<T> = T | Promise<T>;

  2. 配置项详细说明

    • 2.1 轨道控制

      配置项类型说明
      discardboolean丢弃所有音频轨道，只保留视频

    • 2.2 音频重采样与声道配置

      配置项类型说明
      numberOfChannelsnumber输出音频声道数（1=单声道mono，2=立体声stereo）
      sampleRatenumber输出音频采样率（Hz，如44100、48000）
      sampleFormatu8/s16/s32/f32音频采样格式（无符号8位/有符号16位/有符号32位/32位浮点）
      • 声道数与输入不一致时，MediaBunny 会按照 Web Audio API 算法进行上混/下混
      • 采样率与输入不一致时，MediaBunny 会自动执行重采样

    • 2.3 编码与转码控制

      配置项类型说明
      codecAudioCodec输出音频编码格式（需与输出文件兼容）
      bitratenumber / Quality音频码率（bps）或主观画质等级，用于压缩音频
      forceTranscodeboolean强制转码，禁止直接复制媒体流数据
      • 配置 bitrate 会强制触发转码
      • 未配置 bitrate 但需要转码时，默认使用 QUALITY_HIGH
      • 编码不兼容输出文件时，音频轨道会被丢弃

    • 2.4 自定义音频采样处理（核心：音效/变换/时间修改）

      配置项类型说明
      process函数声道混音、重采样后，逐音频采样处理（添加音效、修改时间戳等）
      processedNumberOfChannelsnumber手动处理后声道数（告知编码器）
      processedSampleRatenumber手动处理后采样率（告知编码器）

      process 函数可返回：

      • AudioSample（处理后的单采样）
      • AudioSample[]（处理后的多采样）
      • null（丢弃当前采样）

  3. 执行顺序（固定）

    丢弃检查 → 声道混音 → 重采样 → 自定义 process → 编码/转码

  4. 最常用标准配置示例

    • 4.1 转换为单声道 + 48000Hz 采样率

      audio: {
    numberOfChannels: 1,
    sampleRate: 48000
}

    • 4.2 丢弃所有音频轨道

      audio: {
    discard: true;
}

    • 4.3 强制转码 + 高质量音频

      audio: {
    codec: 'aac',
    bitrate: QUALITY_HIGH,
    forceTranscode: true
}

  5. 核心注意事项

    • 配置会作用于所有音频轨道
    • 需要单独轨道配置 → 使用 track-specific options
    • forceTranscode: true 可避免直接流复制，强制走音频编码流程
    • process 是实现音频音效、音量调整、混音、时间修改的唯一入口
    • 声道混音、重采样、自定义处理、编码有固定执行顺序

  6. 与 AudioEncodingConfig 的区别

    特性ConversionAudioOptionsAudioEncodingConfig
    作用位置Conversion 转换级别AudioSampleSource 编码源级别
    使用方式Conversion.init({ audio })new AudioSampleSource(config)
    控制范围全局所有音频轨道单条编码轨道
    功能声道、采样率、混音、process、丢弃音频编码参数、码率、编码器配置

4.2.3 元数据信息

1. 配置信息

tags 是 MediaBunny Conversion 转换实例的顶层配置项，用于精细化控制输出文件的元数据（Metadata），支持三种核心操作：直接设置、增量修改、完全清空。

  1. 核心特性

    • 默认行为：输入文件的所有元数据会自动复制到输出文件
    • 控制方式：支持「对象配置」或「回调函数」两种形式
    • 支持的元数据类型：标题、艺术家、封面、评论、专辑信息等（取决于输出格式支持）

  2. 三种核心用法详解

    • 2.1 直接设置：完全替换元数据

      直接传入一个对象，会完全覆盖输入的所有元数据，只保留你指定的标签。

      // 示例：设置自定义标题和艺术家
const conversion = await Conversion.init({
    // ...
    tags: {
        title: "re:Turning",
        artist: "Alexander Panos",
    },
    // ...
});

    • 2.2 增量修改：基于输入元数据扩展/删除

      传入回调函数，接收原始输入元数据 inputTags，返回修改后的对象，可实现「保留+新增/删除」的效果。

      // 示例：保留原元数据 + 新增封面 + 删除评论
const conversion = await Conversion.init({
  // ...
  tags: (inputTags) => ({
    ...inputTags,                // 保留所有原有元数据
    images: [{                    // 新增封面图片
      data: new Uint8Array(...),  // 图片二进制数据
      mimeType: 'image/jpeg',
      kind: 'coverFront'          // 封面类型：正面封面
    }],
    comment: undefined            // 删除原有的 comment 字段
  }),
  // ...
});

    • 2.3 完全清空：移除所有元数据

      传入空对象 {}，会清空输出文件的所有元数据，只保留文件必要信息。

      // 示例：输出无任何元数据的文件
const conversion = await Conversion.init({
    // ...
    tags: {},
    // ...
});

  3. 关键注意事项

    1. 格式兼容性：不同容器格式（如 MP4、WebM、MP3）支持的元数据字段不同，超出格式支持的标签会被自动忽略。
    2. 封面图片特殊说明：images 数组仅在音频/部分视频格式中支持，需提供完整的二进制数据、MIME 类型和 kind（如 coverFront）。
    3. 回调函数权限：回调函数中修改 inputTags 本身不会生效，必须返回一个新对象作为最终元数据。
    4. 默认值优先级：未设置 tags 时，默认行为是「复制所有输入元数据」；设置了 tags 后，默认行为被覆盖，以配置为准。

  4. 常见使用场景

    场景推荐用法
    统一批量修改标题/艺术家直接设置对象
    为视频/音频添加封面回调函数 + images 字段
    隐私保护/脱敏处理传入空对象 {}
    保留原数据，仅修改个别字段回调函数扩展 inputTags

            const outputEl = document.getElementById("output");
            const player = document.getElementById("player");

            // 视频源
            const VIDEO_URL = "../../video/04.mp4";
            const COVER_URL = "../../image/002.jpg";

            // 文件大小
            let currentFileSize = 0;
            let lastProgressLine = -1;

            // 普通日志
            function log(msg) {
                outputEl.textContent += msg + "\n";
            }

            // 进度覆盖日志
            function updateProgress(p) {
                const mb = (currentFileSize / 1024 / 1024).toFixed(2);
                const text = outputEl.textContent;
                const lines = text.split("\n");

                const info = `进度：${p}%  | 已输出：${mb} MB`;

                if (lastProgressLine === -1) {
                    lastProgressLine = lines.length;
                    lines.push(info);
                } else {
                    lines[lastProgressLine] = info;
                }

                outputEl.textContent = lines.join("\n");
            }

            // 自动执行
            (async () => {
                try {
                    log("=== 自动开始 音频处理转换 ===");

                    log("0. 图片加载并改为 Uint8Array...");
                    const coverBlob = await fetch(COVER_URL).then(res => res.blob());
                    const coverBuffer = await coverBlob.arrayBuffer();
                    const coverData = new Uint8Array(coverBuffer);

                    log("1. 创建输入源...");
                    const input = new Input({
                        formats: ALL_FORMATS,
                        source: new UrlSource(VIDEO_URL),
                    });

                    log("2. 创建输出（WebM）...");
                    const output = new Output({
                        format: new Mp4OutputFormat(),
                        target: new BufferTarget(),
                    });

                    // 监听输出文件大小
                    output.target.on("write", ({ end }) => {
                        currentFileSize = Math.max(currentFileSize, end);
                    });

                    log("3. 初始化转换任务（处理音频）...");

                    const conversion = await Conversion.init({
                        input,
                        output,
                        // 视频保持原样
                        video: {},
                        // 音频保持原样
                        audio: {},
                        tags: inputTags => ({
                            ...inputTags, // 保留所有原有元数据
                            images: [
                                {
                                    // 新增封面图片
                                    data: coverData, // 图片二进制数据
                                    mimeType: "image/jpeg",
                                    kind: "coverFront", // 封面类型：正面封面
                                },
                            ],
                            comment: undefined, // 删除原有的 comment 字段
                        }),
                    });

                    if (!conversion.isValid) {
                        log("转换无效：" + conversion.discardedTracks.length);
                        return;
                    }

                    log("4. 正在转换（视频处理中）...");
                    conversion.onProgress = progress => {
                        const p = (progress * 100).toFixed(1);
                        updateProgress(p);
                    };

                    await conversion.execute();
                    updateProgress("100");
                    log("视频转换完成！");

                    log("5. 播放处理后的视频...");
                    const blob = new Blob([output.target.buffer], { type: "video/mp4" });
                    player.src = URL.createObjectURL(blob);
                } catch (err) {
                    log("错误：" + err.message);
                    console.error(err);
                }
            })();
        </script>
    </body>
</html>
