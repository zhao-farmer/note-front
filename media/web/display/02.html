2.1 快速理解

2.1.1 基础使用

  1. 代码示例

    <!DOCTYPE html>
<html>

<head>
  <meta charset="UTF-8">
  <title>PIXI8 GIF 案例</title>
  <script src="./js/pixi.js"></script>
  <script src="./js/packages/gif.js"></script>
</head>

<body>
  <script>
    (async () => {

      // 1.初始化 PIXI 应用
      const app = new PIXI.Application();
      await app.init({
        width: 600,
        height: 400,
        backgroundColor: 0x222222
      });
      document.body.appendChild(app.canvas);

      // 2. 加载 GIF 资源（官方方式）
      const gif = await PIXI.Assets.load('./img/demo.gif');

      // 2. 创建gif精灵
      const sprite = new PIXI.GifSprite(gif);
      // 或者直接使用动画精灵
      // const sprite = new PIXI.AnimatedSprite(gif.textures);
      // 添加到舞台
      app.stage.addChild(sprite);

      // 4. 播放
      sprite.animationSpeed = 0.2; // 播放速度
      sprite.play();

       // 居中
      sprite.anchor.set(0.5);
      sprite.x = app.screen.width / 2;
      sprite.y = app.screen.height / 2;

      // 缩放
      sprite.scale.set(0.6);
    })();
  </script>
</body>

</html>

      // 2. 添加一个精灵
      const app = new PIXI.Application();

  <script>
    (async () => {

      // 1. 添加一个精灵
      const app = new PIXI.Application();
      await app.init({
        width: 600,
        height: 400,
        backgroundColor: 0x222222
      });
      document.body.appendChild(app.canvas);

      // 2. 添加gif动画
      const gif = await PIXI.Assets.load('./img/demo.gif');
      const sprite = new PIXI.GifSprite({
        source: gif,
      });


      // 2. 打印精灵属性
      const totalDurationMs = sprite.duration;
      const totalFrames = sprite.totalFrames;
      const frames = gif.frames;
      console.log('GIF 总时长:', totalDurationMs, 'ms');
      console.log('总帧数:', totalFrames);
      console.log('帧数组:', frames);
      console.log('==============================');

      // 4. 打印帧属性
      for (let i = 0; i < frames.length; i++) {
        if(i !== 0 && i !== frames.length - 1) {
          continue;
        }
        const frame = gif.frames[i];
        console.log('帧纹理:', frame.texture);
        console.log('帧起始时间:', frame.start, 'ms');
        console.log('帧结束时间:', frame.end, 'ms');
        console.log('帧时长:', frame.end - frame.start, 'ms');
        console.log(`帧索引-${i}属性:`, frame);
      }
      console.log('-----------------');

      // 5. 帧变化时获取的属性
      sprite.loop = false;      
      sprite.onFrameChange = (frameIndex) => {
        // 索引与起始时间对应
        console.log('当前帧索引:', frameIndex);
        console.log('当前帧开始时间:', frames[frameIndex].start, 'ms');

        // 精灵相关
        console.log('精灵上的帧索引:', sprite.currentFrame);
        console.log('精灵上的播放进度:', sprite.progress);
      };
    })();
  </script>
</body>

2.2.1 帧展示与随机下载

  1. 代码示例

    <!DOCTYPE html>
<html>

<head>
  <meta charset="UTF-8">
  <title>PIXI8 GIF 帧展示 + 随机下载</title>
  <script src="./js/pixi.js"></script>
  <script src="./js/packages/gif.js"></script>

</head>

<body>
  <div style="position:fixed; top:10px; left:10px;">
    <button onclick="doDownload()">随机下载</button>
  </div>
  <script>
    (async () => {
      // 1. 初始化 PIXI
      const app = new PIXI.Application();
      await app.init({
        width: 600,
        height: 200,
        backgroundColor: 0x222222
      });
      document.body.appendChild(app.canvas);

      // 2. 加载gif文件
      const gif = await PIXI.Assets.load('./img/demo.gif');
      let gifHeight = gif.height;
      let gifWidth = gif.width;

      // 2. 获取所有纹理
      const frames = gif.frames;
      for (let i = 0; i < frames.length; i++) {
        const texture = frames[i].texture;
        const sprite = new PIXI.Sprite({
          texture,
          x: i * 120,
          y: 50,
          scale: (120 / gifWidth),
        });
        app.stage.addChild(sprite);
      }

      // 开始随机下载
      window.doDownload = async function () {
        const randomIndex = Math.floor(Math.random() * frames.length);
        const randomFrame = frames[randomIndex];

        // 创建临时精灵，使用原始纹理大小
        const tempSprite = new PIXI.Sprite(randomFrame);
        tempSprite.x = -1000; // 移出可视区域
        tempSprite.y = -1000;
        app.stage.addChild(tempSprite);

        // 下载原始大小的图片
        app.renderer.extract.download({
            target: tempSprite,
            filename: `${Date.now()}-${randomIndex}.png`
        });

        // 移除临时精灵
        app.stage.removeChild(tempSprite);
        tempSprite.destroy(); // 销毁自身
      };
    })();
  </script>
</body>

</html>

2.2.2 基础拉伸

  1. 代码示例

    <!DOCTYPE html>
<html lang="zh-CN">

<head>
  <meta charset="UTF-8">
  <title>PixiJS 8 图片重复填充 + 拉伸</title>
  <script src="./js/pixi.js"></script>
  <script src="./js/packages/gif.js"></script>
</head>

<body>
  <script>
    (async () => {
      // 1. 初始化 Pixi8
      const app = new PIXI.Application();
      await app.init({
        width: 800,
        height: 200,
        background: '#111',
      });
      document.body.appendChild(app.canvas);

      // 2. 新建容器
      const container = new PIXI.Container();
      app.stage.addChild(container);
      container.y = 50;
      // 初始宽度 & 拉伸状态
      let maxWidth = app.screen.width - 200;
      let isResizing = false;
      // 添加背景
      const graphics = new PIXI.Graphics();
      container.addChild(graphics);

      // 2. 最重要，将gif图转为纹理全部放到容器里
      const texture = await PIXI.Assets.load('./img/2.gif');
      let gifWidth = texture.width;
      let gifHeight = texture.height;

      const tempContainer = new PIXI.Container();
      const frames = texture.frames;

      // 临时帧数 如果帧数过多进行采样
      let imageSize = 5;
      let iamgeGap = 1;
      if (frames.length <= imageSize) {
        imageSize = frames.length;
      } else {
        iamgeGap = Math.floor(frames.length / imageSize);
      }

      for (let i = 0; i < imageSize; i++) {
        const texture = frames[i * iamgeGap].texture;
        const sprite = new PIXI.Sprite({
          texture,
          x: i * gifWidth,
          y: 0,
        });
        tempContainer.addChild(sprite);
      }


      // 4. 将容器渲染成新的纹理
      const mergedTexture = PIXI.RenderTexture.create({
        width: gifWidth * imageSize,       // 合成图宽
        height: gifHeight,      // 合成图高
      });
      mergedTexture.source.scaleMode = 'nearest'; // 尽量保持高清
      // 渲染容器到纹理
      app.renderer.render({
        container: tempContainer,  // 要渲染的容器
        target: mergedTexture,     // 目标纹理
        clear: true,               // 清空再画
      });


      // 5. 验证纹理是否渲染成功
      const tempSprite = new PIXI.Sprite(mergedTexture);
      tempSprite.x = 0;
      tempSprite.y = 300;
      tempSprite.scale.set(0.2);
      app.stage.addChild(tempSprite);


      // 6. 创建 TilingSprite 并添加到容器中
      let targetHeight = 60;
      const tilingSprite = new PIXI.TilingSprite(mergedTexture, maxWidth, targetHeight);
      tilingSprite.y = (100 - targetHeight) / 2;
      container.addChild(tilingSprite);

      // 缩放到目标大小 高度时固定了这里计算高度
      const textureScale = targetHeight / texture.height;
      tilingSprite.tileScale.set(textureScale);

      // 7. 创建拉伸手柄并完成拉伸功能
      const handle = new PIXI.Graphics();
      container.addChild(handle);
      handle.eventMode = 'static';
      handle.cursor = 'ew-resize';

      // 初始化渲染
      update();

      // 拖拽开始
      handle.onpointerdown = () => {
        isResizing = true;
      };

      // 拖拽结束（绑定到 canvas）
      app.canvas.addEventListener('pointerup', () => {
        isResizing = false;
      });

      // 拖拽移动（绑定到 canvas）
      app.canvas.addEventListener('pointermove', (e) => {
        if (!isResizing) return;

        // 把鼠标坐标转成容器内的局部坐标
        const rect = app.canvas.getBoundingClientRect();
        const global = new PIXI.Point(
          e.clientX - rect.left,
          e.clientY - rect.top
        );
        const localX = container.toLocal(global).x;

        // 更新宽度（限制最小100px）
        maxWidth = Math.max(100, localX);
        update();
      });

      // 更新视图
      function update() {
        // 1. 更新背景
        graphics.clear();
        graphics.rect(0, 0, maxWidth, 100);
        graphics.fill(0xc34288);

        // 2. 更新平铺宽度
        tilingSprite.width = maxWidth;

        // 2. 更新手柄位置（白色小方块）
        handle.clear();
        handle.rect(maxWidth - 10, 35, 10, 30);
        handle.fill(0xffffff);
      }
    })();
  </script>
</body>

</html>

  1. 代码示例

    <!DOCTYPE html>
<html lang="zh-CN">

<head>
  <meta charset="UTF-8">
  <title>PixiJS 8 GIF 平均抽帧 + 固定宽度 + 固定尺寸按钮</title>

  <script src="./js/packages/gif.js"></script>
  <style>

      display: flex;

    }
    .controls button {
      padding: 8px 16px;
      font-size: 16px;
      cursor: pointer;
      background: #4a90d9;
      color: white;
      border: none;
      border-radius: 4px;
    }
    .controls button:hover { background: #357abd; }
  </style>

<body>
  <div class="controls">
    <button data-width="200">200</button>
    <button data-width="400">400</button>
    <button data-width="600">600</button>
    <button data-width="800">800</button>
    <button data-width="1000">1000</button>
  </div>

  <script>
    (async () => {
      const app = new PIXI.Application();
      await app.init({
        width: 1200,
        height: 200,
        background: '#111',
      });
      document.body.insertBefore(app.canvas, document.body.firstChild);

      const container = new PIXI.Container();
      app.stage.addChild(container);
      container.y = 50;

      // 默认宽度
      let maxWidth = 400;
      const targetHeight = 60;
      const targetWidth = 120; // 单张图片固定宽度

      const graphics = new PIXI.Graphics();
      container.addChild(graphics);

      // 加载 GIF
      const texture = await PIXI.Assets.load('./img/demo.gif');
      const allFrames = texture.frames;
      const totalFrames = allFrames.length;

      // 视口 + 遮罩
      const viewport = new PIXI.Container();
      container.addChild(viewport);
      viewport.y = (100 - targetHeight) / 2;

      const maskGraphics = new PIXI.Graphics();
      viewport.addChild(maskGraphics);
      viewport.mask = maskGraphics;

      const markerLine = new PIXI.Graphics();
      container.addChild(markerLine);

      let gifSprites = [];

      // ======================================================
      // 核心：固定宽度 + 平均抽帧 + 无缝平铺
      // ======================================================
      function updateGifLayout() {
        gifSprites.forEach(s => s.destroy());
        gifSprites = [];

        const count = Math.ceil(maxWidth / targetWidth) + 1;

        // 平均抽帧
        const frameIndices = [];
        for (let i = 0; i < count; i++) {
          const idx = Math.round((i / (count - 1 || 1)) * (totalFrames - 1));
          frameIndices.push(idx);
        }

        // 平铺（不拉伸、不留间隙）
        for (let i = 0; i < count; i++) {
          const idx = frameIndices[i];
          const spr = new PIXI.Sprite(allFrames[idx].texture);
          spr.width = targetWidth;
          spr.height = targetHeight;
          spr.x = i * targetWidth;
          spr.y = 0;
          viewport.addChild(spr);
          gifSprites.push(spr);
        }
      }

      function updateView() {
        graphics.clear();
        maskGraphics.clear();
        markerLine.clear();

        // 背景
        graphics.rect(0, 0, app.screen.width, 100).fill(0xc34288);

        // 遮罩
        maskGraphics.rect(0, 0, maxWidth, targetHeight).fill(0xffffff);

        // 右侧线
        markerLine
          .moveTo(maxWidth, 0)
          .lineTo(maxWidth, 100)
          .stroke({ width: 2, color: 0x0000ff });

        updateGifLayout();
      }

      updateView();

      // ======================================================
      // 绑定固定宽度按钮
      // ======================================================
      const buttons = document.querySelectorAll('.controls button');
      buttons.forEach(btn => {
        btn.addEventListener('click', () => {
          maxWidth = parseInt(btn.dataset.width);
          updateView();
        });
      });

    })();
  </script>
</body>

</html>

      handle.eventMode = 'static';
      handle.cursor = 'ew-resize';
      handle.onpointerdown = () => { isResizing = true; };
      app.canvas.addEventListener('pointerup', () => { isResizing = false; });
      app.canvas.addEventListener('pointermove', (e) => {
        if (!isResizing) return;
        const rect = app.canvas.getBoundingClientRect();
        const local = container.toLocal(new PIXI.Point(e.clientX - rect.left, e.clientY -   rect.top));
        totalWidth = Math.max(segmentWidth, local.x);
        refresh();
      });

      // ================== 核心修复：切换宽度必重新抽帧 ==================
      function refresh() {
        allSprites.forEach(s => s.destroy());
        allSprites = [];
        bg.clear();
        lineGroup.clear();
        maskGraphics.clear();

        // 遮罩 + 背景
        maskGraphics.rect(0, 0, totalWidth, 100).fill(0xffffff);
        bg.rect(0, 0, totalWidth, 100).fill(0xc34288);

        // 蓝线
        for (let x = segmentWidth; x < totalWidth; x += segmentWidth) {
          lineGroup.moveTo(x, 0).lineTo(x, 100).stroke(0x0000ff, 1);
        }


        const totalImages = Math.ceil(totalWidth / itemWidth);
        const totalFrameIndexs = [];

        // 设置所帧的索引
        let startX = 0;
        for (let i = 0; i < totalImages; i++) {
          // 计算处于第几帧
          const frameIndex = Math.round((startX / segmentWidth) * (totalFrames-1));

          totalFrameIndexs.push(frameIndex);
          startX += itemWidth;
          if(startX > segmentWidth) {
            startX = startX - segmentWidth 
          }
        }

        // 设置所有精灵
        let currentX = 0;
        for (let i = 0; i < totalImages; i++) {
          // 获取当前帧的纹理
          const frameIndex = totalFrameIndexs[i];
          // 精灵添加
          const spr = new PIXI.Sprite(allFrames[frameIndex].texture);
          spr.width = itemWidth;
          spr.height = targetHeight;
          spr.x = currentX;
          spr.y = (100 - targetHeight) / 2;

          imageContainer.addChild(spr);
          allSprites.push(spr);
          currentX += itemWidth;
        }

        // 手柄
        handle.clear();
        handle.rect(totalWidth - 10, 35, 10, 30).fill(0xffffff);
      }

      refresh();
    })();
  </script>
