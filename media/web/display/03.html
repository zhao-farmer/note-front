3.1 视频纹理

3.1.1 基础使用

  1. 代码示例

    <!DOCTYPE html>
<html>
<head>
  <meta charset="UTF-8">
  <title>PIXI Video 基础使用</title>
</head>
<body>
  <script src="./js/pixi.js"></script>
  <script>
    (async () => {
      // 1. 初始化应用
      const app = new PIXI.Application();
      await app.init({
        width: 800,
        height: 600,
        backgroundColor: 0x222222,
        antialias: true
      });
      document.body.appendChild(app.canvas);

      // 2. 加载视频
      const videoTexture = await PIXI.Assets.load('./video/demo.mp4');
      // 这个video是HTML5的video元素
      const video = videoTexture.source.resource;

      // 3. 创建视频精灵
      const sprite = new PIXI.Sprite(videoTexture);
      sprite.anchor.set(0.5);
      sprite.x = app.screen.width / 2;
      sprite.y = app.screen.height / 2;
      sprite.scale.set(0.6);
      app.stage.addChild(sprite);

      // 3. 视频基础配置
      video.loop = true;          // 循环
      video.muted = true;        // 静音
      video.volume = 1;           // 音量
      video.playbackRate = 2;     // 播放速度
      video.autoplay = true;      // 自动播放

      // 5. 播放
      video.play();

    })();
  </script>
</body>
</html>

    <!DOCTYPE html>

  <meta charset="UTF-8">
  <title>Pixi 视频抽帧</title>
  <script src="./js/pixi.js"></script>

  <script>
    (async () => {
      // 1. 初始化 Pixi 应用
      const app = new PIXI.Application();
      await app.init({ width: 800, height: 600, backgroundColor: 0x222222 });
      document.body.appendChild(app.canvas);

      // 2. 加载视频
      const videoTexture = await PIXI.Assets.load('./video/demo.mp4');
      const video = videoTexture.source.resource;
      video.loop = true;
      video.muted = true;
      video.play();

      // 3. 创建视频精灵
      const sprite = new PIXI.Sprite(videoTexture);

      sprite.y = app.screen.height * 0.5;
      sprite.scale.set(0.5);
      app.stage.addChild(sprite);


      function captureFrame() {
        const w = video.videoWidth;
        const h = video.videoHeight;

        // 1. 创建临时容器 + 临时精灵（不修改原精灵）
        const tempContainer = new PIXI.Container();
        const tempSprite = new PIXI.Sprite(videoTexture);

        app.renderer.render(tempContainer, { renderTexture: frameTex });

        // 3. 清理临时对象
        tempSprite.destroy();
        tempContainer.destroy();

        console.log("成功截取一帧");
        return frameTex;
      }

      // 3.点击抽帧
      document.onclick = () => {
        const oneFrame = captureFrame();
        const frameSprite = new PIXI.Sprite(oneFrame);
        frameSprite.scale.set(0.15);
        frameSprite.x = 10;
        frameSprite.y = 10;
        app.stage.addChild(frameSprite);
      };

    })();
  </script>

3.1.4 抽帧列表展示

  1. 代码示例

    <!DOCTYPE html>
<html>

<head>
  <meta charset="UTF-8">
  <title>视频平均抽取18帧 + 排成一条线</title>
</head>

<body>
  <script src="./js/pixi.js"></script>
  <script>
    (async () => {
      // 1. 初始化应用
      const app = new PIXI.Application();
      await app.init({ width: 800, height: 300, backgroundColor: 0x222222 });
      document.body.appendChild(app.canvas);

      // 2. 加载视频
      const videoTexture = await PIXI.Assets.load('./video/demo.mp4');
      const video = videoTexture.source.resource;
      video.muted = true;

      // 3. 创建视频精灵（隐藏，不显示）
      const sprite = new PIXI.Sprite(videoTexture);
      sprite.alpha = 0;
      app.stage.addChild(sprite);

      // 抽帧（修复版：不改变位置）
      function captureFrame() {
        const w = video.videoWidth;
        const h = video.videoHeight;

        // 关键：用临时精灵渲染，不影响原精灵
        const tempContainer = new PIXI.Container();
        const tempSprite = new PIXI.Sprite(videoTexture);
        tempContainer.addChild(tempSprite);

        const frameTex = PIXI.RenderTexture.create({ width: w, height: h });
        app.renderer.render(tempContainer, { renderTexture: frameTex });

        tempSprite.destroy();
        tempContainer.destroy();
        return frameTex;
      }

      // 抽取帧的方法
      async function extractAverageFrames(total = 18) {
        const frames = [];
        const duration = video.duration;
        const step = duration / total;

        for (let i = 0; i < total; i++) {
          video.currentTime = i * step;

          await new Promise(resolve => {
            video.onseeked = resolve;
          });

          const frameTex = captureFrame();
          frames.push(frameTex);
          console.log(`已截取第 ${i + 1}/${total} 帧`);
        }

        console.log(` ${total}帧截取完成！`);
        return frames;
      }

      // 排成一条横线（整齐排列）
      function showFramesInLine(frames) {
        const startX = 15;       // 起始X
        const startY = 100;       // 高度Y（整条线统一）
        const scale = 0.08;      // 缩小倍数
        const spacing = 8;       // 图片间距

        frames.forEach((tex, i) => {
          const s = new PIXI.Sprite(tex);
          s.scale.set(scale);
          s.x = startX + i * (s.width + spacing);
          s.y = startY;
          app.stage.addChild(s);
        });
      }

      // 执行
      const t0 = performance.now();
      const frames = await extractAverageFrames(200);
      showFramesInLine(frames);
      const t1 = performance.now();
      const costSeconds = (t1 - t0) / 1000;
      console.log(`extractAndShowFrames 执行耗时：${costSeconds.toFixed(3)} 秒`);

    })();
  </script>
</body>

3.2.3 纹理复制解决

  1. 代码示例

    <!DOCTYPE html>
<html>

<head>
  <meta charset="UTF-8">
  <title>VideoFrame 抽帧优化版 (RenderTexture 高性能版)</title>
</head>

<body>
  <script src="./js/pixi.js"></script>

  <!-- 隐藏 video 标签 -->
  <video id="video" src="./video/demo.mp4" muted playsinline crossorigin="anonymous"
    style="position:absolute;top:-1000px;left:-1000px;"></video>

  <script>
    (async () => {
      // 1. 初始化 PIXI (v8)
      const app = new PIXI.Application();
      await app.init({
        width: 800,
        height: 300,
        backgroundColor: 0x222222,
        antialias: true
      });
      document.body.appendChild(app.canvas);

      const video = document.getElementById('video');
      await new Promise(resolve => { video.onloadedmetadata = resolve; });

      // 视频原始尺寸
      const videoWidth = video.videoWidth;
      const videoHeight = video.videoHeight;

      // 统一配置
      const SPRITE_CONFIG = {
        startX: 15,
        startY: 100,
        scale: 0.08,
        spacing: 8
      };

      // 计算缩略图最终在舞台上占据的宽度（用于计算横向排版）
      const thumbnailWidth = videoWidth * SPRITE_CONFIG.scale;

      // 【核心优化】使用 RenderTexture 离屏烘焙
      async function captureFrameToSprite(index) {
        // 1. 根据当前视频状态创建 VideoFrame
        const vf = new VideoFrame(video, {
          timestamp: video.currentTime * 1e6
        });

        // 2. 用当前帧的 VideoFrame 创建临时纹理与临时精灵
        const tempTexture = PIXI.Texture.from(vf);
        const tempSprite = new PIXI.Sprite(tempTexture);

        // 3. 创建一块等大、独立的离屏渲染纹理（RenderTexture）
        const renderTexture = PIXI.RenderTexture.create({
          width: videoWidth,
          height: videoHeight
        });

        // 3. 将临时精灵渲染到 RenderTexture 中（数据在 GPU 内部完成拷贝）
        app.renderer.render({
          container: tempSprite,
          target: renderTexture
        });

        // 5. 关键：解除引用并立即销毁临时纹理，同时关闭 VideoFrame 释放内存
        tempSprite.destroy();
        tempTexture.destroy(true); // 内部会解除对 vf 的显存占用
